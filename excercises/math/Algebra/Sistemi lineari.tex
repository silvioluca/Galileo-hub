\begin{question}[ID=1.1, subtitle={Sistemi lineari di due equazioni}, difficulty=facile, tags={sistemi, equazioni lineari}]
Risolvi i seguenti sistemi lineari di due equazioni:
\begin{multicols}{2}
\begin{enumerate}[leftmargin=3.2cm]
    \item
    \[
    \begin{cases}
    x+y=5\\
    x-y=1
    \end{cases}
    \]

    \item
    \[
    \begin{cases}
    2x+y=7\\
    x-y=2
    \end{cases}
    \]

    \item
    \[
    \begin{cases}
    x+2y=8\\
    x-y=2
    \end{cases}
    \]

    \item
    \[
    \begin{cases}
    3x+y=10\\
    x+y=6
    \end{cases}
    \]
\end{enumerate}
\end{multicols}
\begin{solution}
\begin{enumerate}
    \item $(x,y)=(3,2)$
    \item $(x,y)=(3,1)$
    \item $(x,y)=(4,2)$
    \item $(x,y)=(2,4)$
\end{enumerate}
\end{solution}
\begin{explanation}
\textbf{Metodo di riduzione}: quando una stessa incognita ha coefficienti opposti (o uguali) nelle due equazioni, sommando (o sottraendo) le equazioni membro a membro quell'incognita sparisce. Trovata una incognita, si sostituisce in una delle equazioni per trovare l'altra.
\begin{enumerate}
    \item La $y$ ha coefficienti opposti ($+1$ e $-1$): sommo le equazioni.
    \[\begin{aligned}
    (x+y)+(x-y) &= 5+1 && \textcolor{gray}{\text{sommo membro a membro}}\\
    2x &= 6\\
    x &= 3\\
    3+y &= 5 && \textcolor{gray}{\text{sostituisco nella prima}}\\
    y &= 2
    \end{aligned}\]
    \item La $y$ ha coefficienti opposti: sommo.
    \[\begin{aligned}
    3x &= 9 && \textcolor{gray}{\text{$(2x+y)+(x-y)=7+2$}}\\
    x &= 3\\
    3-y &= 2 && \textcolor{gray}{\text{sostituisco nella seconda}}\\
    y &= 1
    \end{aligned}\]
    \item La $x$ ha lo stesso coefficiente: sottraggo la seconda dalla prima.
    \[\begin{aligned}
    (x+2y)-(x-y) &= 8-2\\
    3y &= 6\\
    y &= 2\\
    x-2 &= 2 && \textcolor{gray}{\text{sostituisco nella seconda}}\\
    x &= 4
    \end{aligned}\]
    \item La $y$ ha lo stesso coefficiente: sottraggo.
    \[\begin{aligned}
    (3x+y)-(x+y) &= 10-6\\
    2x &= 4\\
    x &= 2\\
    2+y &= 6 && \textcolor{gray}{\text{sostituisco nella seconda}}\\
    y &= 4
    \end{aligned}\]
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=1.2, subtitle={Sistemi lineari con coefficienti interi}, difficulty=facile, tags={sistemi, equazioni lineari}]
Risolvi i seguenti sistemi lineari di due equazioni:
\begin{multicols}{2}
\begin{enumerate}[leftmargin=3.2cm]
    \item
    \[
    \begin{cases}
    2x+3y=12\\
    x-y=1
    \end{cases}
    \]

    \item
    \[
    \begin{cases}
    3x-2y=4\\
    x+y=3
    \end{cases}
    \]

    \item
    \[
    \begin{cases}
    4x+y=11\\
    2x-y=1
    \end{cases}
    \]

    \item
    \[
    \begin{cases}
    5x+2y=16\\
    x-2y=-4
    \end{cases}
    \]
\end{enumerate}
\end{multicols}
\begin{solution}
\begin{enumerate}
    \item $(x,y)=(3,2)$
    \item $(x,y)=(2,1)$
    \item $(x,y)=(2,3)$
    \item $(x,y)=(2,3)$
\end{enumerate}
\end{solution}
\begin{explanation}
\textbf{Metodo di sostituzione}: si ricava un'incognita da un'equazione (conviene quella con coefficiente $1$) e la si sostituisce nell'altra, che diventa un'equazione in una sola incognita.
\begin{enumerate}
    \item Dalla seconda ricavo $x=y+1$ e sostituisco nella prima:
    \[\begin{aligned}
    2(y+1)+3y &= 12\\
    2y+2+3y &= 12 && \textcolor{gray}{\text{svolgo la parentesi}}\\
    5y &= 10\\
    y &= 2\\
    x &= 2+1=3 && \textcolor{gray}{\text{torno a $x=y+1$}}
    \end{aligned}\]
    \item Dalla seconda ricavo $x=3-y$:
    \[\begin{aligned}
    3(3-y)-2y &= 4\\
    9-3y-2y &= 4\\
    -5y &= -5\\
    y &= 1\\
    x &= 3-1=2
    \end{aligned}\]
    \item La $y$ ha coefficienti opposti: conviene la riduzione (sommo).
    \[\begin{aligned}
    6x &= 12 && \textcolor{gray}{\text{$(4x+y)+(2x-y)=11+1$}}\\
    x &= 2\\
    y &= 11-4\cdot2=3 && \textcolor{gray}{\text{dalla prima}}
    \end{aligned}\]
    \item La $y$ ha coefficienti opposti ($+2y$ e $-2y$): sommo.
    \[\begin{aligned}
    6x &= 12 && \textcolor{gray}{\text{$(5x+2y)+(x-2y)=16-4$}}\\
    x &= 2\\
    2-2y &= -4 && \textcolor{gray}{\text{sostituisco nella seconda}}\\
    y &= 3
    \end{aligned}\]
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=1.3, subtitle={Sistemi lineari con parentesi e frazioni}, difficulty=media, tags={sistemi, equazioni lineari}]
Risolvi i seguenti sistemi lineari di due equazioni:
\begin{multicols}{2}
\begin{enumerate}[leftmargin=3.2cm]
    \item
    \[
    \begin{cases}
    2(x+y)=10\\
    3x-y=5
    \end{cases}
    \]

    \item
    \[
    \begin{cases}
    \dfrac{x}{2}+\dfrac{y}{3}=3\\
    x-y=1
    \end{cases}
    \]

    \item
    \[
    \begin{cases}
    3(x-1)+2y=7\\
    x+y=3
    \end{cases}
    \]

    \item
    \[
    \begin{cases}
    \dfrac{2x-1}{3}+y=4\\
    x-\dfrac{y+1}{2}=2
    \end{cases}
    \]
\end{enumerate}
\end{multicols}
\begin{solution}
\begin{enumerate}
    \item $(x,y)=\left(\tfrac52,\tfrac52\right)$
    \item $(x,y)=(4,3)$
    \item $(x,y)=(4,-1)$
    \item $(x,y)=\left(\tfrac72,2\right)$
\end{enumerate}
\end{solution}
\begin{explanation}
Prima si porta ogni equazione nella \textbf{forma normale} $ax+by=c$ (svolgendo le parentesi ed eliminando i denominatori), poi si risolve.
\begin{enumerate}
    \item Divido la prima equazione per $2$: diventa $x+y=5$. La $y$ ha coefficienti opposti nelle due equazioni: sommo.
    \[\begin{aligned}
    (x+y)+(3x-y) &= 5+5\\
    4x &= 10\\
    x &= \dfrac52\\
    y &= 5-\dfrac52=\dfrac52 && \textcolor{gray}{\text{da $x+y=5$}}
    \end{aligned}\]
    \item Moltiplico la prima per il m.c.m. $6$: $3x+2y=18$. Dalla seconda: $x=y+1$.
    \[\begin{aligned}
    3(y+1)+2y &= 18 && \textcolor{gray}{\text{sostituisco}}\\
    5y+3 &= 18\\
    y &= 3\\
    x &= 4
    \end{aligned}\]
    \item Prima equazione: $3x-3+2y=7$, cioè $3x+2y=10$. Dalla seconda: $x=3-y$.
    \[\begin{aligned}
    3(3-y)+2y &= 10 && \textcolor{gray}{\text{sostituisco}}\\
    9-y &= 10\\
    y &= -1\\
    x &= 3-(-1)=4
    \end{aligned}\]
    \item Elimino i denominatori:
    \[\begin{aligned}
    2x-1+3y &= 12 && \textcolor{gray}{\text{prima equazione per $3$}}\\
    2x+3y &= 13\\
    2x-(y+1) &= 4 && \textcolor{gray}{\text{seconda equazione per $2$}}\\
    2x-y &= 5
    \end{aligned}\]
    La $x$ ha lo stesso coefficiente: sottraggo la seconda dalla prima.
    \[\begin{aligned}
    (2x+3y)-(2x-y) &= 13-5\\
    4y &= 8\\
    y &= 2\\
    2x &= 5+2 \;\Rightarrow\; x=\dfrac72 && \textcolor{gray}{\text{da $2x-y=5$}}
    \end{aligned}\]
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=1.4, subtitle={Sistemi lineari con parametro numerico e coefficienti frazionari}, difficulty=difficile, tags={sistemi, equazioni lineari}]
Risolvi i seguenti sistemi lineari di due equazioni:
\begin{multicols}{2}
\begin{enumerate}[leftmargin=3.2cm]
    \item
    \[
    \begin{cases}
    4x-3y=1\\
    5x+2y=19
    \end{cases}
    \]

    \item
    \[
    \begin{cases}
    \dfrac{3x+y}{2}=5\\
    \dfrac{x-y}{3}=1
    \end{cases}
    \]

    \item
    \[
    \begin{cases}
    2(x-2)-3(y+1)=4\\
    5x+2y=13
    \end{cases}
    \]

    \item
    \[
    \begin{cases}
    \dfrac{2x-3y}{4}=2\\
    \dfrac{x+y}{3}=2
    \end{cases}
    \]
\end{enumerate}
\end{multicols}
\begin{solution}
\begin{enumerate}
    \item $(x,y)=\left(\tfrac{59}{23},\tfrac{71}{23}\right)$
    \item $(x,y)=\left(\tfrac{13}{4},\tfrac14\right)$
    \item $(x,y)=\left(\tfrac{61}{19},-\tfrac{29}{19}\right)$
    \item $(x,y)=\left(\tfrac{26}{5},\tfrac45\right)$
\end{enumerate}
\end{solution}
\begin{explanation}
\begin{enumerate}
    \item Nessun coefficiente vale $1$: uso la riduzione. Per eliminare la $y$ rendo i suoi coefficienti opposti ($-6y$ e $+6y$):
    \[\begin{aligned}
    8x-6y &= 2 && \textcolor{gray}{\text{prima equazione per $2$}}\\
    15x+6y &= 57 && \textcolor{gray}{\text{seconda equazione per $3$}}\\
    23x &= 59 && \textcolor{gray}{\text{sommo}}\\
    x &= \dfrac{59}{23}
    \end{aligned}\]
    Ricavo $y$ dalla prima equazione, $3y=4x-1$:
    \[ 3y=\dfrac{236}{23}-\dfrac{23}{23}=\dfrac{213}{23} \;\Rightarrow\; y=\dfrac{71}{23} \]
    \item Elimino i denominatori: $3x+y=10$ (per $2$) e $x-y=3$ (per $3$). La $y$ ha coefficienti opposti: sommo.
    \[\begin{aligned}
    4x &= 13\\
    x &= \dfrac{13}{4}\\
    y &= x-3=\dfrac{13}{4}-\dfrac{12}{4}=\dfrac14
    \end{aligned}\]
    \item Prima equazione in forma normale: $2x-4-3y-3=4$, cioè $2x-3y=11$. Riduzione sulla $y$:
    \[\begin{aligned}
    4x-6y &= 22 && \textcolor{gray}{\text{prima per $2$}}\\
    15x+6y &= 39 && \textcolor{gray}{\text{seconda per $3$}}\\
    19x &= 61 && \textcolor{gray}{\text{sommo}}\\
    x &= \dfrac{61}{19}
    \end{aligned}\]
    Da $3y=2x-11$: \ $3y=\tfrac{122}{19}-\tfrac{209}{19}=-\tfrac{87}{19} \;\Rightarrow\; y=-\tfrac{29}{19}$.
    \item Elimino i denominatori: $2x-3y=8$ (per $4$) e $x+y=6$ (per $3$). Dalla seconda: $x=6-y$.
    \[\begin{aligned}
    2(6-y)-3y &= 8 && \textcolor{gray}{\text{sostituisco}}\\
    12-5y &= 8\\
    y &= \dfrac45\\
    x &= 6-\dfrac45=\dfrac{26}{5}
    \end{aligned}\]
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=1.5, subtitle={Sistemi lineari: casi particolari}, difficulty=difficile, tags={sistemi, equazioni lineari}]
Risolvi e classifica i seguenti sistemi lineari di due equazioni:
\begin{multicols}{2}
\begin{enumerate}[leftmargin=3.2cm]
    \item
    \[
    \begin{cases}
    2x+4y=6\\
    x+2y=3
    \end{cases}
    \]

    \item
    \[
    \begin{cases}
    3x-6y=9\\
    x-2y=1
    \end{cases}
    \]

    \item
    \[
    \begin{cases}
    4x+2y=8\\
    2x+y=5
    \end{cases}
    \]

    \item
    \[
    \begin{cases}
    2x-3y=7\\
    4x-6y=14
    \end{cases}
    \]
\end{enumerate}
\end{multicols}
\begin{solution}
\begin{enumerate}
    \item indeterminato: infinite soluzioni $(3-2t,\ t)$
    \item impossibile
    \item impossibile
    \item indeterminato: infinite soluzioni $\left(\tfrac{7+3t}{2},\ t\right)$
\end{enumerate}
\end{solution}
\begin{explanation}
Per classificare un sistema $\begin{cases} ax+by=c\\ a'x+b'y=c'\end{cases}$ si confrontano i rapporti tra i coefficienti:
\begin{itemize}
    \item se $\tfrac{a}{a'}\neq\tfrac{b}{b'}$ il sistema è \textbf{determinato} (una soluzione: le rette si incontrano);
    \item se $\tfrac{a}{a'}=\tfrac{b}{b'}=\tfrac{c}{c'}$ è \textbf{indeterminato} (le equazioni sono equivalenti: rette coincidenti);
    \item se $\tfrac{a}{a'}=\tfrac{b}{b'}\neq\tfrac{c}{c'}$ è \textbf{impossibile} (rette parallele distinte).
\end{itemize}
\begin{enumerate}
    \item $\tfrac21=\tfrac42=\tfrac63=2$: la prima equazione è il doppio della seconda, quindi è un'unica equazione $x+2y=3$. \\
    Ogni coppia con $y=t$ qualsiasi e $x=3-2t$ è soluzione: indeterminato.
    \item Divido la prima per $3$: diventa $x-2y=3$, mentre la seconda dice $x-2y=1$. La stessa espressione non può valere sia $3$ sia $1$: impossibile.
    \item Divido la prima per $2$: $2x+y=4$, mentre la seconda dice $2x+y=5$: impossibile.
    \item $\tfrac24=\tfrac{-3}{-6}=\tfrac7{14}=\tfrac12$: la seconda è il doppio della prima. \\
    Rimane $2x-3y=7$: posto $y=t$, si ha $x=\tfrac{7+3t}{2}$. Indeterminato.
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=2.1, subtitle={Sistemi con coefficienti frazionari}, difficulty=facile, tags={sistemi, equazioni lineari, frazioni}]
Risolvi i seguenti sistemi lineari di due equazioni:
\begin{multicols}{2}
\begin{enumerate}[leftmargin=3.2cm]
    \item
    \[
    \begin{cases}
    \dfrac{1}{2}x+\dfrac{1}{3}y=3\\
    x-y=1
    \end{cases}
    \]

    \item
    \[
    \begin{cases}
    \dfrac{1}{3}x+\dfrac{1}{2}y=3\\
    2x-y=2
    \end{cases}
    \]

    \item
    \[
    \begin{cases}
    \dfrac{1}{4}x+y=4\\
    \dfrac{1}{2}x-\dfrac{1}{3}y=1
    \end{cases}
    \]

    \item
    \[
    \begin{cases}
    \dfrac{1}{2}x+\dfrac{1}{5}y=4\\
    x-\dfrac{1}{2}y=-1
    \end{cases}
    \]
\end{enumerate}
\end{multicols}
\begin{solution}
\begin{enumerate}
    \item $(x,y)=(4,3)$
    \item $(x,y)=(3,4)$
    \item $(x,y)=(4,3)$
    \item $(x,y)=(4,10)$
\end{enumerate}
\end{solution}
\begin{explanation}
Si moltiplica ogni equazione per il m.c.m. dei suoi denominatori, così i coefficienti diventano interi; poi si risolve per sostituzione o riduzione.
\begin{enumerate}
    \item Prima equazione per $6$: $3x+2y=18$. Dalla seconda: $x=y+1$.
    \[\begin{aligned}
    3(y+1)+2y &= 18 && \textcolor{gray}{\text{sostituisco}}\\
    5y &= 15\\
    y &= 3, \qquad x=4
    \end{aligned}\]
    \item Prima equazione per $6$: $2x+3y=18$. Dalla seconda: $y=2x-2$.
    \[\begin{aligned}
    2x+3(2x-2) &= 18 && \textcolor{gray}{\text{sostituisco}}\\
    8x-6 &= 18\\
    x &= 3, \qquad y=4
    \end{aligned}\]
    \item Prima per $4$: $x+4y=16$; \ seconda per $6$: $3x-2y=6$. Dalla prima: $x=16-4y$.
    \[\begin{aligned}
    3(16-4y)-2y &= 6 && \textcolor{gray}{\text{sostituisco}}\\
    48-14y &= 6\\
    y &= 3, \qquad x=16-12=4
    \end{aligned}\]
    \item Prima per $10$: $5x+2y=40$; \ seconda per $2$: $2x-y=-2$, da cui $y=2x+2$.
    \[\begin{aligned}
    5x+2(2x+2) &= 40 && \textcolor{gray}{\text{sostituisco}}\\
    9x+4 &= 40\\
    x &= 4, \qquad y=10
    \end{aligned}\]
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=2.2, subtitle={Sistemi con frazioni e numeratori letterali}, difficulty=media, tags={sistemi, equazioni lineari, frazioni}]
Risolvi i seguenti sistemi lineari di due equazioni:
\begin{multicols}{2}
\begin{enumerate}[leftmargin=3.2cm]
    \item
    \[
    \begin{cases}
    \dfrac{x+1}{2}+\dfrac{1}{3}y=3\\
    x-\dfrac{1}{3}y=2
    \end{cases}
    \]

    \item
    \[
    \begin{cases}
    \dfrac{1}{4}x+\dfrac{y-1}{3}=2\\
    x-\dfrac{1}{2}y=2
    \end{cases}
    \]

    \item
    \[
    \begin{cases}
    \dfrac{x-2}{3}+\dfrac{1}{2}y=3\\
    2x-y=6
    \end{cases}
    \]

    \item
    \[
    \begin{cases}
    \dfrac{1}{2}x+\dfrac{y+1}{4}=3\\
    \dfrac{x-1}{3}+y=4
    \end{cases}
    \]
\end{enumerate}
\end{multicols}
\begin{solution}
\begin{enumerate}
    \item $(x,y)=(3,3)$
    \item $(x,y)=(4,4)$
    \item $(x,y)=(5,4)$
    \item $(x,y)=(4,3)$
\end{enumerate}
\end{solution}
\begin{explanation}
\begin{enumerate}
    \item Elimino i denominatori:
    \[\begin{aligned}
    3(x+1)+2y &= 18 \;\Rightarrow\; 3x+2y=15 && \textcolor{gray}{\text{prima per $6$}}\\
    3x-y &= 6 && \textcolor{gray}{\text{seconda per $3$}}
    \end{aligned}\]
    La $x$ ha lo stesso coefficiente: sottraggo la seconda dalla prima.
    \[ 3y=9 \;\Rightarrow\; y=3, \qquad 3x=6+3 \;\Rightarrow\; x=3 \]
    \item Prima per $12$: $3x+4(y-1)=24$, cioè $3x+4y=28$; \ seconda per $2$: $2x-y=4$, da cui $y=2x-4$.
    \[\begin{aligned}
    3x+4(2x-4) &= 28 && \textcolor{gray}{\text{sostituisco}}\\
    11x-16 &= 28\\
    x &= 4, \qquad y=4
    \end{aligned}\]
    \item Prima per $6$: $2(x-2)+3y=18$, cioè $2x+3y=22$; \ dalla seconda $y=2x-6$.
    \[\begin{aligned}
    2x+3(2x-6) &= 22\\
    8x &= 40\\
    x &= 5, \qquad y=4
    \end{aligned}\]
    \item Prima per $4$: $2x+y+1=12$, cioè $2x+y=11$; \ seconda per $3$: $x-1+3y=12$, cioè $x+3y=13$. \\
    Dalla prima $y=11-2x$:
    \[\begin{aligned}
    x+3(11-2x) &= 13\\
    -5x+33 &= 13\\
    x &= 4, \qquad y=3
    \end{aligned}\]
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=2.3, subtitle={Sistemi con frazioni in entrambe le equazioni}, difficulty=media, tags={sistemi, equazioni lineari, frazioni}]
Risolvi i seguenti sistemi lineari di due equazioni:
\begin{multicols}{2}
\begin{enumerate}[leftmargin=3.2cm]
    \item
    \[
    \begin{cases}
    \dfrac{2x-1}{3}+\dfrac{y+1}{2}=3\\
    \dfrac{1}{2}x-\dfrac{1}{3}y=0
    \end{cases}
    \]

    \item
    \[
    \begin{cases}
    \dfrac{x+y}{2}+\dfrac{1}{3}x=4\\
    x-\dfrac{1}{3}y=2
    \end{cases}
    \]

    \item
    \[
    \begin{cases}
    \dfrac{3x-y}{4}+\dfrac{1}{2}x=3\\
    \dfrac{1}{2}x-\dfrac{1}{4}y=0
    \end{cases}
    \]

    \item
    \[
    \begin{cases}
    \dfrac{1}{2}x-\dfrac{y-1}{3}=1\\
    \dfrac{x+2}{3}-\dfrac{1}{4}y=1
    \end{cases}
    \]
\end{enumerate}
\end{multicols}
\begin{solution}
\begin{enumerate}
    \item $(x,y)=(2,3)$
    \item $(x,y)=(3,3)$
    \item $(x,y)=(4,8)$
    \item $(x,y)=(4,4)$
\end{enumerate}
\end{solution}
\begin{explanation}
\begin{enumerate}
    \item Prima per $6$: $2(2x-1)+3(y+1)=18$, cioè $4x+3y=17$; \ seconda per $6$: $3x-2y=0$, da cui $y=\tfrac32x$.
    \[\begin{aligned}
    4x+3\cdot\dfrac32x &= 17 && \textcolor{gray}{\text{sostituisco}}\\
    \dfrac{17}{2}x &= 17\\
    x &= 2, \qquad y=3
    \end{aligned}\]
    \item Prima per $6$: $3(x+y)+2x=24$, cioè $5x+3y=24$; \ seconda per $3$: $3x-y=6$, da cui $y=3x-6$.
    \[\begin{aligned}
    5x+3(3x-6) &= 24\\
    14x &= 42\\
    x &= 3, \qquad y=3
    \end{aligned}\]
    \item Prima per $4$: $3x-y+2x=12$, cioè $5x-y=12$; \ seconda per $4$: $2x-y=0$, da cui $y=2x$.
    \[\begin{aligned}
    5x-2x &= 12 && \textcolor{gray}{\text{sostituisco}}\\
    x &= 4, \qquad y=8
    \end{aligned}\]
    \item Prima per $6$: $3x-2(y-1)=6$, cioè $3x-2y=4$; \ seconda per $12$: $4(x+2)-3y=12$, cioè $4x-3y=4$. \\
    Riduzione sulla $y$:
    \[\begin{aligned}
    9x-6y &= 12 && \textcolor{gray}{\text{prima per $3$}}\\
    8x-6y &= 8 && \textcolor{gray}{\text{seconda per $2$}}\\
    x &= 4 && \textcolor{gray}{\text{sottraggo}}\\
    2y &= 3\cdot4-4=8 \;\Rightarrow\; y=4
    \end{aligned}\]
\end{enumerate}
\end{explanation}
\end{question}

\begin{enumerate}[label=\alph*),leftmargin=1.8cm]
    \item $S_A(x)=5+0{,}10x$, \ $S_B(x)=20+0{,}04x$
    \item A $250$ minuti, con una spesa di $30$ euro: $P(250;\,30)$
    \item Sotto i $250$ minuti conviene il piano A, sopra i $250$ minuti il piano B.
\begin{center}
\begin{tikzpicture}
\begin{axis}[
    width=10cm,
    height=6cm,
    xmin=0, xmax=500,
    ymin=0, ymax=62,
    axis lines=left,

]
\addplot[red,thick,domain=0:500] {5+0.1*x};

\addplot[blue,thick,domain=0:500] {20+0.04*x};

\end{axis}

\end{center}
\end{enumerate}
\end{solution}
\begin{explanation}
\begin{enumerate}[label=\alph*),leftmargin=1.8cm]
    \item Ogni piano è canone fisso più costo al minuto per i minuti usati:
    \[ S_A(x)=5+0{,}10x, \qquad S_B(x)=20+0{,}04x \]

\begin{question}[ID=4.4, subtitle={Noleggio auto: quale agenzia conviene}, difficulty=difficile, tags={sistemi, break-even, grafici}]
Due agenzie noleggiano la stessa auto. L'agenzia Alfa chiede $50$ euro di quota fissa più $0{,}20$ euro per ogni chilometro percorso. L'agenzia Beta chiede $20$ euro di quota fissa più $0{,}30$ euro per ogni chilometro.
\begin{enumerate}[label=\alph*),leftmargin=1.8cm]
    \item Scrivi le due funzioni di costo in funzione dei chilometri $x$ percorsi.
    \item Dopo quanti chilometri i due costi coincidono? Quanto si spende in quel caso?
    \item Traccia i grafici e spiega, usando il punto di intersezione, per quali percorrenze conviene ciascuna agenzia.
    \item Un cliente deve percorrere $450\,\text{km}$. Quale agenzia sceglie e quanto risparmia rispetto all'altra?
\end{enumerate}
\begin{solution}
\begin{enumerate}[label=\alph*),leftmargin=1.8cm]
    \item $C_A(x)=50+0{,}2x$, \ $C_B(x)=20+0{,}3x$
    \item A $300\,\text{km}$, con una spesa di $110$ euro: $P(300;\,110)$
    \item Sotto i $300\,\text{km}$ conviene Beta, sopra i $300\,\text{km}$ conviene Alfa.
\begin{center}
\begin{tikzpicture}
\begin{axis}[
    width=10cm,
    height=6cm,
    xmin=0, xmax=500,
    ymin=0, ymax=190,
    axis lines=left,
    xlabel={$x$ (km)},
    ylabel={$y$ (euro)},
    grid=both,
]
\addplot[red,thick,domain=0:500] {50+0.2*x};
\node[red] at (axis cs:440,130) {Alfa};
\addplot[blue,thick,domain=0:500] {20+0.3*x};
\node[blue] at (axis cs:440,165) {Beta};
\node[black] at (axis cs:300,110) {● P (300; 110)};
\end{axis}
\end{tikzpicture}
\end{center}
    \item Sceglie Alfa ($140$ euro contro $155$): risparmia $15$ euro.
\end{enumerate}
\end{solution}
\begin{explanation}
\begin{enumerate}[label=\alph*),leftmargin=1.8cm]
    \item Quota fissa più costo per chilometro:
    \[ C_A(x)=50+0{,}2x, \qquad C_B(x)=20+0{,}3x \]
    \item Uguaglio i due costi:
    \[\begin{aligned}
    50+0{,}2x &= 20+0{,}3x\\
    -0{,}1x &= -30\\
    x &= 300\\
    y &= 50+0{,}2\cdot300=110
    \end{aligned}\]
    \item Beta parte più in basso (quota fissa $20$) ma cresce più rapidamente (pendenza $0{,}3$): fino a $300$ km la sua retta sta sotto quella di Alfa, poi sopra. Conviene sempre l'agenzia la cui retta sta più in basso.
    \item $450>300$, quindi conviene Alfa. Calcolo i due costi:
    \[\begin{aligned}
    C_A(450) &= 50+0{,}2\cdot450=140 \text{ euro}\\
    C_B(450) &= 20+0{,}3\cdot450=155 \text{ euro}
    \end{aligned}\]
    Risparmio: $155-140=15$ euro.
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=4.5, subtitle={Break-even e utile desiderato}, difficulty=difficile, tags={sistemi, break-even, grafici}]
Un laboratorio artigianale produce zaini. Ogni mese sostiene costi fissi di $1\,200$ euro e un costo di produzione di $8$ euro per zaino. Ogni zaino viene venduto a $14$ euro.
\begin{enumerate}[label=\alph*),leftmargin=1.8cm]
    \item Determina il break-even point e rappresentalo graficamente come intersezione delle rette dei costi e dei ricavi.
    \item Quanti zaini deve vendere il laboratorio per ottenere un utile mensile di $900$ euro?
    \item Il prezzo di vendita sale a $16$ euro. Come cambia il break-even point? Come cambia la retta dei ricavi nel grafico?
\end{enumerate}
\begin{solution}
\begin{enumerate}[label=\alph*),leftmargin=1.8cm]
    \item Break-even point $B(200;\,2800)$.
\begin{center}
\begin{tikzpicture}
\begin{axis}[
    width=10cm,
    height=6cm,
    xmin=0, xmax=300,
    ymin=0, ymax=4500,
    axis lines=left,
    xlabel={$x$},
    ylabel={$y$ (euro)},
    grid=both,
]
\addplot[red,thick,domain=0:300] {1200+8*x};
\node[red] at (axis cs:260,3050) {C};
\addplot[blue,thick,domain=0:300] {14*x};
\node[blue] at (axis cs:260,3800) {R};
\node[black] at (axis cs:200,2800) {● B (200; 2800)};
\end{axis}
\end{tikzpicture}
\end{center}
    \item $350$ zaini
    \item Il break-even scende a $B'(150;\,2400)$: la retta dei ricavi diventa più inclinata e incontra prima quella dei costi.
\end{enumerate}
\end{solution}
\begin{explanation}
\begin{enumerate}[label=\alph*),leftmargin=1.8cm]
    \item Costi: $C(x)=1200+8x$; \ ricavi: $R(x)=14x$. Uguaglio:
    \[\begin{aligned}
    14x &= 1200+8x\\
    6x &= 1200\\
    x &= 200\\
    y &= 14\cdot200=2800
    \end{aligned}\]
    Nel grafico la retta dei costi parte da $(0;\,1200)$, quella dei ricavi dall'origine; si incontrano in $B$.
    \item L'utile è $U(x)=R(x)-C(x)=14x-1200-8x=6x-1200$. Impongo che valga $900$:
    \[\begin{aligned}
    6x-1200 &= 900\\
    6x &= 2100\\
    x &= 350
    \end{aligned}\]
    Graficamente è il punto in cui la retta dei ricavi sta $900$ euro sopra quella dei costi.
    \item Con il nuovo prezzo $R(x)=16x$:
    \[\begin{aligned}
    16x &= 1200+8x\\
    8x &= 1200\\
    x &= 150, \qquad y=16\cdot150=2400
    \end{aligned}\]
    La retta dei costi non cambia; quella dei ricavi diventa più ripida (pendenza da $14$ a $16$) e quindi la incontra prima: bastano $150$ zaini invece di $200$.
\end{enumerate}
\end{explanation}
\end{question}
