\begin{question}[ID=1.1, subtitle={Teorema di Cauchy per $x^3+x$ e $x^2+1$ su $[0,2]$}, tags={cauchy, teoremi}, difficulty=media]
Verifica se le funzioni \(f(x) = x^3 + x\) e \(g(x) = x^2 + 1\) soddisfano le ipotesi del teorema di Cauchy nell'intervallo \([0,\ 2]\) e, in caso affermativo, trova il punto \(c\) che soddisfa il teorema.
\begin{solution}
Ipotesi soddisfatte; \ $c=\dfrac{5\pm\sqrt{13}}{6}$ \ (entrambi in $(0,2)$: $c\approx0{,}23$ e $c\approx1{,}43$).
\end{solution}
\begin{explanation}
\textbf{Teorema di Cauchy}: se $f$ e $g$ sono continue su $[a,b]$, derivabili in $(a,b)$ e $g'(x)\neq0$ in $(a,b)$, allora esiste $c\in(a,b)$ tale che
\[ \dfrac{f(b)-f(a)}{g(b)-g(a)}=\dfrac{f'(c)}{g'(c)}. \]
\textbf{Ipotesi.} $f$ e $g$ sono polinomi, quindi continue e derivabili ovunque. $g'(x)=2x\neq0$ per $x\in(0,2)$.

\textbf{Calcolo di $c$.}
\[\begin{aligned}
\dfrac{f(2)-f(0)}{g(2)-g(0)} &= \dfrac{10-0}{5-1}=\dfrac52\\
\dfrac{f'(c)}{g'(c)} &= \dfrac{3c^2+1}{2c} && \textcolor{gray}{\text{$f'=3x^2+1$, \ $g'=2x$}}\\
\dfrac{3c^2+1}{2c} &= \dfrac52\\
3c^2+1 &= 5c && \textcolor{gray}{\text{moltiplico per $2c$}}\\
3c^2-5c+1 &= 0\\
c &= \dfrac{5\pm\sqrt{25-12}}{6}=\dfrac{5\pm\sqrt{13}}{6}
\end{aligned}\]
Valori approssimati $0{,}23$ e $1{,}43$: entrambi appartengono a $(0,2)$.
\end{explanation}
\end{question}

\begin{question}[ID=1.2, subtitle={Teorema di Cauchy per $\sin(2x)$ e $\cos x$ su $[0,\pi/2]$}, tags={cauchy, teoremi}, difficulty=media]
Verifica se le funzioni \(f(x) = \sin(2x)\) e \(g(x) = \cos(x)\) soddisfano le ipotesi del teorema di Cauchy nell'intervallo \([0;\ \pi/2]\) e, in caso affermativo, trova il punto \(c\) che soddisfa il teorema.
\begin{solution}
Ipotesi soddisfatte; \ $c=\dfrac\pi4$.
\end{solution}
\begin{explanation}
\textbf{Teorema di Cauchy}: se $f$ e $g$ sono continue su $[a,b]$, derivabili in $(a,b)$ e $g'(x)\neq0$ in $(a,b)$, allora esiste $c\in(a,b)$ tale che
\[ \dfrac{f(b)-f(a)}{g(b)-g(a)}=\dfrac{f'(c)}{g'(c)}. \]
\textbf{Ipotesi.} Seno e coseno sono continui e derivabili ovunque. $g'(x)=-\sin x\neq0$ per $x\in\left(0,\tfrac\pi2\right)$, dove $\sin x>0$.

\textbf{Calcolo di $c$.}
\[\begin{aligned}
\dfrac{f(\frac\pi2)-f(0)}{g(\frac\pi2)-g(0)} &= \dfrac{\sin\pi-\sin0}{0-1}=\dfrac{0}{-1}=0\\
\dfrac{2\cos(2c)}{-\sin c} &= 0 && \textcolor{gray}{\text{$f'=2\cos2x$, \ $g'=-\sin x$}}\\
\cos(2c) &= 0 && \textcolor{gray}{\text{una frazione è nulla se lo è il numeratore}}\\
2c &= \dfrac\pi2 \;\Rightarrow\; c=\dfrac\pi4 && \textcolor{gray}{\text{unica soluzione con $c\in(0,\dfrac\pi2)$}}
\end{aligned}\]
\end{explanation}
\end{question}

\begin{question}[ID=1.3, subtitle={Teorema di Cauchy per $e^{2x}$ e $e^x$ su $[0,1]$}, tags={cauchy, teoremi}, difficulty=media]
Verifica se le funzioni \(f(x) = e^{2x}\) e \(g(x) = e^x\) soddisfano le ipotesi del teorema di Cauchy nell'intervallo \([0;\ 1]\) e, in caso affermativo, trova il punto \(c\) che soddisfa il teorema.
\begin{solution}
Ipotesi soddisfatte; \ $c=\ln\dfrac{e+1}{2}\approx0{,}62$.
\end{solution}
\begin{explanation}
\textbf{Teorema di Cauchy}: se $f$ e $g$ sono continue su $[a,b]$, derivabili in $(a,b)$ e $g'(x)\neq0$ in $(a,b)$, allora esiste $c\in(a,b)$ tale che
\[ \dfrac{f(b)-f(a)}{g(b)-g(a)}=\dfrac{f'(c)}{g'(c)}. \]
\textbf{Ipotesi.} Le esponenziali sono continue e derivabili ovunque; $g'(x)=e^x>0$ sempre.

\textbf{Calcolo di $c$.}
\[\begin{aligned}
\dfrac{e^2-e^0}{e^1-e^0} &= \dfrac{e^2-1}{e-1}=\dfrac{(e-1)(e+1)}{e-1}=e+1 && \textcolor{gray}{\text{differenza di quadrati}}\\
\dfrac{2e^{2c}}{e^c} &= 2e^c && \textcolor{gray}{\text{$f'=2e^{2x}$, \ $g'=e^x$}}\\
2e^c &= e+1\\
c &= \ln\dfrac{e+1}{2} && \textcolor{gray}{\text{logaritmo naturale di entrambi i membri}}
\end{aligned}\]
$\dfrac{e+1}{2}\approx1{,}86$, quindi $c\approx0{,}62\in(0,1)$.
\end{explanation}
\end{question}

\begin{question}[ID=1.4, subtitle={Teorema di Cauchy per $\ln x$ e $1/x$ su $[1,e]$}, tags={cauchy, teoremi}, difficulty=media]
Date le funzioni \(f(x) = \ln(x)\) e \(g(x) = \dfrac{1}{x}\), verifica se soddisfano le ipotesi del teorema di Cauchy nell'intervallo \([1;\ e]\) e, in caso affermativo, determina i punti che verificano il teorema.
\begin{solution}
Ipotesi soddisfatte; \ $c=\dfrac{e}{e-1}\approx1{,}58$.
\end{solution}
\begin{explanation}
\textbf{Teorema di Cauchy}: se $f$ e $g$ sono continue su $[a,b]$, derivabili in $(a,b)$ e $g'(x)\neq0$ in $(a,b)$, allora esiste $c\in(a,b)$ tale che
\[ \dfrac{f(b)-f(a)}{g(b)-g(a)}=\dfrac{f'(c)}{g'(c)}. \]
\textbf{Ipotesi.} Su $[1,e]$ entrambe sono continue e derivabili ($x>0$); $g'(x)=-\dfrac{1}{x^2}\neq0$.

\textbf{Calcolo di $c$.}
\[\begin{aligned}
\dfrac{\ln e-\ln1}{\frac1e-1} &= \dfrac{1}{\frac{1-e}{e}}=\dfrac{e}{1-e}=-\dfrac{e}{e-1}\\
\dfrac{f'(c)}{g'(c)} &= \dfrac{1/c}{-1/c^2}=-c && \textcolor{gray}{\text{$f'=\dfrac1x$, \ $g'=-\dfrac1{x^2}$}}\\
-c &= -\dfrac{e}{e-1} \;\Rightarrow\; c=\dfrac{e}{e-1}
\end{aligned}\]
$c\approx1{,}58$, compreso tra $1$ ed $e\approx2{,}72$.
\end{explanation}
\end{question}

\begin{question}[ID=1.5, subtitle={Teorema di Cauchy per $\arctan x$ e $x$ su $[0,1]$}, tags={cauchy, teoremi}, difficulty=media]
Date le funzioni \(f(x) = \arctan(x)\) e \(g(x) = x\), verifica se soddisfano le ipotesi del teorema di Cauchy nell'intervallo \([0;\ 1]\) e, in caso affermativo, determina i punti che verificano il teorema.
\begin{solution}
Ipotesi soddisfatte; \ $c=\sqrt{\dfrac{4-\pi}{\pi}}\approx0{,}52$.
\end{solution}
\begin{explanation}
\textbf{Teorema di Cauchy}: se $f$ e $g$ sono continue su $[a,b]$, derivabili in $(a,b)$ e $g'(x)\neq0$ in $(a,b)$, allora esiste $c\in(a,b)$ tale che
\[ \dfrac{f(b)-f(a)}{g(b)-g(a)}=\dfrac{f'(c)}{g'(c)}. \]
\textbf{Ipotesi.} $\arctan x$ e $x$ sono continue e derivabili ovunque; $g'(x)=1\neq0$. \\
(Con $g(x)=x$ il teorema di Cauchy coincide con quello di Lagrange.)

\textbf{Calcolo di $c$.}
\[\begin{aligned}
\dfrac{\arctan1-\arctan0}{1-0} &= \dfrac\pi4\\
\dfrac{1}{1+c^2} &= \dfrac\pi4 && \textcolor{gray}{\text{$f'=\dfrac{1}{1+x^2}$, \ $g'=1$}}\\
1+c^2 &= \dfrac4\pi && \textcolor{gray}{\text{passo ai reciproci}}\\
c^2 &= \dfrac{4-\pi}{\pi} \;\Rightarrow\; c=\sqrt{\dfrac{4-\pi}{\pi}} && \textcolor{gray}{\text{scelgo la radice positiva, in $(0,1)$}}
\end{aligned}\]
\end{explanation}
\end{question}
