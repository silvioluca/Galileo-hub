\begin{question}[ID=1.1, subtitle={Teorema di Lagrange per $f(x)=x^3+2x$ su $[-2,1]$}, tags={lagrange, teoremi}, difficulty=facile]
Verifica se la funzione \(f(x) = x^3+2x\) soddisfa le ipotesi del teorema di Lagrange nell'intervallo \([-2;\ 1]\) e, in caso affermativo, trova il punto \(c\) che soddisfa il teorema.
\begin{solution}
Ipotesi soddisfatte; \ $c=-1$.
\end{solution}
\begin{explanation}
\textbf{Teorema di Lagrange}: se $f$ è continua su $[a,b]$ e derivabile in $(a,b)$, esiste almeno un $c\in(a,b)$ in cui la tangente è parallela alla corda che unisce gli estremi:
\[ f'(c)=\dfrac{f(b)-f(a)}{b-a}. \]
$f$ è un polinomio: continua e derivabile ovunque, quindi le ipotesi valgono.
\[\begin{aligned}
f(1) &= 1+2=3, \qquad f(-2)=-8-4=-12\\
\dfrac{f(1)-f(-2)}{1-(-2)} &= \dfrac{3+12}{3}=5 && \textcolor{gray}{\text{pendenza della corda}}\\
3c^2+2 &= 5 && \textcolor{gray}{\text{$f'(x)=3x^2+2$}}\\
c^2 &= 1 \;\Rightarrow\; c=\pm1
\end{aligned}\]
$c=1$ è un estremo dell'intervallo, non interno: resta $c=-1$.
\end{explanation}
\end{question}

\begin{question}[ID=1.2, subtitle={Teorema di Lagrange per $f(x)=3/\sqrt{x}$ su $[4,9]$}, tags={lagrange, teoremi}, difficulty=media]
Verifica se la funzione \(f(x) = \dfrac{3}{\sqrt{x}}\) soddisfa le ipotesi del teorema di Lagrange nell'intervallo \([4;\ 9]\) e, in caso affermativo, trova il punto \(c\) che soddisfa il teorema.
\begin{solution}
Ipotesi soddisfatte; \ $c=15^{2/3}=\sqrt[3]{225}\approx6{,}08$.
\end{solution}
\begin{explanation}
\textbf{Teorema di Lagrange}: se $f$ è continua su $[a,b]$ e derivabile in $(a,b)$, esiste almeno un $c\in(a,b)$ in cui la tangente è parallela alla corda che unisce gli estremi:
\[ f'(c)=\dfrac{f(b)-f(a)}{b-a}. \]
Su $[4,9]$ si ha $x>0$: $f=3x^{-1/2}$ è continua e derivabile.
\[\begin{aligned}
\dfrac{f(9)-f(4)}{9-4} &= \dfrac{1-\frac32}{5}=-\dfrac{1}{10} && \textcolor{gray}{\text{$f(9)=\dfrac33$, $f(4)=\dfrac32$}}\\
f'(x) &= -\dfrac32x^{-3/2}=-\dfrac{3}{2\sqrt{x^3}}\\
-\dfrac{3}{2c^{3/2}} &= -\dfrac1{10}\\
c^{3/2} &= 15 && \textcolor{gray}{\text{passo ai reciproci e moltiplico}}\\
c &= 15^{2/3} && \textcolor{gray}{\text{elevo alla $\dfrac23$}}
\end{aligned}\]
$c\approx6{,}08\in(4,9)$.
\end{explanation}
\end{question}

\begin{question}[ID=1.3, subtitle={Teorema di Lagrange non applicabile: funzione a tratti su $[0,2]$}, tags={lagrange, teoremi, non applicabile}, difficulty=media]
Data la funzione
\[
f(x) = \begin{cases}
x^2 + 1, & \text{se } x \leq 1 \\
3 - x, & \text{se } x > 1
\end{cases}
\]
verifica se soddisfa le ipotesi del teorema di Lagrange nell'intervallo \([0;\ 2]\) e, in caso affermativo, trova il punto \(c\) che soddisfa il teorema.
\begin{solution}
Il teorema \textbf{non} è applicabile: $f$ non è derivabile in $x=1$.
\end{solution}
\begin{explanation}
\textbf{Teorema di Lagrange}: se $f$ è continua su $[a,b]$ e derivabile in $(a,b)$, esiste almeno un $c\in(a,b)$ in cui la tangente è parallela alla corda che unisce gli estremi:
\[ f'(c)=\dfrac{f(b)-f(a)}{b-a}. \]
\textbf{Continuità in $x=1$:} \ $f(1)=1+1=2$ e $\displaystyle\lim_{x\to1^+}(3-x)=2$: è continua.

\textbf{Derivabilità in $x=1$:}
\[ f'_-(1)=2x\big|_{x=1}=2, \qquad f'_+(1)=-1 \]
Derivate diverse: in $x=1$ c'è un punto angoloso. Poiché $1\in(0,2)$, manca l'ipotesi di derivabilità.
\end{explanation}
\end{question}

\[
f(x) = \begin{cases}
x^2 -4x+5, & \text{se } x \leq 2 \\
x, & \text{se } x > 2
\end{cases}

Trova il valore di \(a\) tale che la funzione \(f(x) = x^3 + ax\) soddisfi il teorema di Lagrange nell'intervallo \([-1;\ 1]\) e determina il punto \(c\) che soddisfa il teorema.
\begin{solution}
Il teorema vale per \textbf{ogni} valore di $a$; \ $c=\pm\dfrac{1}{\sqrt3}$.

3c^2+a &= 1+a && \textcolor{gray}{\text{$f'(x)=3x^2+a$}}\\
c^2 &= \dfrac13 && \textcolor{gray}{\text{il parametro $a$ si elimina}}\\
c &= \pm\dfrac{1}{\sqrt3}
\end{aligned}\]
Entrambi i valori ($\approx\pm0{,}58$) sono in $(-1,1)$.
\end{explanation}
\end{question}

\begin{question}[ID=2.2, subtitle={Parametro $b$ per Lagrange: $1/x$ su $[1,b]$ con $c=2$}, tags={lagrange, teoremi, parametrico}, difficulty=media]
Data la funzione \(f(x) = \dfrac{1}{x}\) nell'intervallo \([1;\ b]\), determina il valore di \(b > 1\) tale che il punto \(c\) che soddisfa il teorema di Lagrange sia \(c = 2\).
\begin{solution}
$b=4$
\end{solution}
\begin{explanation}
\textbf{Teorema di Lagrange}: se $f$ è continua su $[a,b]$ e derivabile in $(a,b)$, esiste almeno un $c\in(a,b)$ in cui la tangente è parallela alla corda che unisce gli estremi:
\[ f'(c)=\dfrac{f(b)-f(a)}{b-a}. \]
Per $b>1$, $f=\dfrac1x$ è continua e derivabile su $[1,b]$. Impongo che la pendenza della corda sia uguale a $f'(2)$:
\[\begin{aligned}
f'(2) &= -\dfrac1{2^2}=-\dfrac14 && \textcolor{gray}{\text{$f'(x)=-\dfrac1{x^2}$}}\\
\dfrac{\frac1b-1}{b-1} &= \dfrac{\frac{1-b}{b}}{b-1}=-\dfrac1b && \textcolor{gray}{\text{$1-b=-(b-1)$}}\\
-\dfrac1b &= -\dfrac14 \;\Rightarrow\; b=4
\end{aligned}\]
Verifica: $c=2\in(1,4)$.
\end{explanation}
\end{question}

\begin{question}[ID=2.3, subtitle={Parametri $a,b$ per Lagrange: funzione a tratti su $[-2,0]$}, tags={lagrange, teoremi, parametrico}, difficulty=difficile]
Trova il valore di \(a\) e \(b\) tale che la funzione
\[
f(x) = \begin{cases}
x^2 +ax-1, & \text{se } x < -1 \\
-4x^2+x+b, & \text{se } x \ge -1
\end{cases}
\]
soddisfi il teorema di Lagrange nell'intervallo \([-2;\ 0]\) e determina il punto \(c\) che soddisfa il teorema.
\begin{solution}
$a=11$, \ $b=-6$; \ $c=-\dfrac{11}{16}$.
\end{solution}
\begin{explanation}
\textbf{Teorema di Lagrange}: se $f$ è continua su $[a,b]$ e derivabile in $(a,b)$, esiste almeno un $c\in(a,b)$ in cui la tangente è parallela alla corda che unisce gli estremi:
\[ f'(c)=\dfrac{f(b)-f(a)}{b-a}. \]
Il punto di raccordo $x=-1$ è interno a $[-2,0]$: servono continuità e derivabilità lì.
\[\begin{aligned}
\lim_{x\to-1^-}(x^2+ax-1) &= 1-a-1=-a && \textcolor{gray}{\text{ramo sinistro}}\\
f(-1) &= -4-1+b=b-5 && \textcolor{gray}{\text{ramo destro}}\\
-a &= b-5 \;\Rightarrow\; b=5-a && \textcolor{gray}{\text{continuità}}\\
f'_-(-1) &= 2(-1)+a=a-2, \qquad f'_+(-1)=-8(-1)+1=9\\
a-2 &= 9 \;\Rightarrow\; a=11, \quad b=-6 && \textcolor{gray}{\text{derivabilità}}
\end{aligned}\]
Ora cerco $c$:
\[\begin{aligned}
f(-2) &= 4-22-1=-19, \qquad f(0)=b=-6\\
\dfrac{f(0)-f(-2)}{0-(-2)} &= \dfrac{13}{2}
\end{aligned}\]
\begin{itemize}
    \item Ramo sinistro ($-2<c<-1$): $2c+11=\tfrac{13}2 \Rightarrow c=-\tfrac94=-2{,}25$, \textbf{fuori} dall'intervallo: da scartare.
    \item Ramo destro ($-1\leq c<0$): $-8c+1=\tfrac{13}2 \Rightarrow c=-\tfrac{11}{16}\approx-0{,}69$, accettabile.
\end{itemize}
\end{explanation}
\end{question}

\begin{question}[ID=2.4, subtitle={Lagrange per $|x^2-6x+5|$ su $[0,2]$ e $[2,a]$}, tags={lagrange, teoremi, valore assoluto}, difficulty=difficile]
Data la funzione $f(x) = |x^2 - 6x + 5|$, determina se soddisfa il teorema di Lagrange nell'intervallo \([0;\ 2]\). Determina inoltre il valore massimo del parametro $a$ affinché il teorema di Lagrange sia valido nell'intervallo $[2, a]$ e, in caso affermativo, determina i valori che soddisfano il teorema.
\begin{solution}
Su $[0,2]$ il teorema \textbf{non} è applicabile (non derivabile in $x=1$). \ Su $[2,a]$ vale per $2<a\leq5$, quindi il valore massimo è $a=5$; \ $c=1+\dfrac a2$ \ (per $a=5$: $c=\tfrac72$).
\end{solution}
\begin{explanation}
\textbf{Teorema di Lagrange}: se $f$ è continua su $[a,b]$ e derivabile in $(a,b)$, esiste almeno un $c\in(a,b)$ in cui la tangente è parallela alla corda che unisce gli estremi:
\[ f'(c)=\dfrac{f(b)-f(a)}{b-a}. \]
Scompongo: $x^2-6x+5=(x-1)(x-5)$, negativo tra $1$ e $5$. Il valore assoluto crea punti angolosi dove si annulla, cioè in $x=1$ e $x=5$.

\textbf{Intervallo $[0,2]$.} Contiene $x=1$ al suo interno, dove $f$ non è derivabile (derivata sinistra $-4$, destra $+4$): il teorema non si applica.

\textbf{Intervallo $[2,a]$.} Finché $a\leq5$ l'intervallo non contiene punti angolosi interni e lì $f(x)=-(x^2-6x+5)$. Per $a>5$ conterrebbe $x=5$: il valore massimo è $a=5$.
\[\begin{aligned}
\dfrac{f(a)-f(2)}{a-2} &= \dfrac{-a^2+6a-5-3}{a-2}=-\dfrac{(a-2)(a-4)}{a-2}=4-a && \textcolor{gray}{\text{$f(2)=3$}}\\
-2c+6 &= 4-a && \textcolor{gray}{\text{$f'(x)=-2x+6$}}\\
c &= \dfrac{a+2}{2}=1+\dfrac a2
\end{aligned}\]
$c$ è il punto medio tra $2$ e $a$, quindi sta sempre in $(2,a)$.
\end{explanation}
\end{question}
