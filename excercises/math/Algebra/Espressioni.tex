\begin{question}[ID=1.1, subtitle={Semplificazione di espressioni con potenze}, tags={potenze, semplificazione}, difficulty=facile]
Semplificare le seguenti espressioni usando le proprietà delle potenze e scrivere il risultato sotto forma di potenza con esponente intero, quando possibile:

\begin{center}\begin{minipage}{0.92\linewidth}
\begin{multicols}{2}
\begin{enumerate}[leftmargin=3.2cm]
\item $2^{5}\cdot 2^{-3} : 2^{2}$
\item $(3^{2}\cdot 3^{-1})^2$
\item $\Big(\dfrac{2}{3}\Big)^3 : \Big(\dfrac{2}{3}\Big)^{-1} \cdot\Big(\dfrac{3}{2}\Big)^{-2} \cdot \Big(\dfrac{2}{3}\Big)^{-4}$
\item $(5^{-1})^{-3}\cdot 5^{2}$
\end{enumerate}
\end{multicols}
\end{minipage}\end{center}
\begin{solution}$a)\ 2^0=1$; $b)\ 3^2=9$; $c)\ \left(\tfrac{2}{3}\right)^2=\tfrac{4}{9}$; $d)\ 5^5=3125$\end{solution}
\end{question}

\begin{question}[ID=2.1, subtitle={Calcolo di espressioni frazionarie — serie 1}, tags={espressioni, frazionarie}, difficulty=facile]
Calcola il valore delle seguenti espressioni:

\begin{center}\begin{minipage}{0.92\linewidth}
\begin{multicols}{2}
\begin{enumerate}[leftmargin=3.2cm]
\item $-\frac{2}{3} -\Biggl\{ \frac{1}{6} -\left(\frac{1}{2}-\frac{1}{3}\right) +3 -\left(\frac{7}{12}+\frac{1}{4}\right) \Biggr\}$
\item $\left[ \left(\frac{5}{2}\right)^{4} \cdot \left(3-\frac{1}{2}\right)^{2} \right] :\left(\frac{5}{2}\right)^{2} :\left[ \left(-\frac{5}{2}\right)^{6} :\frac{5}{2} \right]$
\item $\left\{ \frac{15}{16} - \left[ \left(\frac{3}{2}-\frac14\right)^{2} :\frac{5}{4} - \left(\frac12+\frac14\right)^{2} \right] \right\}^{2} +\frac14$
\item $\left[ \left(0{,}20+\frac{2}{3}\right) : \left(\frac{7}{3}-2\right) \right] : \left[ 1{,}3 : \left(0{,}75-\frac{2}{3}\right)^{-1} \right]$
\end{enumerate}
\end{multicols}
\end{minipage}\end{center}
\begin{solution}
$a)$ Le parentesi danno $\tfrac12-\tfrac13=\tfrac16$ e $\tfrac{7}{12}+\tfrac14=\tfrac56$; la graffa vale $\tfrac16-\tfrac16+3-\tfrac56=\tfrac{13}{6}$, quindi $-\tfrac23-\tfrac{13}{6}=\boxed{-\tfrac{17}{6}}$. \\
$b)$ $3-\tfrac12=\tfrac52$, perciò il numeratore è $\big(\tfrac52\big)^4\big(\tfrac52\big)^2=\big(\tfrac52\big)^6$; diviso $\big(\tfrac52\big)^2$ dà $\big(\tfrac52\big)^4$. La parentesi finale è $\big(\tfrac52\big)^6:\tfrac52=\big(\tfrac52\big)^5$. Quindi $\big(\tfrac52\big)^4:\big(\tfrac52\big)^5=\big(\tfrac52\big)^{-1}=\boxed{\tfrac25}$. \\
$c)$ $\tfrac32-\tfrac14=\tfrac54$, $\big(\tfrac54\big)^2:\tfrac54=\tfrac54$; $\big(\tfrac12+\tfrac14\big)^2=\tfrac{9}{16}$; quindi $\tfrac54-\tfrac{9}{16}=\tfrac{11}{16}$ e $\tfrac{15}{16}-\tfrac{11}{16}=\tfrac14$. Infine $\big(\tfrac14\big)^2+\tfrac14=\tfrac{1}{16}+\tfrac14=\boxed{\tfrac{5}{16}}$. \\
$d)$ $\tfrac15+\tfrac23=\tfrac{13}{15}$, $\tfrac73-2=\tfrac13$, rapporto $\tfrac{13}{5}$; $0{,}75-\tfrac23=\tfrac{1}{12}$, $\big(\tfrac{1}{12}\big)^{-1}=12$, $1{,}3:12=\tfrac{13}{120}$. Quindi $\tfrac{13}{5}:\tfrac{13}{120}=\boxed{24}$.
\end{solution}
\end{question}

\begin{question}[ID=3.1, subtitle={Espressioni con numeri periodici}, tags={espressioni, periodici}, difficulty=facile]
Calcola il valore delle seguenti espressioni con numeri periodici:
\begin{center}\begin{minipage}{0.92\linewidth}
\begin{multicols}{2}
\begin{enumerate}[leftmargin=3.2cm]
\item $1{,}2 - \frac{3}{5} \cdot 2{,}\overline{8} \cdot (1{,}3 \cdot 10)^{-1}$
\item $\left[ 0{,}2\overline{6} : (0{,}4)^{2} \right]^{2} : \left(\frac{5}{3}\right)^{4}$
\end{enumerate}
\end{multicols}
\end{minipage}\end{center}
\begin{solution}
Si convertono i numeri periodici in frazione. \\
$a)$ $1{,}2=\tfrac65$, $2{,}\overline{8}=\tfrac{26}{9}$, $1{,}3\cdot10=13$. Allora $\tfrac35\cdot\tfrac{26}{9}\cdot\tfrac{1}{13}=\tfrac{2}{15}$ e $\tfrac65-\tfrac{2}{15}=\tfrac{18-2}{15}=\boxed{\tfrac{16}{15}}$. \\
$b)$ $0{,}2\overline{6}=\tfrac{4}{15}$, $(0{,}4)^2=\tfrac{4}{25}$, quindi $\tfrac{4}{15}:\tfrac{4}{25}=\tfrac53$ e $\big(\tfrac53\big)^2=\tfrac{25}{9}$. Infine $\tfrac{25}{9}:\big(\tfrac53\big)^4=\tfrac{25}{9}\cdot\tfrac{81}{625}=\boxed{\tfrac{9}{25}}$.
\end{solution}
\end{question}
