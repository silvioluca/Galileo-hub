\begin{question}[ID=1.1, subtitle={Integrali indefiniti — serie 1}, tags={integrali, indefiniti}, difficulty=media]
Risolvi i seguenti integrali indefiniti

\begin{center}\begin{minipage}{0.7\linewidth}
\begin{enumerate}
\item $\int (3x^2 - 5x + 2) \ dx$
\item $\int_{0}^{1} \frac{1}{2x+1} \ dx$
\item $\int \frac{\ln^3 x}{x} \ dx$
\item $\int \sqrt{9 - x^2} \ dx$
\item $\int x \cdot \sin(x) \ dx$
\item $\int \tan(x) \ dx$
\item $\int \frac{2x + 3}{x^2 + 4x - 5} \ dx$
\item $\int \frac{1}{x^2 + 4x + 7} \ dx$
\end{enumerate}
\end{minipage}\end{center}
\begin{solution}
\begin{enumerate}
    \item $x^3-\tfrac52x^2+2x+C$
    \item $\tfrac{\ln3}{2}$
    \item $\tfrac14\ln^4x+C$
    \item $\tfrac x2\sqrt{9-x^2}+\tfrac92\arcsin\tfrac x3+C$
    \item $-x\cos x+\sin x+C$
    \item $-\ln|\cos x|+C$
    \item $\tfrac56\ln|x-1|+\tfrac76\ln|x+5|+C$
    \item $\tfrac{1}{\sqrt3}\arctan\tfrac{x+2}{\sqrt3}+C$
\end{enumerate}
\end{solution}
\begin{explanation}
\begin{enumerate}
    \item Integrale di un polinomio: si integra termine per termine con $\int x^n\,dx=\dfrac{x^{n+1}}{n+1}$.
    \[ \int(3x^2-5x+2)\,dx=3\cdot\dfrac{x^3}{3}-5\cdot\dfrac{x^2}{2}+2x+C=x^3-\dfrac52x^2+2x+C \]
    \item Questo è un integrale \textbf{definito}. Il numeratore è (a meno del fattore $2$) la derivata del denominatore, quindi la primitiva è un logaritmo:
    \[\begin{aligned}
    \int\dfrac{1}{2x+1}\,dx &= \dfrac12\int\dfrac{2}{2x+1}\,dx=\dfrac12\ln|2x+1| && \textcolor{gray}{\text{moltiplico e divido per $2$}}\\
    \Big[\dfrac12\ln|2x+1|\Big]_0^1 &= \dfrac12\ln3-\dfrac12\ln1=\dfrac{\ln3}{2} && \textcolor{gray}{\text{teorema fondamentale}}
    \end{aligned}\]
    \item $\dfrac1x$ è la derivata di $\ln x$: l'integrale ha la forma $\int t^3\,t'\,dx$ con $t=\ln x$.
    \[ \int\ln^3x\cdot\dfrac1x\,dx=\dfrac{\ln^4x}{4}+C \]
    \item Sostituzione $x=3\sin t$ (con $dx=3\cos t\,dt$ e $\sqrt{9-x^2}=3\cos t$):
    \[\begin{aligned}
    \int\sqrt{9-x^2}\,dx &= \int9\cos^2t\,dt\\
    &= \dfrac92\int(1+\cos2t)\,dt && \textcolor{gray}{\text{formula $\cos^2t=\dfrac{1+\cos2t}{2}$}}\\
    &= \dfrac92\left(t+\sin t\cos t\right)+C && \textcolor{gray}{\text{$\dfrac12\sin2t=\sin t\cos t$}}\\
    &= \dfrac92\arcsin\dfrac x3+\dfrac x2\sqrt{9-x^2}+C && \textcolor{gray}{\text{torno alla $x$: $\sin t=\dfrac x3$, $\cos t=\dfrac{\sqrt{9-x^2}}{3}$}}
    \end{aligned}\]
    \item Integrazione per parti $\int f\,g'=fg-\int f'g$, con $f=x$ (si semplifica derivando) e $g'=\sin x$:
    \[\begin{aligned}
    \int x\sin x\,dx &= x\cdot(-\cos x)-\int1\cdot(-\cos x)\,dx\\
    &= -x\cos x+\sin x+C
    \end{aligned}\]
    \item Scrivo $\tan x=\dfrac{\sin x}{\cos x}$: il numeratore è l'opposto della derivata del denominatore.
    \[ \int\dfrac{\sin x}{\cos x}\,dx=-\int\dfrac{-\sin x}{\cos x}\,dx=-\ln|\cos x|+C \]
    \item Il denominatore si scompone: $x^2+4x-5=(x-1)(x+5)$. Scompongo in fratti semplici:
    \[\begin{aligned}
    \dfrac{2x+3}{(x-1)(x+5)} &= \dfrac{A}{x-1}+\dfrac{B}{x+5}\\
    2x+3 &= A(x+5)+B(x-1) && \textcolor{gray}{\text{denominatore comune}}\\
    x=1:\ 5 &= 6A \;\Rightarrow\; A=\dfrac56 && \textcolor{gray}{\text{sostituisco gli zeri del denominatore}}\\
    x=-5:\ -7 &= -6B \;\Rightarrow\; B=\dfrac76
    \end{aligned}\]
    Integro ogni frazione come un logaritmo: $\tfrac56\ln|x-1|+\tfrac76\ln|x+5|+C$.
    \item Il denominatore non si scompone ($\Delta<0$): completo il quadrato e riconduco all'arcotangente.
    \[\begin{aligned}
    x^2+4x+7 &= (x+2)^2+3 && \textcolor{gray}{\text{completamento del quadrato}}\\
    \int\dfrac{dx}{(x+2)^2+3} &= \dfrac{1}{\sqrt3}\arctan\dfrac{x+2}{\sqrt3}+C && \textcolor{gray}{\text{$\int\dfrac{dt}{t^2+a^2}=\dfrac1a\arctan\dfrac ta$}}
    \end{aligned}\]
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=1.2, subtitle={Integrali indefiniti — serie 2}, tags={integrali, indefiniti}, difficulty=media]
Risolvi i seguenti integrali indefiniti

\begin{center}\begin{minipage}{0.7\linewidth}
\begin{enumerate}
\item $\int (2x^3 - x^2 + 4x -1) \ dx$
\item $\int \frac{2x+1}{x^2+x} \ dx$
\item $\int \frac{1}{x \cdot \ln x} \ dx$
\item $\int \sqrt{16- x^2} \ dx$
\item $\int x \cdot e^{x} \ dx$
\item $\int \frac{dx}{\cos^2(3x)}$
\item $\int \frac{1}{x^2 + 2x + 10} \ dx$
\item $\int \frac{2x+1}{x^2-9} \ dx$
\end{enumerate}
\end{minipage}\end{center}
\begin{solution}
\begin{enumerate}
    \item $\tfrac12x^4-\tfrac13x^3+2x^2-x+C$
    \item $\ln|x^2+x|+C$
    \item $\ln|\ln x|+C$
    \item $\tfrac x2\sqrt{16-x^2}+8\arcsin\tfrac x4+C$
    \item $e^x(x-1)+C$
    \item $\tfrac13\tan(3x)+C$
    \item $\tfrac13\arctan\tfrac{x+1}{3}+C$
    \item $\tfrac76\ln|x-3|+\tfrac56\ln|x+3|+C$
\end{enumerate}
\end{solution}
\begin{explanation}
\begin{enumerate}
    \item Termine per termine: \ $2\cdot\dfrac{x^4}{4}-\dfrac{x^3}{3}+4\cdot\dfrac{x^2}{2}-x+C=\dfrac12x^4-\dfrac13x^3+2x^2-x+C$.
    \item Il numeratore $2x+1$ è esattamente la derivata del denominatore $x^2+x$:
    \[ \int\dfrac{f'(x)}{f(x)}\,dx=\ln|f(x)|+C \;\Rightarrow\; \ln|x^2+x|+C \]
    \item $\dfrac1x$ è la derivata di $\ln x$: l'integrale ha la forma $\int\dfrac{t'}{t}\,dx$ con $t=\ln x$.
    \[ \int\dfrac{1/x}{\ln x}\,dx=\ln|\ln x|+C \]
    \item Come per $\sqrt{9-x^2}$, con la sostituzione $x=4\sin t$:
    \[\begin{aligned}
    \int\sqrt{16-x^2}\,dx &= \int16\cos^2t\,dt=8(t+\sin t\cos t)+C\\
    &= 8\arcsin\dfrac x4+\dfrac x2\sqrt{16-x^2}+C
    \end{aligned}\]
    \item Per parti con $f=x$ e $g'=e^x$:
    \[\begin{aligned}
    \int xe^x\,dx &= xe^x-\int e^x\,dx\\
    &= xe^x-e^x+C=e^x(x-1)+C
    \end{aligned}\]
    \item $\dfrac{1}{\cos^2t}$ è la derivata di $\tan t$; con $t=3x$ serve il fattore $3$:
    \[ \int\dfrac{dx}{\cos^2(3x)}=\dfrac13\int\dfrac{3}{\cos^2(3x)}\,dx=\dfrac13\tan(3x)+C \]
    \item Completo il quadrato: $x^2+2x+10=(x+1)^2+9$, poi uso $\int\dfrac{dt}{t^2+a^2}=\dfrac1a\arctan\dfrac ta$ con $a=3$:
    \[ \dfrac13\arctan\dfrac{x+1}{3}+C \]
    \item $x^2-9=(x-3)(x+3)$: fratti semplici.
    \[\begin{aligned}
    2x+1 &= A(x+3)+B(x-3)\\
    x=3:\ 7 &= 6A \;\Rightarrow\; A=\dfrac76\\
    x=-3:\ -5 &= -6B \;\Rightarrow\; B=\dfrac56
    \end{aligned}\]
    Quindi $\tfrac76\ln|x-3|+\tfrac56\ln|x+3|+C$ \ (equivale a $\ln|x^2-9|+\tfrac16\ln\left|\tfrac{x-3}{x+3}\right|+C$).
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=2.1, subtitle={Primitiva di $f(x)=\ln x^2$ passante per $(1,1)$}, tags={integrali, primitiva}, difficulty=media]
Tra le primitive della funzione $f(x) = \ln x^2$ determina quella il cui grafico passa per il punto $(1,1)$.
\begin{solution}
$F(x)=2x\ln|x|-2x+3$
\end{solution}
\begin{explanation}
Per le proprietà dei logaritmi $\ln x^2=2\ln|x|$. Calcolo $\int\ln x\,dx$ per parti, con $f=\ln x$ e $g'=1$:
\[\begin{aligned}
\int\ln x\,dx &= x\ln x-\int x\cdot\dfrac1x\,dx=x\ln x-x\\
F(x) &= 2x\ln|x|-2x+C && \textcolor{gray}{\text{moltiplico per $2$}}
\end{aligned}\]
Impongo il passaggio per $(1,1)$:
\[ F(1)=2\cdot1\cdot0-2+C=1 \;\Rightarrow\; C=3 \]
\end{explanation}
\end{question}

\begin{question}[ID=3.1, subtitle={Valor medio di $\ln x$ su $[1,2]$}, tags={integrali, valor medio}, difficulty=media]
Trova il valor medio $M$ della funzione $f(x) = \ln x$ su $[1, 2]$ e stabilisci per quale ascissa $c$ vale $f(c) = M$.
\begin{solution}
$M=\ln4-1\approx0{,}386$, \quad $c=\dfrac4e\approx1{,}47$
\end{solution}
\begin{explanation}
Il valor medio di $f$ su $[a,b]$ è $M=\dfrac{1}{b-a}\displaystyle\int_a^bf(x)\,dx$; il teorema della media garantisce un $c\in[a,b]$ con $f(c)=M$.
\[\begin{aligned}
\int_1^2\ln x\,dx &= \Big[x\ln x-x\Big]_1^2 && \textcolor{gray}{\text{primitiva calcolata per parti}}\\
&= (2\ln2-2)-(0-1)=2\ln2-1=\ln4-1\\
M &= \dfrac{\ln4-1}{2-1}=\ln4-1
\end{aligned}\]
Cerco $c$: \ $\ln c=\ln4-1 \;\Rightarrow\; c=e^{\ln4-1}=\dfrac{e^{\ln4}}{e}=\dfrac4e\approx1{,}47$, che appartiene a $[1,2]$.
\end{explanation}
\end{question}

\begin{question}[ID=4.1, subtitle={Area della versiera di Agnesi nel primo quadrante}, tags={integrali, area, applicazioni}, difficulty=difficile]
La \textit{versiera di Agnesi} è una curva a cui Gaetana Agnesi ha legato il suo nome nel 1748, una curva che peraltro non fu scoperta da lei, ma da Guido Grandi. Analiticamente è il luogo dei punti di intersezione tra due rette che si muovono in relazione a una circonferenza. La funzione è esprimibile come:
\begin{equation*}
    f(x) = \frac{8a^3}{4a^2 + x^2}
\end{equation*}
\begin{enumerate}
    \item Determina l'area delimitata dalla curva nel primo quadrante, l'asse $x$ e la retta verticale di equazione $x=2$.
    \item Calcolane il valore con $a=1$
\end{enumerate}
\begin{solution}
\begin{enumerate}
    \item $A=4a^2\arctan\dfrac1a$
    \item $A=\pi$
\end{enumerate}
\end{solution}
\begin{explanation}
\begin{enumerate}
    \item Nel primo quadrante la curva è sopra l'asse $x$, quindi l'area è l'integrale da $0$ a $2$. Raccolgo per ricondurmi alla forma $\int\dfrac{dx}{x^2+k^2}=\dfrac1k\arctan\dfrac xk$ con $k=2a$:
    \[\begin{aligned}
    A &= \int_0^2\dfrac{8a^3}{4a^2+x^2}\,dx=8a^3\int_0^2\dfrac{dx}{x^2+(2a)^2}\\
    &= 8a^3\cdot\dfrac{1}{2a}\Big[\arctan\dfrac{x}{2a}\Big]_0^2\\
    &= 4a^2\left(\arctan\dfrac1a-0\right)=4a^2\arctan\dfrac1a
    \end{aligned}\]
    \item Con $a=1$: \ $A=4\arctan1=4\cdot\dfrac\pi4=\pi$.
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=4.2, subtitle={Area tra la parabola $y=-x^2+2x$ e la retta $y=-x$}, tags={integrali, area tra curve}, difficulty=media]
\begin{minipage}[t]{0.65\textwidth}
Calcolare l'area compresa tra la parabola $y = -x^2 + 2x$ \\ e la retta $y = -x$
\end{minipage}
\begin{minipage}[t]{0.35\textwidth}
\centering \vspace{-3\baselineskip}
\includegraphics[width=\textwidth]{images/area_tra_curve.png}
\end{minipage}
\begin{solution}
$A=\dfrac92$
\end{solution}
\begin{explanation}
Trovo dove si incontrano le due curve:
\[\begin{aligned}
-x^2+2x &= -x\\
-x^2+3x &= 0\\
x(3-x) &= 0 \;\Rightarrow\; x=0,\ x=3
\end{aligned}\]
Tra $0$ e $3$ la parabola sta sopra la retta (ad esempio in $x=1$: $1>-1$). L'area è l'integrale della differenza \emph{sopra meno sotto}:
\[\begin{aligned}
A &= \int_0^3\big[(-x^2+2x)-(-x)\big]\,dx=\int_0^3(-x^2+3x)\,dx\\
&= \Big[-\dfrac{x^3}{3}+\dfrac{3x^2}{2}\Big]_0^3=-9+\dfrac{27}{2}=\dfrac92
\end{aligned}\]
\end{explanation}
\end{question}

\begin{question}[ID=5.1, subtitle={Volume della tromba di Torricelli su $[1,2]$}, tags={integrali, volume, solido di rotazione}, difficulty=difficile]
La tromba di Torricelli è un solido ottenuto dalla rivoluzione intorno all'asse $x$ della curva di equazione $y=\dfrac{1}{x}$. Determina il volume del solido di rotazione limitandoti all'intervallo $x \in [1, 2]$
\begin{solution}
$V=\dfrac\pi2$
\end{solution}
\begin{explanation}
Il volume del solido ottenuto ruotando il grafico di $f$ attorno all'asse $x$ tra $a$ e $b$ è $V=\pi\displaystyle\int_a^bf(x)^2\,dx$ (somma di tanti dischi di raggio $f(x)$).
\[\begin{aligned}
V &= \pi\int_1^2\dfrac{1}{x^2}\,dx\\
&= \pi\Big[-\dfrac1x\Big]_1^2 && \textcolor{gray}{\text{$\int x^{-2}dx=-x^{-1}$}}\\
&= \pi\left(-\dfrac12+1\right)=\dfrac\pi2
\end{aligned}\]
\end{explanation}
\end{question}
