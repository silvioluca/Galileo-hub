\begin{question}[ID=1.1, subtitle={Completamento con percentuali — serie 1}, tags={percentuali}, difficulty=facile]
Completa con i numeri opportuni:
\begin{center}\begin{minipage}{0.92\linewidth}
\begin{multicols}{2}
\begin{enumerate}
\item $\text{Il }20\%\text{ di }150\text{ è } \rule{1.5cm}{0.4pt}$
\item $\text{Il }15\%\text{ di }200\text{ è } \rule{1.5cm}{0.4pt}$
\item $\text{Il }18\%\text{ di } \rule{1.5cm}{0.4pt} \text{ è }3{,}6$
\item $\text{Il } \rule{0.5cm}{0.4pt}\%\text{ di }80\text{ è }4$
\end{enumerate}
\end{multicols}
\end{minipage}\end{center}
\begin{solution}
\begin{enumerate}
    \item $30$
    \item $30$
    \item $20$
    \item $5\%$
\end{enumerate}
\end{solution}
\begin{explanation}
Calcolare il $p\%$ di un numero $N$ significa moltiplicare $N$ per $\dfrac{p}{100}$. Quando manca il numero o la percentuale, si scrive la stessa relazione e si ricava l'incognita.
\begin{enumerate}
    \item \[ \dfrac{20}{100}\cdot150=0{,}2\cdot150=30 \]
    \item \[ \dfrac{15}{100}\cdot200=0{,}15\cdot200=30 \]
    \item Cerco il numero $N$ di cui $3{,}6$ è il $18\%$:
    \[\begin{aligned}
    0{,}18\cdot N &= 3{,}6\\
    N &= \dfrac{3{,}6}{0{,}18}=20 && \textcolor{gray}{\text{divido per $0{,}18$}}
    \end{aligned}\]
    \item Cerco la percentuale $p$ tale che il $p\%$ di $80$ sia $4$:
    \[\begin{aligned}
    \dfrac{p}{100}\cdot80 &= 4\\
    p &= \dfrac{4}{80}\cdot100=5 && \textcolor{gray}{\text{rapporto parte/totale, per $100$}}
    \end{aligned}\]
    Quindi $5\%$.
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=1.2, subtitle={Completamento con percentuali — serie 2}, tags={percentuali}, difficulty=facile]
Completa con i numeri opportuni:
\begin{center}\begin{minipage}{0.92\linewidth}
\begin{multicols}{2}
\begin{enumerate}
\item $\text{Il }25\%\text{ di }160\text{ è } \rule{1.5cm}{0.4pt}$
\item $\text{Il }30\%\text{ di }90\text{ è } \rule{1.5cm}{0.4pt}$
\item $\text{Il }12\%\text{ di } \rule{1.5cm}{0.4pt} \text{ è }6$
\item $\text{Il } \rule{0.5cm}{0.4pt}\%\text{ di }250\text{ è }50$
\end{enumerate}
\end{multicols}
\end{minipage}\end{center}
\begin{solution}
\begin{enumerate}
    \item $40$
    \item $27$
    \item $50$
    \item $20\%$
\end{enumerate}
\end{solution}
\begin{explanation}
\begin{enumerate}
    \item \[ 0{,}25\cdot160=40 \]
    \item \[ 0{,}30\cdot90=27 \]
    \item \[\begin{aligned}
    0{,}12\cdot N &= 6\\
    N &= \dfrac{6}{0{,}12}=50
    \end{aligned}\]
    \item La percentuale è il rapporto tra la parte e il totale, moltiplicato per $100$:
    \[ p=\dfrac{50}{250}\cdot100=20 \quad\Rightarrow\quad 20\% \]
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=2.1, subtitle={Prezzo scontato del maglione natalizio}, tags={percentuali}, difficulty=facile]
Quest'anno per Natale ho deciso: solo regali utili. Un negozio di abbigliamento applica uno sconto del $\text{25\%}$ su un maglione a tema natalizio.

\begin{enumerate}
    \item Sapendo che il prezzo iniziale del maglione è di $\text{39,99}$ euro, determina il prezzo scontato.
    \color{gray}\item In un altro negozio, lo stesso maglione viene venduto a $\text{25}$ euro già scontato del $\text{20 \%}$. Determina il prezzo iniziale del maglione prima dell'applicazione dello sconto.
\end{enumerate}
\begin{solution}
\begin{enumerate}
    \item $29{,}99$ €
    \item $31{,}25$ €
\end{enumerate}
\end{solution}
\begin{explanation}
\begin{enumerate}
    \item Con uno sconto del $25\%$ si paga il $100\%-25\%=75\%$ del prezzo:
    \[ 39{,}99\cdot0{,}75=29{,}9925\approx29{,}99 \text{ euro} \]
    \item Dopo uno sconto del $20\%$ si paga l'$80\%$ del prezzo iniziale $P$, e questo vale $25$ €:
    \[\begin{aligned}
    0{,}80\cdot P &= 25\\
    P &= \dfrac{25}{0{,}80}=31{,}25 \text{ euro} && \textcolor{gray}{\text{divido per $0{,}80$}}
    \end{aligned}\]
    Attenzione: non basta aggiungere il $20\%$ a $25$ €, perché lo sconto era calcolato sul prezzo iniziale, non su quello scontato.
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=3.1, subtitle={Conversione in notazione scientifica}, tags={notazione scientifica}, difficulty=facile]
Trasforma le seguenti misure in notazione scientifica:
\begin{center}\begin{minipage}{0.92\linewidth}
\begin{multicols}{2}
\begin{enumerate}
\item $0{,}0045 = \rule{3cm}{0.4pt}$
\item $32000 = \rule{3cm}{0.4pt}$
\item $0{,}00000082 = \rule{3cm}{0.4pt}$
\item $5\,600\,000 = \rule{3cm}{0.4pt}$
\item $123{,}45 = \rule{3cm}{0.4pt}$
\item $34{,}5 \cdot 10^{-2} = \rule{3cm}{0.4pt}$
\end{enumerate}
\end{multicols}
\end{minipage}\end{center}
\begin{solution}
\begin{enumerate}
    \item $4{,}5\times10^{-3}$
    \item $3{,}2\times10^{4}$
    \item $8{,}2\times10^{-7}$
    \item $5{,}6\times10^{6}$
    \item $1{,}2345\times10^{2}$
    \item $3{,}45\times10^{-1}$
\end{enumerate}
\end{solution}
\begin{explanation}
Un numero è in notazione scientifica se è scritto come $a\times10^n$ con $1\leq a<10$. L'esponente $n$ è il numero di posti di cui si sposta la virgola: positivo se il numero di partenza è maggiore di $10$, negativo se è minore di $1$.
\begin{enumerate}
    \item $0{,}0045$: la virgola si sposta di $3$ posti a destra per avere $4{,}5$: \ $4{,}5\times10^{-3}$.
    \item $32000$: la virgola si sposta di $4$ posti a sinistra per avere $3{,}2$: \ $3{,}2\times10^{4}$.
    \item $0{,}00000082$: $7$ posti a destra per avere $8{,}2$: \ $8{,}2\times10^{-7}$.
    \item $5\,600\,000$: $6$ posti a sinistra: \ $5{,}6\times10^{6}$.
    \item $123{,}45$: $2$ posti a sinistra: \ $1{,}2345\times10^{2}$.
    \item $34{,}5$ non è tra $1$ e $10$: lo scrivo come $3{,}45\times10^1$ e poi moltiplico le potenze:
    \[ 3{,}45\times10^{1}\cdot10^{-2}=3{,}45\times10^{-1} \]
\end{enumerate}
\end{explanation}
\end{question}
