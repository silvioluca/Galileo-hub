\begin{question}[ID=1.1, subtitle={Verifica della circonferenza $x^2+y^2-4x+6y-3=0$}, difficulty=facile, tags={circonferenza}]
Verifica se l'equazione $x^2+y^2-4x+6y-3=0$ rappresenta una circonferenza e determina eventualmente centro e raggio.
\begin{solution}
Sì: \ $C(2,-3)$, \ $r=4$.
\end{solution}
\begin{explanation}
L'equazione $x^2+y^2+ax+by+c=0$ rappresenta una circonferenza reale se $r^2=\left(\tfrac a2\right)^2+\left(\tfrac b2\right)^2-c>0$; in tal caso il centro è $C\left(-\tfrac a2,\,-\tfrac b2\right)$ e il raggio $r=\sqrt{\left(\tfrac a2\right)^2+\left(\tfrac b2\right)^2-c}$.
Qui $a=-4$, $b=6$, $c=-3$.
\[\begin{aligned}
C &= \left(-\dfrac{-4}{2},\,-\dfrac62\right)=(2,-3)\\
r^2 &= 2^2+(-3)^2-(-3)=4+9+3=16>0 \;\Rightarrow\; r=4
\end{aligned}\]
\end{explanation}
\end{question}

\begin{question}[ID=1.2, subtitle={Verifica di una circonferenza non reale}, difficulty=facile, tags={circonferenza}]
Verifica se l'equazione $x^2+y^2+2x-8y+20=0$ rappresenta una circonferenza.
\begin{solution}
No: non rappresenta una circonferenza reale ($r^2=-3<0$).
\end{solution}
\begin{explanation}
L'equazione $x^2+y^2+ax+by+c=0$ rappresenta una circonferenza reale se $r^2=\left(\tfrac a2\right)^2+\left(\tfrac b2\right)^2-c>0$; in tal caso il centro è $C\left(-\tfrac a2,\,-\tfrac b2\right)$ e il raggio $r=\sqrt{\left(\tfrac a2\right)^2+\left(\tfrac b2\right)^2-c}$.
$a=2$, $b=-8$, $c=20$:
\[ r^2=1^2+(-4)^2-20=1+16-20=-3<0 \]
Nessun punto del piano soddisfa l'equazione.
\end{explanation}
\end{question}

\begin{question}[ID=1.3, subtitle={Valori di $k$ per la forma canonica della circonferenza}, difficulty=media, tags={circonferenza, parametri}]
Studia per quali valori di $k$ l'equazione $x^2+y^2+(k-1)x+2y-5=0$ rappresenta una circonferenza.
\begin{solution}
Per ogni $k\in\mathbb{R}$.
\end{solution}
\begin{explanation}
L'equazione $x^2+y^2+ax+by+c=0$ rappresenta una circonferenza reale se $r^2=\left(\tfrac a2\right)^2+\left(\tfrac b2\right)^2-c>0$; in tal caso il centro è $C\left(-\tfrac a2,\,-\tfrac b2\right)$ e il raggio $r=\sqrt{\left(\tfrac a2\right)^2+\left(\tfrac b2\right)^2-c}$.
$a=k-1$, $b=2$, $c=-5$:
\[ r^2=\left(\dfrac{k-1}{2}\right)^2+1+5=\dfrac{(k-1)^2}{4}+6 \]
È somma di un quadrato e di un numero positivo: sempre $>0$.
\end{explanation}
\end{question}

\begin{question}[ID=1.4, subtitle={Intervallo di $k$ per cui esiste la circonferenza}, difficulty=media, tags={circonferenza, parametri}]
Studia per quali valori di $k$ l'equazione $x^2+y^2+2(k-2)x+(k+1)y+1=0$ rappresenta una circonferenza.
\begin{solution}
Per ogni $k\in\mathbb{R}$.
\end{solution}
\begin{explanation}
L'equazione $x^2+y^2+ax+by+c=0$ rappresenta una circonferenza reale se $r^2=\left(\tfrac a2\right)^2+\left(\tfrac b2\right)^2-c>0$; in tal caso il centro è $C\left(-\tfrac a2,\,-\tfrac b2\right)$ e il raggio $r=\sqrt{\left(\tfrac a2\right)^2+\left(\tfrac b2\right)^2-c}$.
$a=2(k-2)$, $b=k+1$, $c=1$:
\[\begin{aligned}
r^2 &= (k-2)^2+\dfrac{(k+1)^2}{4}-1>0\\
4(k^2-4k+4)+(k^2+2k+1)-4 &> 0 && \textcolor{gray}{\text{moltiplico per $4$}}\\
5k^2-14k+13 &> 0
\end{aligned}\]
Il discriminante è $\tfrac\Delta4=49-65=-16<0$ e il coefficiente di $k^2$ è positivo: la disequazione è sempre verificata.
\end{explanation}
\end{question}

\begin{question}[ID=2.8, subtitle={Circonferenza per $A(1,0)$ e $B(0,1)$ con centro su $x=y$}, difficulty=media, tags={circonferenza}]
Trova la circonferenza passante per $A(1,0)$, $B(0,1)$ con centro sulla retta $x=y$.
\begin{solution}
Infinite circonferenze: $(x-a)^2+(y-a)^2=2a^2-2a+1$, per ogni $a\in\mathbb{R}$. \ Quella di raggio minimo (con $AB$ come diametro) è $\left(x-\tfrac12\right)^2+\left(y-\tfrac12\right)^2=\tfrac12$.
\end{solution}
\begin{explanation}
Il centro sta sulla retta $y=x$: $C(a,a)$. Impongo $|CA|=|CB|$:
\[\begin{aligned}
(1-a)^2+a^2 &= a^2+(1-a)^2
\end{aligned}\]
L'uguaglianza è vera per \textbf{ogni} $a$: infatti la retta $y=x$ è proprio l'asse del segmento $AB$, quindi tutti i suoi punti sono equidistanti da $A$ e $B$. Il problema ha infinite soluzioni:
\[ r^2=(1-a)^2+a^2=2a^2-2a+1 \]
Il raggio è minimo per $a=\tfrac12$ (vertice della parabola $2a^2-2a+1$), cioè quando il centro è il punto medio di $AB$.
\end{explanation}
\end{question}

Secante
\end{solution}
\begin{explanation}
Posizione retta–circonferenza: si confronta la distanza $d$ del centro dalla retta con il raggio $r$: \ $d<r$ secante, $d=r$ tangente, $d>r$ esterna. La distanza di $(x_0,y_0)$ da $ax+by+c=0$ è $d=\dfrac{|ax_0+by_0+c|}{\sqrt{a^2+b^2}}$.
Centro $O(0,0)$, $r=3$. La retta $y=2$ dista $2$ dall'origine: $d=2<3$, secante (incontra la circonferenza in $(\pm\sqrt5,\,2)$).
\end{explanation}

L'equazione $x^2+y^2+ax+by+c=0$ rappresenta una circonferenza reale se $r^2=\left(\tfrac a2\right)^2+\left(\tfrac b2\right)^2-c>0$; in tal caso il centro è $C\left(-\tfrac a2,\,-\tfrac b2\right)$ e il raggio $r=\sqrt{\left(\tfrac a2\right)^2+\left(\tfrac b2\right)^2-c}$.
\begin{enumerate}
    \item $a=2$, $b=-4$, $c=k$: \ $C(-1,2)$, \ $r^2=1+4-k=5-k$, positivo per $k<5$.
    \item Sostituisco $O(0,0)$: $0+0+0-0+k=0 \;\Rightarrow\; k=0$.
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=6.2, subtitle={Fascio $x^2+y^2-4x+k(x^2+y^2-4y)=0$: centro e degenere}, difficulty=media, tags={fasci, circonferenze}]
Considera il fascio $x^2+y^2-4x+k(x^2+y^2-4y)=0$.
\begin{enumerate}
\item Determina il centro del fascio.
\item Trova $k$ per cui la circonferenza è tangente all'asse $x$.
\item Studia per quali $k$ la circonferenza è degenere.
\end{enumerate}
\begin{solution}
\begin{enumerate}
    \item Centro $\left(\dfrac{2}{1+k},\ \dfrac{2k}{1+k}\right)$ \ ($k\neq-1$); i centri stanno sulla retta $x+y=2$.
    \item Nessun valore (finito) di $k$; l'unica circonferenza del fascio tangente all'asse $x$ è la generatrice $x^2+y^2-4y=0$.
    \item Degenere per $k=-1$: la retta $y=x$ (asse radicale).
\end{enumerate}
\end{solution}
\begin{explanation}
In un fascio $\gamma_1+k\gamma_2=0$ si raccolgono i termini: $(1+k)(x^2+y^2)+\dots=0$. Per $k\neq-1$ si divide per $1+k$ e si leggono centro e raggio; per $k=-1$ i termini di secondo grado si annullano e si ottiene una retta (asse radicale). I \textbf{punti base} sono i punti comuni alle due generatrici: appartengono a tutte le circonferenze del fascio.
\begin{enumerate}
    \item \[\begin{aligned}
    (1+k)(x^2+y^2)-4x-4ky &= 0\\
    x^2+y^2-\dfrac{4}{1+k}x-\dfrac{4k}{1+k}y &= 0 && \textcolor{gray}{\text{divido per $1+k$}}\\
    C &= \left(\dfrac{2}{1+k},\ \dfrac{2k}{1+k}\right)
    \end{aligned}\]
    Sommando le coordinate: $\dfrac{2+2k}{1+k}=2$, quindi i centri stanno sulla retta $x+y=2$ (l'asse dei centri).
    \item Tutte le circonferenze passano per l'origine (non c'è termine noto). Tangente all'asse $x$ vuol dire $|y_C|=r$:
    \[\begin{aligned}
    r^2 &= \dfrac{4}{(1+k)^2}+\dfrac{4k^2}{(1+k)^2}\\
    \dfrac{4k^2}{(1+k)^2} &= \dfrac{4+4k^2}{(1+k)^2} \;\Rightarrow\; 0=4
    \end{aligned}\]
    Impossibile. Solo la seconda generatrice $x^2+y^2-4y=0$ (centro $(0,2)$, raggio $2$), che si ottiene solo al limite per $k\to\infty$, è tangente all'asse $x$ nell'origine.
    \item Per $k=-1$ spariscono i termini di secondo grado: $-4x+4y=0$, cioè $y=x$.
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=6.3, subtitle={Fascio $(x^2+y^2-2x)+k(x^2+y^2+2y)=0$: punti base e asse radicale}, difficulty=media, tags={fasci, circonferenze}]
Considera il fascio $(x^2+y^2-2x) + k(x^2+y^2+2y)=0$.
\begin{enumerate}
\item Determina i punti base del fascio.
\item Trova l'asse radicale.
\item Determina per quali $k$ la circonferenza passa per $P(1,-1)$.
\end{enumerate}
\begin{solution}
\begin{enumerate}
    \item Punti base $O(0,0)$ e $(1,-1)$.
    \item Asse radicale: $x+y=0$.
    \item $P(1,-1)$ è un punto base: appartiene a \textbf{tutte} le circonferenze del fascio, per ogni $k$.
\end{enumerate}
\end{solution}
\begin{explanation}
In un fascio $\gamma_1+k\gamma_2=0$ si raccolgono i termini: $(1+k)(x^2+y^2)+\dots=0$. Per $k\neq-1$ si divide per $1+k$ e si leggono centro e raggio; per $k=-1$ i termini di secondo grado si annullano e si ottiene una retta (asse radicale). I \textbf{punti base} sono i punti comuni alle due generatrici: appartengono a tutte le circonferenze del fascio.
\begin{enumerate}
    \item Metto a sistema le due generatrici e sottraggo:
    \[\begin{aligned}
    (x^2+y^2-2x)-(x^2+y^2+2y) &= 0 \;\Rightarrow\; y=-x\\
    x^2+x^2-2x &= 0 \;\Rightarrow\; 2x(x-1)=0 && \textcolor{gray}{\text{sostituisco nella prima}}
    \end{aligned}\]
    $x=0 \Rightarrow (0,0)$; \quad $x=1 \Rightarrow (1,-1)$.
    \item Per $k=-1$: $-2x-2y=0$, cioè $x+y=0$ (passa per i due punti base).
    \item Sostituisco $P(1,-1)$: \ $(1+1-2)+k(1+1-2)=0+k\cdot0=0$, vero per ogni $k$.
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=6.4, subtitle={Fascio $x^2+y^2-6x+k(x^2+y^2-2y)=0$: centro e tangenza}, difficulty=media, tags={fasci, circonferenze}]
Considera il fascio $x^2+y^2-6x+k(x^2+y^2-2y)=0$.
\begin{enumerate}
\item Determina centro e raggio in funzione di $k$.
\item Trova $k$ per cui la circonferenza è tangente alla retta $x=0$.
\item Determina per quali $k$ la circonferenza ha area $4\pi$.
\end{enumerate}
\begin{solution}
\begin{enumerate}
    \item $C\left(\dfrac{3}{1+k},\ \dfrac{k}{1+k}\right)$, \ $r^2=\dfrac{9+k^2}{(1+k)^2}$ \ ($k\neq-1$)
    \item $k=0$
    \item $k=\dfrac{-4\pm\sqrt{31}}{3}$
\end{enumerate}
\end{solution}
\begin{explanation}
In un fascio $\gamma_1+k\gamma_2=0$ si raccolgono i termini: $(1+k)(x^2+y^2)+\dots=0$. Per $k\neq-1$ si divide per $1+k$ e si leggono centro e raggio; per $k=-1$ i termini di secondo grado si annullano e si ottiene una retta (asse radicale). I \textbf{punti base} sono i punti comuni alle due generatrici: appartengono a tutte le circonferenze del fascio.
\begin{enumerate}
    \item \[\begin{aligned}
    (1+k)(x^2+y^2)-6x-2ky &= 0\\
    x^2+y^2-\dfrac{6}{1+k}x-\dfrac{2k}{1+k}y &= 0\\
    C=\left(\dfrac{3}{1+k},\ \dfrac{k}{1+k}\right), &\quad r^2=\dfrac{9+k^2}{(1+k)^2}
    \end{aligned}\]
    \item Tangente a $x=0$: la distanza del centro dall'asse $y$, cioè $|x_C|$, deve valere $r$:
    \[\begin{aligned}
    \dfrac{9}{(1+k)^2} &= \dfrac{9+k^2}{(1+k)^2}\\
    k^2 &= 0 \;\Rightarrow\; k=0
    \end{aligned}\]
    Per $k=0$: $(x-3)^2+y^2=9$, tangente all'asse $y$ nell'origine.
    \item Area $\pi r^2=4\pi \Rightarrow r^2=4$:
    \[\begin{aligned}
    9+k^2 &= 4(1+k)^2\\
    9+k^2 &= 4+8k+4k^2\\
    3k^2+8k-5 &= 0\\
    k &= \dfrac{-8\pm\sqrt{64+60}}{6}=\dfrac{-4\pm\sqrt{31}}{3}
    \end{aligned}\]
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=6.5, subtitle={Fascio $x^2+y^2+2x+k(x^2+y^2-4y)=0$: luogo dei centri}, difficulty=difficile, tags={fasci, circonferenze}]
Considera il fascio $x^2+y^2+2x+k(x^2+y^2-4y)=0$.
\begin{enumerate}
\item Determina il luogo dei centri al variare di $k$.
\item Trova il valore di $k$ per cui la circonferenza è tangente alla circonferenza $x^2+y^2=1$.
\item Determina per quali $k$ il centro si trova nel primo quadrante.
\end{enumerate}
\begin{solution}
\begin{enumerate}
    \item Centri $\left(-\dfrac{1}{1+k},\ \dfrac{2k}{1+k}\right)$: il luogo è la retta $y=2x+2$ (privata del punto $(0,2)$).
    \item Nessun valore di $k$: le circonferenze del fascio sono sempre secanti a $x^2+y^2=1$.
    \item $k<-1$
\end{enumerate}
\end{solution}
\begin{explanation}
In un fascio $\gamma_1+k\gamma_2=0$ si raccolgono i termini: $(1+k)(x^2+y^2)+\dots=0$. Per $k\neq-1$ si divide per $1+k$ e si leggono centro e raggio; per $k=-1$ i termini di secondo grado si annullano e si ottiene una retta (asse radicale). I \textbf{punti base} sono i punti comuni alle due generatrici: appartengono a tutte le circonferenze del fascio.
\begin{enumerate}
    \item \[\begin{aligned}
    x^2+y^2+\dfrac{2}{1+k}x-\dfrac{4k}{1+k}y &= 0 && \textcolor{gray}{\text{divido per $1+k$}}\\
    x_C=-\dfrac{1}{1+k}, &\quad y_C=\dfrac{2k}{1+k}
    \end{aligned}\]
    Elimino $k$: \ $y_C=\dfrac{2(1+k)-2}{1+k}=2-\dfrac{2}{1+k}=2+2x_C$. I centri stanno sulla retta $y=2x+2$, che passa per i centri delle generatrici $(-1,0)$ e $(0,2)$.
    \item Punti base: sottraendo le generatrici, $2x+4y=0$, cioè $x=-2y$; sostituendo nella seconda si trovano $O(0,0)$ e $\left(-\tfrac85,\tfrac45\right)$. \\
    Ogni circonferenza del fascio passa per $O$, che è \textbf{interno} alla circonferenza unitaria, e per $\left(-\tfrac85,\tfrac45\right)$, che è \textbf{esterno} (dista $\tfrac{\sqrt{80}}{5}\approx1{,}79$ da $O$). Una curva chiusa che passa per un punto interno e uno esterno deve attraversarla: non può esserle tangente.
    \item Primo quadrante: $x_C>0$ e $y_C>0$.
    \[\begin{aligned}
    -\dfrac{1}{1+k}>0 &\;\Rightarrow\; 1+k<0 \;\Rightarrow\; k<-1\\
    \dfrac{2k}{1+k}>0 &\;\Rightarrow\; k<-1 \;\vee\; k>0
    \end{aligned}\]
    Entrambe: $k<-1$.
\end{enumerate}
\end{explanation}
\end{question}
