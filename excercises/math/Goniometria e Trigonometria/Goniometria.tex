\begin{question}[ID=1.1, subtitle={Formule di addizione con $\sin\alpha=2/3$ e $\sin\beta=3/5$}, tags={goniometria, formule di addizione}, difficulty=media]
Sapendo che $\sin \alpha = \dfrac{2}{3}$ e $\sin \beta = \dfrac{3}{5}$, con $0 < \alpha < \dfrac{\pi}{2}$ e $\dfrac{\pi}{2} < \beta < \pi$, calcola:
\begin{multicols}{2}
\begin{enumerate}
\item $\sin(\alpha + \beta)$
\item $\cos(\alpha - \beta)$
\item $\sin\dfrac{\alpha}{2}$
\item $\tan\dfrac{\beta}{2}$
\end{enumerate}
\end{multicols}
\begin{solution}
\begin{enumerate}
    \item $\dfrac{3\sqrt5-8}{15}$
    \item $\dfrac{6-4\sqrt5}{15}$
    \item $\sqrt{\dfrac{3-\sqrt5}{6}}$
    \item $3$
\end{enumerate}
\end{solution}
\begin{explanation}
Formule: \ $\sin(\alpha\pm\beta)=\sin\alpha\cos\beta\pm\cos\alpha\sin\beta$; \ $\cos(\alpha\pm\beta)=\cos\alpha\cos\beta\mp\sin\alpha\sin\beta$; \ bisezione $\sin\tfrac\alpha2=\pm\sqrt{\tfrac{1-\cos\alpha}{2}}$, $\cos\tfrac\alpha2=\pm\sqrt{\tfrac{1+\cos\alpha}{2}}$, $\tan\tfrac\alpha2=\tfrac{1-\cos\alpha}{\sin\alpha}$. \\
Prima si ricavano seno e coseno mancanti con $\sin^2+\cos^2=1$, scegliendo il segno in base al quadrante; il segno nelle formule di bisezione dipende dal quadrante di $\tfrac\alpha2$.
\textbf{Funzioni mancanti.}
\[\begin{aligned}
\cos\alpha &= +\sqrt{1-\dfrac49}=\dfrac{\sqrt5}{3} && \textcolor{gray}{\text{$\alpha$ nel I quadrante}}\\
\cos\beta &= -\sqrt{1-\dfrac{9}{25}}=-\dfrac45 && \textcolor{gray}{\text{$\beta$ nel II quadrante: coseno negativo}}
\end{aligned}\]
\begin{enumerate}
    \item \[ \sin(\alpha+\beta)=\dfrac23\cdot\left(-\dfrac45\right)+\dfrac{\sqrt5}{3}\cdot\dfrac35=\dfrac{-8+3\sqrt5}{15} \]
    \item \[ \cos(\alpha-\beta)=\dfrac{\sqrt5}{3}\cdot\left(-\dfrac45\right)+\dfrac23\cdot\dfrac35=\dfrac{-4\sqrt5+6}{15} \]
    \item $0<\tfrac\alpha2<\tfrac\pi4$: seno positivo.
    \[ \sin\dfrac\alpha2=\sqrt{\dfrac{1-\frac{\sqrt5}{3}}{2}}=\sqrt{\dfrac{3-\sqrt5}{6}} \]
    \item $\tfrac\pi4<\tfrac\beta2<\tfrac\pi2$: I quadrante.
    \[ \tan\dfrac\beta2=\dfrac{1-\cos\beta}{\sin\beta}=\dfrac{1+\frac45}{\frac35}=\dfrac{9}{5}\cdot\dfrac53=3 \]
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=2.1, subtitle={Semplificazione con formule di duplicazione e prostaferesi}, tags={goniometria, formule di duplicazione, prostaferesi}, difficulty=facile]
Semplifica le seguenti espressioni:
\begin{multicols}{2}
\begin{enumerate}
\item $\dfrac{1 + \cos 2\alpha}{\cos \alpha}$
\item $\sin\!\left(2\alpha + \dfrac{\pi}{4}\right) - \cos\!\left(2\alpha + \dfrac{\pi}{4}\right)$
\item $\sin 3\alpha + \sin \alpha$
\item $\cos 5\alpha - \cos 3\alpha$
\end{enumerate}
\end{multicols}
\begin{solution}
\begin{enumerate}
    \item $2\cos\alpha$
    \item $\sqrt2\sin2\alpha$
    \item $2\sin2\alpha\cos\alpha$
    \item $-2\sin4\alpha\sin\alpha$
\end{enumerate}
\end{solution}
\begin{explanation}
Duplicazione: $1+\cos2\alpha=2\cos^2\alpha$, \ $1-\cos2\alpha=2\sin^2\alpha$, \ $\sin2\alpha=2\sin\alpha\cos\alpha$. \\
Prostaferesi: \ $\sin p+\sin q=2\sin\tfrac{p+q}{2}\cos\tfrac{p-q}{2}$, \ $\sin p-\sin q=2\cos\tfrac{p+q}{2}\sin\tfrac{p-q}{2}$, \\
$\cos p+\cos q=2\cos\tfrac{p+q}{2}\cos\tfrac{p-q}{2}$, \ $\cos p-\cos q=-2\sin\tfrac{p+q}{2}\sin\tfrac{p-q}{2}$.
\begin{enumerate}
    \item \[ \dfrac{1+\cos2\alpha}{\cos\alpha}=\dfrac{2\cos^2\alpha}{\cos\alpha}=2\cos\alpha \]
    \item Pongo $u=2\alpha+\tfrac\pi4$ e uso $\sin u-\cos u=\sqrt2\sin\left(u-\tfrac\pi4\right)$ (che si verifica con la formula di sottrazione):
    \[ \sqrt2\sin\left(2\alpha+\dfrac\pi4-\dfrac\pi4\right)=\sqrt2\sin2\alpha \]
    \item $p=3\alpha$, $q=\alpha$: \ $2\sin\tfrac{4\alpha}{2}\cos\tfrac{2\alpha}{2}=2\sin2\alpha\cos\alpha$.
    \item $-2\sin\tfrac{8\alpha}{2}\sin\tfrac{2\alpha}{2}=-2\sin4\alpha\sin\alpha$.
\end{enumerate}
\end{explanation}
\end{question}

\end{enumerate}

\begin{explanation}

    \item \[\begin{aligned}
    2x+\dfrac\pi4 &= \dfrac\pi3+k\pi && \textcolor{gray}{\text{$\tan\dfrac\pi3=\sqrt3$}}\\

    \end{aligned}\]

    \[ \sin x\left(\sqrt3\sin x-\cos x\right)=0 \]

    x-\dfrac\pi3 &= \dfrac\pi2+2k\pi \;\Rightarrow\; x=\dfrac56\pi+2k\pi
    \end{aligned}\]
\end{enumerate}
\end{explanation}
\end{question}

    \item Esprimo tutto in seno con $\cos^2x=1-\sin^2x$:
    \[\begin{aligned}
    2(1-\sin^2x)-\sin x &= 0\\
    2\sin^2x+\sin x-2 &= 0\\
    \sin x &= \dfrac{-1\pm\sqrt{17}}{4}
    \end{aligned}\]
    $\tfrac{-1-\sqrt{17}}{4}\approx-1{,}28<-1$: impossibile. Resta $\sin x=\tfrac{\sqrt{17}-1}{4}\approx0{,}78$, che ha due soluzioni in un giro (I e II quadrante).
    \item $\cos x=0$ non è soluzione; divido per $\cos x$: $\tan x=-2$.
\end{enumerate}
\end{explanation}
\end{question}


\begin{question}[ID=fac.1, subtitle={Interferenza di onde: sovrapposizione e ampiezza}, tags={goniometria, fisica, interferenza}, difficulty=difficile]
\textbf{FAC)} In un ondoscopio due sorgenti d'onda $S_1$ e $S_2$ producono onde con la stessa ampiezza $A$, lo stesso periodo $T$ e la stessa lunghezza d'onda $\lambda$.
Lo spostamento trasversale che ognuna di esse produce è descritto dalle funzioni d'onda:
\[
y_1 = A \sin \left[ 2\pi \left( \frac{x_1}{\lambda} - \frac{t}{T} \right) \right], \qquad
y_2 = A \sin \left[ 2\pi \left( \frac{x_2}{\lambda} - \frac{t}{T} \right) \right]
\]
dove $x_1$ e $x_2$ rappresentano i cammini percorsi dalle due onde.
In un punto $P$ dell'ondoscopio le due onde si sovrappongono.

\begin{enumerate}[label=\alph*)]
  \item Dimostra che la sovrapposizione è un'onda e che l'ampiezza non dipende dal tempo
  \item Poni $\Delta x = x_1 - x_2$ e dimostra che l'ampiezza del moto è $0$ quando
  \[
  \Delta x = (2n + 1)\frac{\lambda}{2}, \qquad n \in \mathbb{Z},
  \]
  mentre è uguale a $2A$
  quando $\Delta x = n\lambda$.
\end{enumerate}
\begin{solution}
\begin{enumerate}[label=\alph*)]
    \item $y=2A\cos\left(\dfrac{\pi\,\Delta x}{\lambda}\right)\sin\left[2\pi\left(\dfrac{x_1+x_2}{2\lambda}-\dfrac tT\right)\right]$: un'onda di periodo $T$ con ampiezza $\left|2A\cos\dfrac{\pi\,\Delta x}{\lambda}\right|$, indipendente da $t$.
    \item Ampiezza $0$ per $\Delta x=(2n+1)\tfrac\lambda2$; \ ampiezza $2A$ per $\Delta x=n\lambda$.
\end{enumerate}
\end{solution}
\begin{explanation}
\begin{enumerate}[label=\alph*)]
    \item Chiamo $\theta_1$ e $\theta_2$ gli argomenti dei due seni e applico la prostaferesi $\sin\theta_1+\sin\theta_2=2\sin\tfrac{\theta_1+\theta_2}{2}\cos\tfrac{\theta_1-\theta_2}{2}$:
    \[\begin{aligned}
    \dfrac{\theta_1+\theta_2}{2} &= 2\pi\left(\dfrac{x_1+x_2}{2\lambda}-\dfrac tT\right) && \textcolor{gray}{\text{il termine in $t$ resta}}\\
    \dfrac{\theta_1-\theta_2}{2} &= \dfrac{\pi(x_1-x_2)}{\lambda}=\dfrac{\pi\,\Delta x}{\lambda} && \textcolor{gray}{\text{il termine in $t$ si elimina}}\\
    y &= 2A\cos\dfrac{\pi\,\Delta x}{\lambda}\cdot\sin\left[2\pi\left(\dfrac{x_1+x_2}{2\lambda}-\dfrac tT\right)\right]
    \end{aligned}\]
    Il secondo fattore è un'onda sinusoidale con lo stesso periodo $T$; il primo non contiene $t$, quindi fa da ampiezza costante nel tempo.
    \item \textbf{Ampiezza nulla}:
    \[\begin{aligned}
    \cos\dfrac{\pi\,\Delta x}{\lambda} &= 0\\
    \dfrac{\pi\,\Delta x}{\lambda} &= \dfrac\pi2+n\pi \;\Rightarrow\; \Delta x=(2n+1)\dfrac\lambda2 && \textcolor{gray}{\text{interferenza distruttiva}}
    \end{aligned}\]
    \textbf{Ampiezza massima $2A$}:
    \[\begin{aligned}
    \left|\cos\dfrac{\pi\,\Delta x}{\lambda}\right| &= 1\\
    \dfrac{\pi\,\Delta x}{\lambda} &= n\pi \;\Rightarrow\; \Delta x=n\lambda && \textcolor{gray}{\text{interferenza costruttiva}}
    \end{aligned}\]
\end{enumerate}
\end{explanation}
\end{question}
