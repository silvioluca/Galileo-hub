\begin{question}[ID=1.1, subtitle={Parabola $y=x^2$: vertice, asse, fuoco e direttrice}, difficulty=facile, tags={parabola, luogo}]
Considera la parabola di equazione $y=x^2$. Determina vertice, asse, concavità, fuoco e direttrice e rappresentala nel piano cartesiano.
\begin{solution}
$V(0,0)$; \ asse $x=0$; \ concavità verso l'alto; \ $F\left(0,\tfrac14\right)$; \ direttrice $y=-\tfrac14$.
\end{solution}
\begin{explanation}
Per la parabola $y=ax^2+bx+c$ (asse verticale), con $\Delta=b^2-4ac$:
\[ V\left(-\dfrac{b}{2a},\,-\dfrac{\Delta}{4a}\right), \quad F\left(-\dfrac{b}{2a},\,\dfrac{1-\Delta}{4a}\right), \quad \text{direttrice } y=-\dfrac{1+\Delta}{4a}, \quad \text{asse } x=-\dfrac{b}{2a}. \]
Fuoco e direttrice distano entrambi $\tfrac{1}{4|a|}$ dal vertice, da parti opposte; la concavità è verso l'alto se $a>0$.
$a=1$, $b=0$, $c=0$, $\Delta=0$:
\[\begin{aligned}
V &= (0,\,0)\\
F &= \left(0,\ \dfrac{1-0}{4}\right)=\left(0,\dfrac14\right)\\
d:\ y &= -\dfrac{1+0}{4}=-\dfrac14
\end{aligned}\]
$a=1>0$: concavità verso l'alto. Per disegnarla: $(\pm1,1)$, $(\pm2,4)$.
\end{explanation}
\end{question}

\begin{question}[ID=1.2, subtitle={Parabola $x=2y^2-3y+8$: elementi caratteristici}, difficulty=facile, tags={parabola}]
Data la parabola $x=2y^2-3y+8$, determina vertice, asse, fuoco e direttrice e rappresentala nel piano cartesiano.
\begin{solution}
$V\left(\tfrac{55}{8},\tfrac34\right)$; \ asse $y=\tfrac34$; \ $F\left(7,\tfrac34\right)$; \ direttrice $x=\tfrac{27}{4}$; \ concavità verso destra.
\end{solution}
\begin{explanation}
È una parabola ad asse \textbf{orizzontale} ($x$ in funzione di $y$): valgono le stesse formule scambiando i ruoli di $x$ e $y$. Qui $a=2$, $b=-3$, $c=8$:
\[\begin{aligned}
\Delta &= 9-64=-55\\
y_V &= -\dfrac{b}{2a}=\dfrac34 && \textcolor{gray}{\text{asse $y=\dfrac34$}}\\
x_V &= -\dfrac{\Delta}{4a}=\dfrac{55}{8} && \textcolor{gray}{\text{oppure sostituendo $y=\dfrac34$}}\\
x_F &= \dfrac{1-\Delta}{4a}=\dfrac{56}{8}=7\\
d:\ x &= -\dfrac{1+\Delta}{4a}=\dfrac{54}{8}=\dfrac{27}{4}
\end{aligned}\]
$a>0$: si apre verso destra. Fuoco e direttrice distano $\tfrac18$ dal vertice.
\end{explanation}
\end{question}

\begin{question}[ID=1.3, subtitle={Parabola $(x-2)^2=12(y+1)$: elementi caratteristici}, difficulty=media, tags={parabola}]
Data la parabola $(x-2)^2=12(y+1)$, determina vertice, asse, fuoco, direttrice e concavità e rappresentala nel piano cartesiano.
\begin{solution}
$V(2,-1)$; \ asse $x=2$; \ $F(2,2)$; \ direttrice $y=-4$; \ concavità verso l'alto.
\end{solution}
\begin{explanation}
È nella forma $(x-x_V)^2=4p\,(y-y_V)$, con vertice $V(x_V,y_V)$ e distanza fuoco–vertice $p$.
\[\begin{aligned}
V &= (2,-1), \qquad 4p=12 \;\Rightarrow\; p=3\\
F &= (2,\,-1+3)=(2,2)\\
d:\ y &= -1-3=-4
\end{aligned}\]
$p>0$: concavità verso l'alto.
\end{explanation}
\end{question}

\begin{question}[ID=1.4, subtitle={Luogo equidistante da $F(0,2)$ e dalla retta $y=-2$}, difficulty=media, tags={parabola, luogo}]
Determina il luogo dei punti equidistanti da $F(0,2)$ e dalla retta $y=-2$, trovandone equazione, vertice, asse e fuoco.
\begin{solution}
$x^2=8y$ \ (cioè $y=\tfrac18x^2$); \ $V(0,0)$; \ asse $x=0$; \ $F(0,2)$.
\end{solution}
\begin{explanation}
La parabola è il luogo dei punti $P(x,y)$ equidistanti dal fuoco $F$ e dalla direttrice $d$: si impone $|PF|=\text{dist}(P,d)$ e si elevano al quadrato entrambi i membri.
\[\begin{aligned}
\sqrt{x^2+(y-2)^2} &= |y+2| && \textcolor{gray}{\text{distanza da $F$ = distanza dalla retta}}\\
x^2+y^2-4y+4 &= y^2+4y+4 && \textcolor{gray}{\text{elevo al quadrato}}\\
x^2 &= 8y
\end{aligned}\]
Il vertice è a metà tra fuoco e direttrice: $V(0,0)$; asse $x=0$.
\end{explanation}
\end{question}

\begin{question}[ID=1.5, subtitle={Parabola $y=-x^2+4x-3$: forma canonica e fuoco}, difficulty=media, tags={parabola}]
Data la parabola $y=-x^2+4x-3$, determinane vertice, asse, forma canonica, fuoco e direttrice.
\begin{solution}
$y=-(x-2)^2+1$; \ $V(2,1)$; \ asse $x=2$; \ $F\left(2,\tfrac34\right)$; \ direttrice $y=\tfrac54$.
\end{solution}
\begin{explanation}
Per la parabola $y=ax^2+bx+c$ (asse verticale), con $\Delta=b^2-4ac$:
\[ V\left(-\dfrac{b}{2a},\,-\dfrac{\Delta}{4a}\right), \quad F\left(-\dfrac{b}{2a},\,\dfrac{1-\Delta}{4a}\right), \quad \text{direttrice } y=-\dfrac{1+\Delta}{4a}, \quad \text{asse } x=-\dfrac{b}{2a}. \]
Fuoco e direttrice distano entrambi $\tfrac{1}{4|a|}$ dal vertice, da parti opposte; la concavità è verso l'alto se $a>0$.
Completo il quadrato per ottenere la forma canonica:
\[\begin{aligned}
y &= -(x^2-4x)-3\\
&= -(x^2-4x+4)+4-3 && \textcolor{gray}{\text{aggiungo e tolgo $4$}}\\
&= -(x-2)^2+1
\end{aligned}\]
Vertice $(2,1)$. Con $a=-1$, fuoco e direttrice distano $\tfrac{1}{4}$ dal vertice; la concavità è verso il basso, quindi il fuoco sta \emph{sotto} il vertice:
\[ F=\left(2,\ 1-\dfrac14\right)=\left(2,\dfrac34\right), \qquad d:\ y=1+\dfrac14=\dfrac54 \]
\end{explanation}
\end{question}

\begin{question}[ID=2.1, subtitle={Posizione di $y=x^2-4$ e $y=-2x+4$ con corda}, difficulty=facile, tags={parabola, retta, corda}]
Data la parabola $y=x^2-4$ e la retta $y=-2x+4$, determina la posizione reciproca, i punti di intersezione e la lunghezza della corda individuata.
\begin{solution}
Secante; \ punti $(2,0)$ e $(-4,12)$; \ corda $6\sqrt5$.
\end{solution}
\begin{explanation}
Per la posizione retta–parabola si mette a sistema e si ottiene un'equazione di secondo grado: $\Delta>0$ secante (due punti), $\Delta=0$ tangente, $\Delta<0$ esterna.
\[\begin{aligned}
x^2-4 &= -2x+4\\
x^2+2x-8 &= 0\\
(x+4)(x-2) &= 0 \;\Rightarrow\; x=2,\ x=-4
\end{aligned}\]
Ordinate dalla retta: $y(2)=0$, $y(-4)=12$. Lunghezza della corda:
\[ \sqrt{(2+4)^2+(0-12)^2}=\sqrt{36+144}=\sqrt{180}=6\sqrt5 \]
\end{explanation}
\end{question}

\end{explanation}

\begin{question}[ID=3.2, subtitle={Tangenti a $y=-3x^2+2$ negli assi e area del triangolo}, difficulty=facile, tags={parabola, tangente}]
Determina le rette tangenti alla parabola di equazione $y = - 3x^2 + 2$ nei suoi punti d'intersezione con gli assi cartesiani e l'area del triangolo individuato da tali rette.
\begin{solution}
Tangenti: $y=2$ (in $(0,2)$), \ $y=-2\sqrt6\,x+4$ e $y=2\sqrt6\,x+4$ (in $\left(\pm\sqrt{\tfrac23},0\right)$); \ area del triangolo $\dfrac{\sqrt6}{3}\approx0{,}82$.
\end{solution}
\begin{explanation}
Una retta $y=mx+q$ è tangente alla parabola quando l'equazione risolvente ha $\Delta=0$. In alternativa, la pendenza della tangente nel punto di ascissa $x_0$ è $y'(x_0)=2ax_0+b$.
\textbf{Intersezioni con gli assi:} con l'asse $y$ in $(0,2)$; con l'asse $x$ per $3x^2=2$, cioè $x=\pm\sqrt{\tfrac23}$. \\
\textbf{Tangenti} (con $y'=-6x$):
\[\begin{aligned}
x_0=0: &\quad m=0 \;\Rightarrow\; y=2\\
x_0=\sqrt{\dfrac23}: &\quad m=-6\sqrt{\dfrac23}=-2\sqrt6 \;\Rightarrow\; y=-2\sqrt6\left(x-\sqrt{\dfrac23}\right)=-2\sqrt6\,x+4\\
x_0=-\sqrt{\dfrac23}: &\quad m=2\sqrt6 \;\Rightarrow\; y=2\sqrt6\,x+4 && \textcolor{gray}{\text{per simmetria}}
\end{aligned}\]
(Infatti $2\sqrt6\cdot\sqrt{\tfrac23}=2\sqrt4=4$.) \\
\textbf{Triangolo.} Le due tangenti oblique si incontrano in $(0,4)$; ciascuna incontra $y=2$ dove $2\sqrt6\,|x|=2$, cioè $x=\pm\tfrac{1}{\sqrt6}$.
\[ A=\dfrac12\cdot\dfrac{2}{\sqrt6}\cdot(4-2)=\dfrac{2}{\sqrt6}=\dfrac{\sqrt6}{3} \]
\end{explanation}
\end{question}

Scrivi l'equazione della retta tangente alla parabola di equazione $y = x^2 - 3x + 5$ e parallela alla retta $y = 2x$. Determina poi le coordinate del punto di contatto tra la retta e la parabola.
\begin{solution}
$y=2x-\tfrac54$; \ punto di contatto $\left(\tfrac52,\tfrac{15}{4}\right)$.
\end{solution}
\begin{explanation}
Una retta $y=mx+q$ è tangente alla parabola quando l'equazione risolvente ha $\Delta=0$. In alternativa, la pendenza della tangente nel punto di ascissa $x_0$ è $y'(x_0)=2ax_0+b$.
Una retta parallela a $y=2x$ ha pendenza $2$. Cerco il punto dove la derivata vale $2$:
\[\begin{aligned}
y' &= 2x-3=2 \;\Rightarrow\; x_0=\dfrac52\\
y_0 &= \dfrac{25}{4}-\dfrac{15}{2}+5=\dfrac{15}{4}\\
y &= 2\left(x-\dfrac52\right)+\dfrac{15}{4}=2x-\dfrac54
\end{aligned}\]
Verifica con $\Delta$: $x^2-3x+5=2x-\tfrac54 \Rightarrow x^2-5x+\tfrac{25}{4}=\left(x-\tfrac52\right)^2=0$.
\end{explanation}
\end{question}

\begin{question}[ID=3.4, subtitle={Tangente perpendicolare a $y=-2x$ alla parabola $y=-\frac12x^2+x$}, difficulty=media, tags={parabola, tangente}]
Determina la retta tangente alla parabola $y = -\dfrac{1}{2}x^2 + x$ perpendicolare a $y = -2x$.
\begin{solution}
$y=\tfrac12x+\tfrac18$; \ punto di contatto $\left(\tfrac12,\tfrac38\right)$.
\end{solution}
\begin{explanation}
Una retta $y=mx+q$ è tangente alla parabola quando l'equazione risolvente ha $\Delta=0$. In alternativa, la pendenza della tangente nel punto di ascissa $x_0$ è $y'(x_0)=2ax_0+b$.
Perpendicolare a $y=-2x$: pendenza $m=\tfrac12$ (le pendenze di rette perpendicolari hanno prodotto $-1$).
\[\begin{aligned}
y' &= -x+1=\dfrac12 \;\Rightarrow\; x_0=\dfrac12\\
y_0 &= -\dfrac12\cdot\dfrac14+\dfrac12=\dfrac38\\
y &= \dfrac12\left(x-\dfrac12\right)+\dfrac38=\dfrac12x+\dfrac18
\end{aligned}\]
\end{explanation}
\end{question}

\begin{question}[ID=3.5, subtitle={Tangente perpendicolare a $y=-2x$ alla parabola (variante)}, difficulty=difficile, tags={parabola, tangente}]
Determina la retta tangente alla parabola $y = -\dfrac{1}{2}x^2 + x$ perpendicolare a $y = -2x$.

$y=\tfrac12x+\tfrac18$; \ punto di contatto $\left(\tfrac12,\tfrac38\right)$.
\end{solution}
\begin{explanation}

\[\begin{aligned}

\end{aligned}\]

\begin{question}[ID=5.1, subtitle={Fascio $y=x^2-2kx+k-1$: punti base e proprietà}, difficulty=media, tags={parabola, fasci}]
Considera il fascio di parabole di equazione $y=x^2-2kx+k-1$ e determina i punti base e la caratteristica del fascio, la parabola passante per $P(1,1)$, quella avente asse di equazione $x=1$ e le eventuali parabole tangenti alla retta $y=0$.
\begin{solution}
Un solo punto base proprio $B\left(\tfrac12,-\tfrac34\right)$; \ asse $x=k$; \ per $P(1,1)$: $k=-1$ ($y=x^2+2x-2$); \ asse $x=1$: $k=1$ ($y=x^2-2x$); \ nessuna parabola tangente a $y=0$.
\end{solution}
\begin{explanation}
In un fascio $y=f(x)+k\,g(x)$ i \textbf{punti base} sono quelli che soddisfano l'equazione per ogni $k$: si annulla il coefficiente di $k$ e si sostituisce nel resto. Se $g$ è di primo grado c'è un solo punto base proprio.
Raccolgo $k$: \ $y=(x^2-1)+k(1-2x)$.
\begin{itemize}
    \item \textbf{Punti base}: $1-2x=0 \Rightarrow x=\tfrac12$, \ $y=\tfrac14-1=-\tfrac34$.
    \item \textbf{Asse}: $x_V=-\dfrac{b}{2a}=k$.
    \item \textbf{Per $P(1,1)$}: $1=1-2k+k-1 \Rightarrow k=-1$.
    \item \textbf{Asse $x=1$}: $k=1$.
    \item \textbf{Tangenza a $y=0$}: $x^2-2kx+k-1=0$ con $\tfrac\Delta4=k^2-k+1$, che ha discriminante $1-4<0$: è sempre positivo, quindi due intersezioni per ogni $k$.
\end{itemize}
\end{explanation}
\end{question}

\begin{question}[ID=5.2, subtitle={Fascio $y=x^2+kx-2k$: punti base e proprietà}, difficulty=media, tags={parabola, fasci}]

\begin{solution}
Un solo punto base proprio $B(2,4)$; \ per $P(2,0)$: impossibile; \ vertice sull'asse $y$: $k=0$ ($y=x^2$); \ tangenti a $y=-1$: $k=-4\pm2\sqrt5$.
\end{solution}
\begin{explanation}
In un fascio $y=f(x)+k\,g(x)$ i \textbf{punti base} sono quelli che soddisfano l'equazione per ogni $k$: si annulla il coefficiente di $k$ e si sostituisce nel resto. Se $g$ è di primo grado c'è un solo punto base proprio.

    \end{aligned}\]
\end{itemize}
\end{explanation}
\end{question}

\begin{question}[ID=5.5, subtitle={Fascio $y=x^2+kx+k-2$: punti base e tangenza}, difficulty=difficile, tags={parabola, fasci, parametrico}]
Considera il fascio $y=x^2+kx+k-2$ e determina i punti base, la parabola passante per $P(1,0)$, quella avente vertice sulla bisettrice $y=x$ e le parabole tangenti alla retta $y=-2$.
\begin{solution}
Un solo punto base proprio $B(-1,-1)$; \ per $P(1,0)$: $k=\tfrac12$; \ vertice su $y=x$: $k=2$ o $k=4$; \ tangenti a $y=-2$: $k=0$ o $k=4$.
\end{solution}
\begin{explanation}
In un fascio $y=f(x)+k\,g(x)$ i \textbf{punti base} sono quelli che soddisfano l'equazione per ogni $k$: si annulla il coefficiente di $k$ e si sostituisce nel resto. Se $g$ è di primo grado c'è un solo punto base proprio.
$y=(x^2-2)+k(x+1)$.
\begin{itemize}
    \item \textbf{Punti base}: $x=-1 \Rightarrow y=1-2=-1$.
    \item \textbf{Per $P(1,0)$}: $0=1+k+k-2 \Rightarrow k=\tfrac12$.
    \item \textbf{Vertice su $y=x$}:
    \[\begin{aligned}
    x_V &= -\dfrac k2, \qquad y_V=x_V^2+kx_V+k-2=-\dfrac{k^2}{4}+k-2\\
    -\dfrac k2 &= -\dfrac{k^2}{4}+k-2 && \textcolor{gray}{\text{impongo $y_V=x_V$}}\\
    k^2-6k+8 &= 0 \;\Rightarrow\; k=2,\ k=4 && \textcolor{gray}{\text{moltiplico per $4$}}
    \end{aligned}\]
    \item \textbf{Tangenza a $y=-2$}: $x^2+kx+k=0$, \ $\Delta=k^2-4k=k(k-4)=0 \Rightarrow k=0,\ k=4$.
\end{itemize}
\end{explanation}
\end{question}
