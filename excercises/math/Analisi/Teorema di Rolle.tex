\begin{question}[ID=1.1, subtitle={Teorema di Rolle per $f(x)=x^2-4x+3$ su $[1,3]$}, tags={rolle, teoremi}, difficulty=facile]
Verificare se la funzione \(f(x) = x^2 - 4x + 3\) soddisfa le ipotesi del teorema di Rolle nell'intervallo \([1;\ 3]\) e, in caso affermativo, trovare il punto \(c\) in cui \(f'(c) = 0\).
\begin{solution}
Ipotesi soddisfatte; \ $c=2$.
\end{solution}
\begin{explanation}
\textbf{Teorema di Rolle}: se $f$ è continua su $[a,b]$, derivabile in $(a,b)$ e $f(a)=f(b)$, esiste almeno un $c\in(a,b)$ con $f'(c)=0$ (tangente orizzontale). Le tre ipotesi vanno verificate tutte.
Polinomio: continuo e derivabile. Valori agli estremi:
\[ f(1)=1-4+3=0, \qquad f(3)=9-12+3=0 \]
Uguali: le ipotesi valgono. Cerco dove si annulla la derivata:
\[ f'(x)=2x-4=0 \;\Rightarrow\; c=2\in(1,3) \]
\end{explanation}
\end{question}

\begin{question}[ID=1.2, subtitle={Teorema di Rolle per $\sin(x)$ su $[0,\pi]$}, tags={rolle, teoremi}, difficulty=facile]
Sia \(f(x) = \sin(x)\). Verificare che \(f(0) = f(\pi)\) e trovare un valore \(c \in (0;\ \pi)\) tale che \(f'(c) = 0\).
\begin{solution}
$f(0)=f(\pi)=0$; \ $c=\dfrac\pi2$.
\end{solution}
\begin{explanation}
\textbf{Teorema di Rolle}: se $f$ è continua su $[a,b]$, derivabile in $(a,b)$ e $f(a)=f(b)$, esiste almeno un $c\in(a,b)$ con $f'(c)=0$ (tangente orizzontale). Le tre ipotesi vanno verificate tutte.
$\sin x$ è continua e derivabile ovunque, e $\sin0=\sin\pi=0$.
\[ f'(x)=\cos x=0 \;\Rightarrow\; c=\dfrac\pi2 \quad \text{(unico valore in $(0,\pi)$)} \]
\end{explanation}
\end{question}

\begin{question}[ID=1.3, subtitle={Teorema di Rolle per $f(x)=x^3-x$ su $[-1,1]$}, tags={rolle, teoremi}, difficulty=media]
Considera la funzione \(f(x) = x^3 - x\) nell'intervallo \([-1;\ 1]\). Mostrare che soddisfa le ipotesi del teorema di Rolle e determinare il valore di \(c\) in \((-1, 1)\) per cui \(f'(c) = 0\).
\begin{solution}
Ipotesi soddisfatte; \ $c=\pm\dfrac{1}{\sqrt3}$.
\end{solution}
\begin{explanation}
\textbf{Teorema di Rolle}: se $f$ è continua su $[a,b]$, derivabile in $(a,b)$ e $f(a)=f(b)$, esiste almeno un $c\in(a,b)$ con $f'(c)=0$ (tangente orizzontale). Le tre ipotesi vanno verificate tutte.
Polinomio, e $f(-1)=-1+1=0=f(1)$.
\[\begin{aligned}
f'(x) &= 3x^2-1=0\\
x^2 &= \dfrac13 \;\Rightarrow\; c=\pm\dfrac{1}{\sqrt3}\approx\pm0{,}58
\end{aligned}\]
Entrambi in $(-1,1)$: il teorema garantisce almeno un punto, qui sono due.
\end{explanation}
\end{question}

\begin{question}[ID=1.4, subtitle={Teorema di Rolle per $f(x)=e^{-x}\sin(x)$ su $[0,\pi]$}, tags={rolle, teoremi}, difficulty=media]
Applica il teorema di Rolle alla funzione \(f(x) = e^{-x}\sin(x)\) nell'intervallo \([0;\ \pi]\). Trovare il punto \(c\) in cui la derivata si annulla.
\begin{solution}
Ipotesi soddisfatte; \ $c=\dfrac\pi4$.
\end{solution}
\begin{explanation}
\textbf{Teorema di Rolle}: se $f$ è continua su $[a,b]$, derivabile in $(a,b)$ e $f(a)=f(b)$, esiste almeno un $c\in(a,b)$ con $f'(c)=0$ (tangente orizzontale). Le tre ipotesi vanno verificate tutte.
Prodotto di funzioni continue e derivabili; $f(0)=0$ e $f(\pi)=e^{-\pi}\cdot0=0$.
\[\begin{aligned}
f'(x) &= -e^{-x}\sin x+e^{-x}\cos x && \textcolor{gray}{\text{regola del prodotto}}\\
&= e^{-x}(\cos x-\sin x)\\
\cos c-\sin c &= 0 && \textcolor{gray}{\text{$e^{-x}>0$ non si annulla mai}}\\
\tan c &= 1 \;\Rightarrow\; c=\dfrac\pi4 && \textcolor{gray}{\text{unica soluzione in $(0,\pi)$}}
\end{aligned}\]
\end{explanation}
\end{question}

\begin{question}[ID=1.7, subtitle={Teorema di Rolle non applicabile: $f(x)=1/(x-1)$ su $[1/2,3]$}, tags={rolle, teoremi, non applicabile}, difficulty=media]
Applica il teorema di Rolle alla funzione \(f(x) = \dfrac{1}{x-1}\), nell'intervallo \(\bigl[\tfrac{1}{2};\ 3\bigr]\). Trovare il punto \(c\) in cui la derivata si annulla.
\begin{solution}
Il teorema \textbf{non} è applicabile: $f$ non è definita in $x=1\in\left[\tfrac12,3\right]$.

(Inoltre $f\left(\tfrac12\right)=-2\neq f(3)=\tfrac12$.) In effetti $f'(x)=-\dfrac{1}{(x-1)^2}$ non si annulla mai.
\end{explanation}
\end{question}

\begin{question}[ID=2.1, subtitle={Parametro $a$ per Rolle: $ax^3+(a-1)x$ su $[-1,2]$}, tags={rolle, teoremi, parametrico}, difficulty=difficile]
Trova i valori dei parametri in modo che la seguente funzione $f(x) = ax^3 + (a-1)x$ verifichi il teorema di Rolle nell'intervallo $[-1;\ 2]$. Determina il punto (o i punti) la cui esistenza è assicurata dal teorema.
\begin{solution}
$a=\dfrac14$; \ $c=1$.
\end{solution}
\begin{explanation}
\textbf{Teorema di Rolle}: se $f$ è continua su $[a,b]$, derivabile in $(a,b)$ e $f(a)=f(b)$, esiste almeno un $c\in(a,b)$ con $f'(c)=0$ (tangente orizzontale). Le tre ipotesi vanno verificate tutte.
Continuità e derivabilità valgono per ogni $a$ (polinomio). Impongo $f(-1)=f(2)$:
\[\begin{aligned}
f(-1) &= -a-(a-1)=1-2a\\
f(2) &= 8a+2(a-1)=10a-2\\
1-2a &= 10a-2 \;\Rightarrow\; a=\dfrac14
\end{aligned}\]
Con $a=\tfrac14$: \ $f(x)=\tfrac14x^3-\tfrac34x$.
\[ f'(x)=\dfrac34x^2-\dfrac34=0 \;\Rightarrow\; x=\pm1 \]
$x=-1$ è un estremo dell'intervallo: resta $c=1\in(-1,2)$.
\end{explanation}
\end{question}

\begin{question}[ID=2.2, subtitle={Parametro $a$ per Rolle: $(a+2)x^4-2ax+4$ su $[-1,1]$}, tags={rolle, teoremi, parametrico}, difficulty=difficile]
Trova i valori dei parametri in modo che la seguente funzione $f(x) = (a+2)x^4 -2ax +4$ verifichi il teorema di Rolle nell'intervallo $[-1;\ 1]$. Determina il punto (o i punti) la cui esistenza è assicurata dal teorema.
\begin{solution}
$a=0$; \ $c=0$.
\end{solution}
\begin{explanation}
\textbf{Teorema di Rolle}: se $f$ è continua su $[a,b]$, derivabile in $(a,b)$ e $f(a)=f(b)$, esiste almeno un $c\in(a,b)$ con $f'(c)=0$ (tangente orizzontale). Le tre ipotesi vanno verificate tutte.
Polinomio per ogni $a$. Impongo $f(-1)=f(1)$:
\[\begin{aligned}
f(-1) &= (a+2)+2a+4=3a+6\\
f(1) &= (a+2)-2a+4=6-a\\
3a+6 &= 6-a \;\Rightarrow\; a=0
\end{aligned}\]
Con $a=0$: \ $f(x)=2x^4+4$, \ $f'(x)=8x^3=0 \;\Rightarrow\; c=0$.
\end{explanation}
\end{question}
