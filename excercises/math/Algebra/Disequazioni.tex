\begin{question}[ID=1.1, subtitle={Disequazioni di primo grado}, difficulty=facile, tags={disequazioni lineari}]
Risolvi le seguenti disequazioni di primo grado:
\begin{multicols}{2}
\begin{enumerate}[leftmargin=3.2cm]
    \item $x + 3 > 7$
    \item $x - 5 \leq 2$
    \item $2x > 8$
    \item $-3x \leq 9$
    \item $2x + 1 > 5$
    \item $3x - 4 \geq 2$
\end{enumerate}
\end{multicols}
\begin{solution}
\begin{enumerate}
    \item $x>4$
    \item $x\leq 7$
    \item $x>4$
    \item $x\geq -3$
    \item $x>2$
    \item $x\geq 2$
\end{enumerate}
\end{solution}
\end{question}

\begin{question}[ID=2.1, subtitle={Disequazioni frazionarie}, difficulty=facile, tags={disequazioni frazionarie}]
Risolvi le seguenti disequazioni frazionarie:
\begin{multicols}{2}
\begin{enumerate}[leftmargin=3.2cm]
    \item $\dfrac{x}{x-1} > 0$
    \item $\dfrac{x-2}{x+1} \geq 0$
    \item $\dfrac{2x+1}{x-3} > 0$
    \item $\dfrac{x+4}{2x-1} \leq 0$
    \item $\dfrac{x-1}{x+2} < 0$
    \item $\dfrac{3x-2}{x-2} \geq 0$
\end{enumerate}
\end{multicols}
\begin{solution}
\begin{enumerate}
    \item $x<0\vee x>1$
    \item $x\in(-\infty,-1)\cup[2,+\infty)$
    \item $x<-\tfrac12\vee x>3$
    \item $x\in[-4,\tfrac12)$
    \item $x\in(-2,1)$
    \item $x\in(-\infty,\tfrac23]\cup(2,+\infty)$
\end{enumerate}
\end{solution}
\end{question}

\begin{enumerate}

    \item $x\in[-3,3)$
    \item $x<-1$
    \item $x\in(-\infty,-3]\cup(1,+\infty)$
    \item $x\in(-\infty,1)\cup(2,+\infty)$
    \item $x\in(-2,\tfrac52]$
\end{enumerate}
\end{solution}
\end{question}

\begin{question}[ID=2.4, subtitle={Disequazioni frazionarie con più fattori}, difficulty=difficile, tags={disequazioni frazionarie}]
Risolvi le seguenti disequazioni frazionarie:
\begin{multicols}{2}
\begin{enumerate}[leftmargin=3.2cm]
    \item $\dfrac{3}{x-2}+\dfrac{1}{2x+1}>0$
    \item $\dfrac{1}{x+1}-\dfrac{3}{x-1}\leq 0$
    \item $\dfrac{x-1}{x+2}>\dfrac{2x+3}{x-1}$
    \item $\dfrac{2x+1}{x-2}-\dfrac{1}{x+1}\geq 2$
    \item $\dfrac{x+2}{x-1}+\dfrac{x-3}{x+2}\leq 1$
    \item $\dfrac{1}{x-2}-\dfrac{2}{x+1}\leq 1$
\end{enumerate}
\end{multicols}
\begin{solution}
\begin{enumerate}
    \item $x\in(-\tfrac12,-\tfrac17)\cup(2,+\infty)$
    \item $x\in[-2,-1)\cup(1,+\infty)$
    \item $x\in\left(\tfrac{-9-\sqrt{61}}{2},-2\right)\cup\left(\tfrac{-9+\sqrt{61}}{2},1\right)$
    \item $x\in[-\tfrac74,-1)\cup(2,+\infty)$
    \item $x\in(-2,1)$
    \item $x\in(-\infty,-\sqrt7]\cup(-1,2)\cup[\sqrt7,+\infty)$
\end{enumerate}
\end{solution}
\end{question}
