\begin{question}[ID=1.1, subtitle={Dominio di una funzione}, difficulty=facile, tags={funzioni, dominio}]
Qual è il dominio della funzione $f(x)=\frac{1}{x-3}$?

\begin{tabularx}{\linewidth}{@{}X X X X@{}}  
\boxed{A}\hspace{6pt} $\mathbb{R}$ &  
\boxed{B}\hspace{6pt} $\mathbb{R}\setminus\{3\}$ &  
\boxed{C}\hspace{6pt} $[3,+\infty)$ &  
\boxed{D}\hspace{6pt} $\mathbb{R}\setminus\{0\}$  
\end{tabularx}

\begin{solution}
Risposta $\boxed{\text{B}}$: \ $\mathbb{R}\setminus\{3\}$
\end{solution}
\begin{explanation}
L'unica condizione è che il denominatore non si annulli:
\[ x-3\neq0 \;\Rightarrow\; x\neq3 \]
Quindi $D=\mathbb{R}\setminus\{3\}$. \\
La A dimentica la condizione; la C confonde la frazione con una radice; la D esclude lo zero invece del $3$.
\end{explanation}
\end{question}

\begin{question}[ID=1.2, subtitle={Dominio di una funzione irrazionale}, difficulty=facile, tags={funzioni, dominio}]
Qual è il dominio della funzione $f(x)=\sqrt{x+2}$?

\begin{tabularx}{\linewidth}{@{}X X X X@{}}  
\boxed{A}\hspace{6pt} $x>-2$ &  
\boxed{B}\hspace{6pt} $x\geq-2$ &  
\boxed{C}\hspace{6pt} $x\leq-2$ &  
\boxed{D}\hspace{6pt} $\mathbb{R}$  
\end{tabularx}

\begin{solution}
Risposta $\boxed{\text{B}}$: \ $x\geq-2$
\end{solution}
\begin{explanation}
Il radicando di una radice quadrata deve essere \textbf{maggiore o uguale} a zero:
\[ x+2\geq0 \;\Rightarrow\; x\geq-2 \]
La A esclude $x=-2$, dove invece $\sqrt0=0$ esiste; la C ha il verso sbagliato; la D non pone condizioni.
\end{explanation}
\end{question}

\begin{question}[ID=1.3, subtitle={Dominio di una funzione razionale e irrazionale}, difficulty=media, tags={funzioni, dominio}]
Determina il dominio della funzione $f(x)=\frac{\sqrt{x-1}}{x-2}$.

\begin{tabularx}{\linewidth}{@{}X X X X@{}}  
\boxed{A}\hspace{6pt} $[1,+\infty)$ &  
\boxed{B}\hspace{6pt} $[1,+\infty)\setminus\{2\}$ &  
\boxed{C}\hspace{6pt} $(-\infty,1]\setminus\{2\}$ &  
\boxed{D}\hspace{6pt} $\mathbb{R}\setminus\{2\}$  
\end{tabularx}

\begin{solution}
Risposta $\boxed{\text{B}}$: \ $[1,+\infty)\setminus\{2\}$
\end{solution}
\begin{explanation}
Due condizioni da mettere a sistema:
\[ \begin{cases} x-1\geq0 & \text{radicando}\\ x-2\neq0 & \text{denominatore}\end{cases} \;\Rightarrow\; \begin{cases} x\geq1\\ x\neq2\end{cases} \]
La A dimentica il denominatore, la C ha il verso sbagliato per la radice, la D dimentica la radice.
\end{explanation}
\end{question}

\begin{question}[ID=1.4, subtitle={Dominio di una funzione con radice e frazione}, difficulty=media, tags={funzioni, dominio}]
Qual è il dominio della funzione $f(x)=\sqrt{\frac{x-1}{x+2}}$?

\begin{tabularx}{\linewidth}{@{}X X X X@{}}  
\boxed{A}\hspace{6pt} $(-\infty,-2)\cup[1,+\infty)$ &  
\boxed{B}\hspace{6pt} $[-2,1]$ &  
\boxed{C}\hspace{6pt} $(-2,1]$ &  
\boxed{D}\hspace{6pt} $(-\infty,-2]\cup[1,+\infty)$  
\end{tabularx}

\begin{solution}
Risposta $\boxed{\text{A}}$: \ $(-\infty,-2)\cup[1,+\infty)$
\end{solution}
\begin{explanation}
Il radicando è una frazione: deve essere $\dfrac{x-1}{x+2}\geq0$, con $x\neq-2$. \\
Numeratore $\geq0$ per $x\geq1$, denominatore $>0$ per $x>-2$: sono concordi per $x<-2$ e per $x\geq1$ ($x=1$ incluso perché annulla la frazione). \\
La D sembra uguale ma include $x=-2$, dove la frazione non esiste; B e C prendono l'intervallo in cui la frazione è negativa.
\end{explanation}
\end{question}

\begin{question}[ID=1.5, subtitle={Dominio di una funzione logaritmica}, difficulty=difficile, tags={funzioni, dominio}]
Qual è il dominio della funzione $f(x)=\ln\left(\frac{x-1}{x+1}\right)$?

\begin{tabularx}{\linewidth}{@{}X X X X@{}}  
\boxed{A}\hspace{6pt} $(-\infty,-1)\cup(1,+\infty)$ &  
\boxed{B}\hspace{6pt} $(-1,1)$ &  
\boxed{C}\hspace{6pt} $\mathbb{R}\setminus\{-1,1\}$ &  
\boxed{D}\hspace{6pt} $(-\infty,1)\cup(1,+\infty)$  
\end{tabularx}

\begin{solution}
Risposta $\boxed{\text{A}}$: \ $(-\infty,-1)\cup(1,+\infty)$
\end{solution}
\begin{explanation}
L'argomento del logaritmo deve essere \textbf{strettamente} positivo:
\[ \dfrac{x-1}{x+1}>0 \]
Numeratore positivo per $x>1$, denominatore positivo per $x>-1$: concordi per $x<-1$ oppure $x>1$. \\
La C toglie solo i due punti, ma tra $-1$ e $1$ la frazione è negativa; la B è proprio l'intervallo in cui è negativa.
\end{explanation}
\end{question}

\begin{question}[ID=2.1, subtitle={Segno di una funzione lineare}, difficulty=facile, tags={funzioni, segno}]
Per quali valori di $x$ la funzione $f(x)=x-4$ è positiva?

\begin{tabularx}{\linewidth}{@{}X X X X@{}}  
\boxed{A}\hspace{6pt} $x<4$ &  
\boxed{B}\hspace{6pt} $x\leq4$ &  
\boxed{C}\hspace{6pt} $x>4$ &  
\boxed{D}\hspace{6pt} $x\geq4$  
\end{tabularx}

\begin{solution}
Risposta $\boxed{\text{C}}$: \ $x>4$
\end{solution}
\begin{explanation}
\[ x-4>0 \;\Rightarrow\; x>4 \]
Per $x=4$ la funzione vale $0$, che non è positivo: per questo la D (con $\geq$) è sbagliata.
\end{explanation}
\end{question}

\begin{question}[ID=2.2, subtitle={Segno di una funzione quadratica}, difficulty=facile, tags={funzioni, segno}]
Per quali valori di $x$ la funzione $f(x)=x^2-9$ è negativa?

\begin{tabularx}{\linewidth}{@{}X X X X@{}}  
\boxed{A}\hspace{6pt} $x<-3\vee x>3$ &  
\boxed{B}\hspace{6pt} $-3<x<3$ &  
\boxed{C}\hspace{6pt} $x\leq-3\vee x\geq3$ &  
\boxed{D}\hspace{6pt} $-3\leq x\leq3$  
\end{tabularx}

\begin{solution}
Risposta $\boxed{\text{B}}$: \ $-3<x<3$
\end{solution}
\begin{explanation}
Scompongo: $x^2-9=(x-3)(x+3)$. È una parabola rivolta verso l'alto che si annulla in $\pm3$: è negativa \textbf{tra} le radici.
\[ (x-3)(x+3)<0 \;\Rightarrow\; -3<x<3 \]
Gli estremi sono esclusi perché lì la funzione vale $0$ (la D li include); la A è l'insieme dove è positiva.
\end{explanation}
\end{question}

\begin{question}[ID=2.3, subtitle={Segno di una funzione razionale}, difficulty=media, tags={funzioni, segno}]
Per quali valori di $x$ la funzione $f(x)=\frac{x+1}{x-2}$ è positiva?

\begin{tabularx}{\linewidth}{@{}X X X X@{}}  
\boxed{A}\hspace{6pt} $x<-1\vee x>2$ &  
\boxed{B}\hspace{6pt} $-1<x<2$ &  
\boxed{C}\hspace{6pt} $x<-2\vee x>1$ &  
\boxed{D}\hspace{6pt} $x>-1$  
\end{tabularx}

\begin{solution}
Risposta $\boxed{\text{A}}$: \ $x<-1\vee x>2$
\end{solution}
\begin{explanation}
Studio il segno di numeratore e denominatore:
\[ N=x+1>0 \text{ per } x>-1, \qquad D=x-2>0 \text{ per } x>2 \]
La frazione è positiva dove $N$ e $D$ hanno lo stesso segno: entrambi negativi per $x<-1$, entrambi positivi per $x>2$. Tra $-1$ e $2$ sono discordi (B è dove è negativa).
\end{explanation}
\end{question}

\begin{question}[ID=2.4, subtitle={Segno di una funzione polinomiale}, difficulty=media, tags={funzioni, segno}]
Qual è l'insieme dei valori di $x$ per cui la funzione $f(x)=(x-1)(x+2)$ è positiva?

\begin{tabularx}{\linewidth}{@{}X X X X@{}}  
\boxed{A}\hspace{6pt} $-2<x<1$ &  
\boxed{B}\hspace{6pt} $x<-2\vee x>1$ &  
\boxed{C}\hspace{6pt} $x\leq-2\vee x\geq1$ &  
\boxed{D}\hspace{6pt} $x>1$  
\end{tabularx}

\begin{solution}
Risposta $\boxed{\text{B}}$: \ $x<-2\vee x>1$
\end{solution}
\begin{explanation}
Il prodotto si annulla in $x=1$ e $x=-2$. Sviluppato è $x^2+x-2$, una parabola rivolta verso l'alto: è positiva \textbf{all'esterno} delle radici. \\
La C include $-2$ e $1$, dove la funzione vale $0$; la A è dove è negativa.
\end{explanation}
\end{question}

\begin{question}[ID=2.5, subtitle={Studio del segno di una funzione razionale}, difficulty=difficile, tags={funzioni, segno}]
Per quali valori di $x$ la funzione $f(x)=\frac{x^2-4}{x-1}$ è negativa?

\begin{tabularx}{\linewidth}{@{}X X X X@{}}  
\boxed{A}\hspace{6pt} $x<-2\vee1<x<2$ &  
\boxed{B}\hspace{6pt} $x<-2\vee x>2$ &  
\boxed{C}\hspace{6pt} $-2<x<1$ &  
\boxed{D}\hspace{6pt} $-2<x<1\vee x>2$  
\end{tabularx}

\begin{solution}
Risposta $\boxed{\text{A}}$: \ $x<-2\vee1<x<2$
\end{solution}
\begin{explanation}
\[ \dfrac{x^2-4}{x-1}=\dfrac{(x-2)(x+2)}{x-1}<0 \]
Zeri dei fattori in ordine: $-2$, $1$, $2$. Segno di ciascun fattore e del quoziente:
\[\begin{aligned}
x<-2: &\quad (-)(-)/(-)=- && \textcolor{gray}{\text{negativa}}\\
-2<x<1: &\quad (-)(+)/(-)=+\\
1<x<2: &\quad (-)(+)/(+)=- && \textcolor{gray}{\text{negativa}}\\
x>2: &\quad (+)(+)/(+)=+
\end{aligned}\]
La funzione è negativa per $x<-2$ e per $1<x<2$.
\end{explanation}
\end{question}

\begin{question}[ID=3.1, subtitle={Funzione pari}, difficulty=facile, tags={funzioni, parità}]
Quale delle seguenti funzioni è pari?

\begin{tabularx}{\linewidth}{@{}X X X X@{}}  
\boxed{A}\hspace{6pt} $f(x)=x^3+x$ &  
\boxed{B}\hspace{6pt} $f(x)=x^2+1$ &  
\boxed{C}\hspace{6pt} $f(x)=2x-1$ &  
\boxed{D}\hspace{6pt} $f(x)=x^3-2$  
\end{tabularx}

\begin{solution}
Risposta $\boxed{\text{B}}$: \ $f(x)=x^2+1$
\end{solution}
\begin{explanation}
Una funzione è \textbf{pari} se $f(-x)=f(x)$ (grafico simmetrico rispetto all'asse $y$). Sostituisco $-x$:
\[ f(-x)=(-x)^2+1=x^2+1=f(x) \]
Le altre: $x^3+x$ è dispari, $2x-1$ e $x^3-2$ non sono né pari né dispari (cambiano segno solo alcuni termini).
\end{explanation}
\end{question}

%----------------------------------------------------%

\begin{question}[ID=3.2, subtitle={Funzione dispari}, difficulty=facile, tags={funzioni, parità}]
Quale delle seguenti funzioni è dispari?

\begin{tabularx}{\linewidth}{@{}X X X X@{}}  
\boxed{A}\hspace{6pt} $f(x)=x^2+1$ &  
\boxed{B}\hspace{6pt} $f(x)=x^3-2x$ &  
\boxed{C}\hspace{6pt} $f(x)=x^2-3$ &  
\boxed{D}\hspace{6pt} $f(x)=x+1$  
\end{tabularx}

\begin{solution}
Risposta $\boxed{\text{B}}$: \ $f(x)=x^3-2x$
\end{solution}
\begin{explanation}
Una funzione è \textbf{dispari} se $f(-x)=-f(x)$ (grafico simmetrico rispetto all'origine).
\[ f(-x)=(-x)^3-2(-x)=-x^3+2x=-(x^3-2x)=-f(x) \]
A e C sono pari; la D non è né pari né dispari per via del termine costante $+1$.
\end{explanation}
\end{question}

\begin{question}[ID=3.3, subtitle={Pari, dispari o nessuna delle due}, difficulty=media, tags={funzioni, parità}]
La funzione $f(x)=x^3+x^2$ è:

\begin{tabularx}{\linewidth}{@{}X X X X@{}}  
\boxed{A}\hspace{6pt} pari &  
\boxed{B}\hspace{6pt} dispari &  
\boxed{C}\hspace{6pt} né pari né dispari &  
\boxed{D}\hspace{6pt} costante  
\end{tabularx}

\begin{solution}
Risposta $\boxed{\text{C}}$: \ né pari né dispari
\end{solution}
\begin{explanation}
\[\begin{aligned}
f(-x) &= (-x)^3+(-x)^2=-x^3+x^2\\
f(x) &= x^3+x^2, \qquad -f(x)=-x^3-x^2
\end{aligned}\]
$f(-x)$ è diversa sia da $f(x)$ sia da $-f(x)$ (basta provare con $x=1$: $f(1)=2$, $f(-1)=0$). \\
Un polinomio è pari se contiene solo potenze pari, dispari se solo potenze dispari: qui ci sono entrambe.
\end{explanation}
\end{question}

\begin{question}[ID=3.4, subtitle={Parità di una funzione razionale}, difficulty=media, tags={funzioni, parità}]
La funzione $f(x)=\frac{1}{x}$ è:

\begin{tabularx}{\linewidth}{@{}X X X X@{}}  
\boxed{A}\hspace{6pt} pari &  
\boxed{B}\hspace{6pt} dispari &  
\boxed{C}\hspace{6pt} né pari né dispari &  
\boxed{D}\hspace{6pt} non è una funzione  
\end{tabularx}

\begin{solution}
Risposta $\boxed{\text{B}}$: \ dispari
\end{solution}
\begin{explanation}
Il dominio $\mathbb{R}\setminus\{0\}$ è simmetrico rispetto all'origine, e
\[ f(-x)=\dfrac{1}{-x}=-\dfrac1x=-f(x) \]
Il grafico (iperbole) è infatti simmetrico rispetto all'origine.
\end{explanation}
\end{question}

\begin{question}[ID=3.5, subtitle={Parità di una funzione razionale}, difficulty=difficile, tags={funzioni, parità}]
Quale delle seguenti funzioni è pari?

\begin{tabularx}{\linewidth}{@{}X X X X@{}}  
\boxed{A}\hspace{6pt} $f(x)=\frac{x+1}{x^2+1}$ &  
\boxed{B}\hspace{6pt} $f(x)=\frac{x^2+1}{x^2-4}$ &  
\boxed{C}\hspace{6pt} $f(x)=\frac{x}{x^2-1}$ &  
\boxed{D}\hspace{6pt} $f(x)=\frac{x^3+1}{x^2+1}$  
\end{tabularx}

\begin{solution}
Risposta $\boxed{\text{B}}$: \ $f(x)=\dfrac{x^2+1}{x^2-4}$
\end{solution}
\begin{explanation}
Al numeratore e al denominatore compaiono solo potenze pari di $x$, quindi sostituendo $-x$ nulla cambia:
\[ f(-x)=\dfrac{(-x)^2+1}{(-x)^2-4}=\dfrac{x^2+1}{x^2-4}=f(x) \]
Nelle altre: A e D hanno potenze miste al numeratore (né pari né dispari); C è dispari (numeratore dispari, denominatore pari).
\end{explanation}
\end{question}

\begin{question}[ID=4.1, subtitle={Calcolo di una funzione composta}, difficulty=facile, tags={funzioni, composizione}]
Siano $f(x)=2x+1$ e $g(x)=x^2$. Quanto vale $(f\circ g)(3)$?

\begin{tabularx}{\linewidth}{@{}X X X X@{}}  
\boxed{A}\hspace{6pt} $7$ &  
\boxed{B}\hspace{6pt} $9$ &  
\boxed{C}\hspace{6pt} $19$ &  
\boxed{D}\hspace{6pt} $21$  
\end{tabularx}

\begin{solution}
Risposta $\boxed{\text{C}}$: \ $19$
\end{solution}
\begin{explanation}
In $f\circ g$ si applica prima $g$, poi $f$:
\[ g(3)=3^2=9, \qquad f(9)=2\cdot9+1=19 \]
Le opzioni A ($7=f(3)$) e B ($9=g(3)$) applicano una sola delle due funzioni. Attenzione all'ordine: la funzione scritta a destra agisce per prima.
\end{explanation}
\end{question}

\begin{question}[ID=4.2, subtitle={Espressione di una funzione composta}, difficulty=facile, tags={funzioni, composizione}]
Siano $f(x)=x+2$ e $g(x)=3x-1$. Qual è l'espressione di $(g\circ f)(x)$?

\begin{tabularx}{\linewidth}{@{}X X X X@{}}  
\boxed{A}\hspace{6pt} $3x+1$ &  
\boxed{B}\hspace{6pt} $3x+5$ &  
\boxed{C}\hspace{6pt} $x+5$ &  
\boxed{D}\hspace{6pt} $3x-3$  
\end{tabularx}

\begin{solution}
Risposta $\boxed{\text{B}}$: \ $3x+5$
\end{solution}
\begin{explanation}
In $g\circ f$ si applica prima $f$: al posto della $x$ di $g$ si mette $f(x)=x+2$.
\[ g(x+2)=3(x+2)-1=3x+6-1=3x+5 \]
La A, $3x+1$, è invece $f\circ g$: $f(3x-1)=3x-1+2$.
\end{explanation}
\end{question}

\begin{question}[ID=4.3, subtitle={Composizione di funzioni algebriche}, difficulty=media, tags={funzioni, composizione}]
Siano $f(x)=x^2$ e $g(x)=x-1$. Qual è l'espressione di $(f\circ g)(x)$?

\begin{tabularx}{\linewidth}{@{}X X X X@{}}  
\boxed{A}\hspace{6pt} $x^2-1$ &  
\boxed{B}\hspace{6pt} $(x-1)^2$ &  
\boxed{C}\hspace{6pt} $x^2+1$ &  
\boxed{D}\hspace{6pt} $(x+1)^2$  
\end{tabularx}

\begin{solution}
Risposta $\boxed{\text{B}}$: \ $(x-1)^2$
\end{solution}
\begin{explanation}
\[ f\big(g(x)\big)=f(x-1)=(x-1)^2 \]
L'errore tipico è la A: $x^2-1$ è $g\circ f$, cioè prima si eleva al quadrato e poi si sottrae $1$.
\end{explanation}
\end{question}

\begin{question}[ID=4.5, subtitle={Composizione di tre funzioni}, difficulty=difficile, tags={funzioni, composizione}]
Siano $f(x)=2x$, $g(x)=x+1$ e $h(x)=x^2$. Qual è l'espressione di $(f\circ g\circ h)(x)$?

\begin{tabularx}{\linewidth}{@{}X X X X@{}}  
\boxed{A}\hspace{6pt} $2x^2+1$ &  
\boxed{B}\hspace{6pt} $2x^2+2$ &  
\boxed{C}\hspace{6pt} $(2x+1)^2$ &  
\boxed{D}\hspace{6pt} $2(x+1)^2$  
\end{tabularx}

\begin{solution}
Risposta $\boxed{\text{B}}$: \ $2x^2+2$
\end{solution}
\begin{explanation}
Si lavora da destra verso sinistra: prima $h$, poi $g$, infine $f$.
\[\begin{aligned}
h(x) &= x^2\\
g\big(h(x)\big) &= x^2+1\\
f\big(g(h(x))\big) &= 2(x^2+1)=2x^2+2
\end{aligned}\]
La D, $2(x+1)^2$, corrisponde a un ordine diverso: prima $g$, poi $h$, poi $f$ (cioè $f\circ h\circ g$).
\end{explanation}
\end{question}
