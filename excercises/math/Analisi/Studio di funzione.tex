\begin{question}[ID=1.1, subtitle={Studio di una funzione polinomiale}, difficulty=facile, tags={funzioni, studio di funzione}]
Determinare dominio, segno, intersezioni con gli assi cartesiani e parità della funzione seguente:
\begin{equation*}
    f(x)=x^2-4
\end{equation*}
\begin{solution}
\begin{enumerate}
    \item Dominio: $\mathbb{R}$
    \item Segno: $f>0$ per $x<-2\vee x>2$; \ $f<0$ per $-2<x<2$
    \item Intersezioni: $(-2,0)$, $(2,0)$, $(0,-4)$
    \item Parità: pari
\end{enumerate}
\end{solution}
\begin{explanation}
Schema: \ \textbf{dominio} (condizioni di esistenza); \ \textbf{segno} (dove $f>0$, $f<0$, $f=0$); \ \textbf{intersezioni} con l'asse $x$ (zeri, $y=0$) e con l'asse $y$ ($x=0$, se appartiene al dominio); \ \textbf{parità}: pari se $f(-x)=f(x)$, dispari se $f(-x)=-f(x)$ (serve un dominio simmetrico).
\begin{enumerate}
    \item Polinomio: $D=\mathbb{R}$.
    \item $x^2-4=(x-2)(x+2)$: parabola rivolta verso l'alto con zeri $\pm2$, positiva all'esterno degli zeri e negativa tra di essi.
    \item Asse $x$: $x^2-4=0 \Rightarrow x=\pm2$. \ Asse $y$: $f(0)=-4$.
    \item $f(-x)=(-x)^2-4=x^2-4=f(x)$: pari (grafico simmetrico rispetto all'asse $y$).
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=1.2, subtitle={Studio di una funzione razionale}, difficulty=facile, tags={funzioni, studio di funzione}]
Determinare dominio, segno, intersezioni con gli assi cartesiani e parità della funzione seguente:
\begin{equation*}
    f(x)=\dfrac{x-2}{x+1}
\end{equation*}
\begin{solution}
\begin{enumerate}
    \item Dominio: $\mathbb{R}\setminus\{-1\}$
    \item Segno: $f>0$ per $x<-1\vee x>2$; \ $f<0$ per $-1<x<2$
    \item Intersezioni: $(2,0)$, $(0,-2)$
    \item Parità: né pari né dispari
\end{enumerate}
\end{solution}
\begin{explanation}
Schema: \ \textbf{dominio} (condizioni di esistenza); \ \textbf{segno} (dove $f>0$, $f<0$, $f=0$); \ \textbf{intersezioni} con l'asse $x$ (zeri, $y=0$) e con l'asse $y$ ($x=0$, se appartiene al dominio); \ \textbf{parità}: pari se $f(-x)=f(x)$, dispari se $f(-x)=-f(x)$ (serve un dominio simmetrico).
\begin{enumerate}
    \item Denominatore $x+1\neq0$: $x\neq-1$.
    \item Numeratore positivo per $x>2$, denominatore per $x>-1$: concordi per $x<-1$ e per $x>2$.
    \item Asse $x$: numeratore nullo, $x=2$. \ Asse $y$: $f(0)=\dfrac{-2}{1}=-2$.
    \item Il dominio non è simmetrico ($1$ appartiene, $-1$ no): non può essere né pari né dispari.
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=1.3, subtitle={Studio di una funzione razionale fratta}, difficulty=media, tags={funzioni, studio di funzione}]
Determinare dominio, segno, intersezioni con gli assi cartesiani e parità della funzione seguente:
\begin{equation*}
    f(x)=\dfrac{2x-4}{x^2-9}
\end{equation*}
\begin{solution}
\begin{enumerate}
    \item Dominio: $\mathbb{R}\setminus\{-3,3\}$
    \item Segno: $f>0$ per $-3<x<2\vee x>3$; \ $f<0$ per $x<-3\vee2<x<3$
    \item Intersezioni: $(2,0)$, $\left(0,\tfrac49\right)$
    \item Parità: né pari né dispari
\end{enumerate}
\end{solution}
\begin{explanation}
Schema: \ \textbf{dominio} (condizioni di esistenza); \ \textbf{segno} (dove $f>0$, $f<0$, $f=0$); \ \textbf{intersezioni} con l'asse $x$ (zeri, $y=0$) e con l'asse $y$ ($x=0$, se appartiene al dominio); \ \textbf{parità}: pari se $f(-x)=f(x)$, dispari se $f(-x)=-f(x)$ (serve un dominio simmetrico).
\begin{enumerate}
    \item $x^2-9\neq0$: $x\neq\pm3$.
    \item $f(x)=\dfrac{2(x-2)}{(x-3)(x+3)}$. Zeri dei fattori in ordine: $-3$, $2$, $3$. Studiando il segno di ciascun fattore, il quoziente vale da sinistra: $-,\ +,\ -,\ +$.
    \item Asse $x$: $2x-4=0 \Rightarrow x=2$. \ Asse $y$: $f(0)=\dfrac{-4}{-9}=\dfrac49$.
    \item $f(-x)=\dfrac{-2x-4}{x^2-9}$, che non coincide né con $f(x)$ né con $-f(x)$ (ad esempio $f(0)=\tfrac49\neq0$ esclude la disparità).
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=1.5, subtitle={Studio di una funzione razionale con più fattori}, difficulty=difficile, tags={funzioni, studio di funzione}]
Determinare dominio, segno, intersezioni con gli assi cartesiani e parità della funzione seguente:
\begin{equation*}
    f(x)=\dfrac{x^2-1}{x(x^2-4)}
\end{equation*}
\begin{solution}
\begin{enumerate}
    \item Dominio: $\mathbb{R}\setminus\{-2,0,2\}$
    \item Segno: $f>0$ per $-2<x<-1\vee0<x<1\vee x>2$; \ $f<0$ per $x<-2\vee-1<x<0\vee1<x<2$
    \item Intersezioni: $(-1,0)$, $(1,0)$; nessuna con l'asse $y$
    \item Parità: dispari
\end{enumerate}
\end{solution}
\begin{explanation}
Schema: \ \textbf{dominio} (condizioni di esistenza); \ \textbf{segno} (dove $f>0$, $f<0$, $f=0$); \ \textbf{intersezioni} con l'asse $x$ (zeri, $y=0$) e con l'asse $y$ ($x=0$, se appartiene al dominio); \ \textbf{parità}: pari se $f(-x)=f(x)$, dispari se $f(-x)=-f(x)$ (serve un dominio simmetrico).
\begin{enumerate}
    \item $x(x^2-4)\neq0$: $x\neq0$ e $x\neq\pm2$.
    \item $f(x)=\dfrac{(x-1)(x+1)}{x(x-2)(x+2)}$. Zeri dei fattori: $-2,\ -1,\ 0,\ 1,\ 2$. \\
    Per $x$ molto grande il quoziente è positivo (si comporta come $\tfrac1x$); a ogni zero (tutti semplici) il segno cambia. Da destra a sinistra: $+,\ -,\ +,\ -,\ +,\ -$.
    \item Asse $x$: $x^2-1=0 \Rightarrow x=\pm1$. \ Asse $y$: $x=0$ non è nel dominio.
    \item Numeratore pari, denominatore dispari: $f(-x)=\dfrac{x^2-1}{-x(x^2-4)}=-f(x)$, dispari.
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=1.15, subtitle={Studio di una funzione logaritmica composta}, difficulty=difficile, tags={funzioni, studio di funzione, logaritmi, parità}]
Determinare dominio, segno, intersezioni con gli assi cartesiani e parità della funzione seguente:
\begin{equation*}
    f(x)=\ln\left(\frac{x-1}{x+1}\right)
\end{equation*}
\begin{solution}
\begin{enumerate}
    \item Dominio: $(-\infty,-1)\cup(1,+\infty)$
    \item Segno: $f>0$ per $x<-1$; \ $f<0$ per $x>1$; nessuno zero
    \item Intersezioni: nessuna
    \item Parità: dispari
\end{enumerate}
\end{solution}
\begin{explanation}
Schema: \ \textbf{dominio} (condizioni di esistenza); \ \textbf{segno} (dove $f>0$, $f<0$, $f=0$); \ \textbf{intersezioni} con l'asse $x$ (zeri, $y=0$) e con l'asse $y$ ($x=0$, se appartiene al dominio); \ \textbf{parità}: pari se $f(-x)=f(x)$, dispari se $f(-x)=-f(x)$ (serve un dominio simmetrico).
\begin{enumerate}
    \item $\dfrac{x-1}{x+1}>0$: $x<-1$ oppure $x>1$.
    \item $\ln t>0 \iff t>1$. Confronto l'argomento con $1$:
    \[ \dfrac{x-1}{x+1}-1=\dfrac{-2}{x+1} \]
    che è positivo per $x<-1$ (argomento $>1$, $f>0$) e negativo per $x>1$ (argomento $<1$, $f<0$).
    \item $f=0$ richiederebbe $\tfrac{x-1}{x+1}=1$, impossibile; l'asse $y$ è escluso.
    \item $f(-x)=\ln\dfrac{-x-1}{-x+1}=\ln\dfrac{x+1}{x-1}=-\ln\dfrac{x-1}{x+1}=-f(x)$.
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=2.1, subtitle={Studio completo di una funzione polinomiale}, difficulty=facile, tags={funzioni, studio di funzione, polinomi}]
Effettua lo studio completo della funzione seguente. Determina dominio, intersezioni con gli assi cartesiani, segno, parità, limiti, eventuali asintoti, derivata prima, intervalli di monotonia, eventuali estremi relativi, derivata seconda e concavità. Rappresenta infine schematicamente il grafico.
\begin{equation*}
    f(x)=x^2-4x+3
\end{equation*}
\begin{solution}
\begin{enumerate}
    \item $D=\mathbb{R}$
    \item Intersezioni: $(1,0)$, $(3,0)$, $(0,3)$
    \item $f>0$ per $x<1\vee x>3$; \ $f<0$ per $1<x<3$
    \item né pari né dispari (simmetrica rispetto a $x=2$)
    \item $\lim_{x\to\pm\infty}f=+\infty$
    \item nessun asintoto
    \item $f'(x)=2(x-2)$
    \item decrescente per $x<2$, crescente per $x>2$
    \item minimo assoluto $(2,-1)$
    \item $f''=2$: convessa ovunque, nessun flesso
\end{enumerate}
\end{solution}
\begin{explanation}
\begin{enumerate}
    \item Polinomio: $D=\mathbb{R}$.
    \item Asse $x$: $x^2-4x+3=(x-1)(x-3)=0 \Rightarrow x=1,\ 3$. \ Asse $y$: $f(0)=3$.
    \item Parabola rivolta verso l'alto: positiva fuori dalle radici, negativa tra di esse.
    \item $f(-x)=x^2+4x+3$: non coincide con $\pm f(x)$. L'asse di simmetria della parabola è $x=-\tfrac{b}{2a}=2$.
    \item Domina il termine $x^2$: tende a $+\infty$ da entrambe le parti.
    \item Polinomio di grado $2$: nessun asintoto (né verticale, né orizzontale, né obliquo).
    \item $f'(x)=2x-4$.
    \item $f'>0 \iff x>2$.
    \item La derivata passa da negativa a positiva in $x=2$: minimo, $f(2)=4-8+3=-1$ (assoluto, perché la parabola non scende mai più in basso del vertice).
    \item $f''(x)=2>0$ costante.
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=2.2, subtitle={Studio completo di una funzione razionale}, difficulty=media, tags={funzioni, studio di funzione, funzioni razionali}]
Effettua lo studio completo della funzione seguente. Determina dominio, intersezioni con gli assi cartesiani, segno, parità, limiti, eventuali asintoti, derivata prima, intervalli di monotonia, eventuali estremi relativi, derivata seconda e concavità. Rappresenta infine schematicamente il grafico.
\begin{equation*}
    f(x)=\dfrac{x-1}{x+1}
\end{equation*}
\begin{solution}
\begin{enumerate}
    \item $D=\mathbb{R}\setminus\{-1\}$
    \item Intersezioni: $(1,0)$, $(0,-1)$
    \item $f>0$ per $x<-1\vee x>1$; \ $f<0$ per $-1<x<1$
    \item né pari né dispari
    \item $\lim_{x\to-1^\mp}f=\pm\infty$; \ $\lim_{x\to\pm\infty}f=1$
    \item asintoti $x=-1$ e $y=1$
    \item $f'(x)=\dfrac{2}{(x+1)^2}>0$
    \item crescente in $(-\infty,-1)$ e in $(-1,+\infty)$; nessun estremo
    \item $f''(x)=-\dfrac{4}{(x+1)^3}$
    \item convessa per $x<-1$, concava per $x>-1$; nessun flesso
\end{enumerate}
\end{solution}
\begin{explanation}
\begin{enumerate}
    \item $x+1\neq0$.
    \item Asse $x$: $x-1=0$. \ Asse $y$: $f(0)=-1$.
    \item Numeratore e denominatore concordi per $x<-1$ e $x>1$.
    \item Dominio non simmetrico.
    \item In $x=-1$ il numeratore vale $-2$. Per $x\to-1^-$ il denominatore è $0^-$: $\dfrac{-2}{0^-}=+\infty$; per $x\to-1^+$ è $0^+$: $-\infty$. \\
    All'infinito numeratore e denominatore hanno lo stesso grado: il limite è il rapporto dei coefficienti, $1$.
    \item Dai limiti: verticale $x=-1$, orizzontale $y=1$.
    \item Regola del quoziente: $f'(x)=\dfrac{1\cdot(x+1)-(x-1)\cdot1}{(x+1)^2}=\dfrac{2}{(x+1)^2}$.
    \item Sempre positiva: crescente in ciascuno dei due intervalli del dominio (non su tutto $D$, che è spezzato).
    \item Scrivendo $f'=2(x+1)^{-2}$: \ $f''=-4(x+1)^{-3}$.
    \item Per $x<-1$, $(x+1)^3<0$ e quindi $f''>0$; per $x>-1$, $f''<0$. Il cambio avviene in $x=-1$, che non è nel dominio: niente flesso.
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=2.3, subtitle={Studio completo di una funzione razionale fratta}, difficulty=media, tags={funzioni, studio di funzione, funzioni razionali}]
Effettua lo studio completo della funzione seguente. Determina dominio, intersezioni con gli assi cartesiani, segno, parità, limiti, eventuali asintoti, derivata prima, intervalli di monotonia, eventuali estremi relativi, derivata seconda e concavità. Rappresenta infine schematicamente il grafico.
\begin{equation*}
    f(x)=\dfrac{x^2-1}{x-2}
\end{equation*}
\begin{solution}
\begin{enumerate}
    \item $D=\mathbb{R}\setminus\{2\}$
    \item Intersezioni: $(-1,0)$, $(1,0)$, $\left(0,\tfrac12\right)$
    \item $f>0$ per $-1<x<1\vee x>2$; \ $f<0$ per $x<-1\vee1<x<2$
    \item né pari né dispari
    \item $\lim_{x\to2^\mp}f=\mp\infty$; \ $\lim_{x\to\pm\infty}f=\pm\infty$
    \item asintoti $x=2$ e $y=x+2$
    \item $f'(x)=1-\dfrac{3}{(x-2)^2}$
    \item crescente per $x<2-\sqrt3$ e $x>2+\sqrt3$; decrescente per $2-\sqrt3<x<2$ e $2<x<2+\sqrt3$
    \item massimo relativo $(2-\sqrt3,\ 4-2\sqrt3)$; minimo relativo $(2+\sqrt3,\ 4+2\sqrt3)$
    \item $f''=\dfrac{6}{(x-2)^3}$: concava per $x<2$, convessa per $x>2$; nessun flesso
\end{enumerate}
\end{solution}
\begin{explanation}
\begin{enumerate}
    \item $x\neq2$.
    \item Asse $x$: $x^2-1=0 \Rightarrow x=\pm1$. \ Asse $y$: $f(0)=\dfrac{-1}{-2}=\dfrac12$.
    \item $f=\dfrac{(x-1)(x+1)}{x-2}$: zeri $-1,\ 1,\ 2$, segni da sinistra $-,\ +,\ -,\ +$.
    \item Dominio non simmetrico.
    \item In $x=2$ il numeratore vale $3>0$: $\dfrac{3}{0^-}=-\infty$ a sinistra, $+\infty$ a destra. All'infinito il grado del numeratore è maggiore: $\pm\infty$.
    \item I gradi differiscono di $1$: divido i polinomi.
    \[\begin{aligned}
    x^2-1 &= (x-2)(x+2)+3\\
    f(x) &= x+2+\dfrac{3}{x-2} && \textcolor{gray}{\text{l'ultimo termine tende a $0$}}
    \end{aligned}\]
    Asintoto obliquo $y=x+2$; verticale $x=2$.
    \item Derivo la forma divisa: $f'(x)=1-\dfrac{3}{(x-2)^2}$.
    \item \[\begin{aligned}
    f'(x)>0 &\iff (x-2)^2>3\\
    &\iff x-2<-\sqrt3 \;\vee\; x-2>\sqrt3
    \end{aligned}\]
    \item In $2-\sqrt3$ la derivata passa da $+$ a $-$: massimo; in $2+\sqrt3$ da $-$ a $+$: minimo.
    \[\begin{aligned}
    f(2-\sqrt3) &= 4-\sqrt3+\dfrac{3}{-\sqrt3}=4-2\sqrt3 && \textcolor{gray}{\text{$\dfrac3{\sqrt3}=\sqrt3$}}\\
    f(2+\sqrt3) &= 4+\sqrt3+\sqrt3=4+2\sqrt3
    \end{aligned}\]
    \item $f''(x)=\dfrac{6}{(x-2)^3}$: segno di $(x-2)^3$. Il cambio di concavità avviene in $x=2$, fuori dal dominio.
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=2.4, subtitle={Studio completo di una funzione esponenziale fratta}, difficulty=difficile, tags={funzioni, studio di funzione, esponenziali}]
Effettua lo studio completo della funzione seguente. Determina dominio, intersezioni con gli assi cartesiani, segno, parità, limiti, eventuali asintoti, derivata prima, intervalli di monotonia, eventuali estremi relativi, derivata seconda e concavità. Rappresenta infine schematicamente il grafico.
\begin{equation*}
    f(x)=\dfrac{e^x}{x}
\end{equation*}
\begin{solution}
\begin{enumerate}
    \item $D=\mathbb{R}\setminus\{0\}$
    \item nessuna intersezione con gli assi
    \item $f<0$ per $x<0$; \ $f>0$ per $x>0$
    \item né pari né dispari
    \item $\lim_{x\to0^\mp}f=\mp\infty$; \ $\lim_{x\to+\infty}f=+\infty$, \ $\lim_{x\to-\infty}f=0^-$
    \item asintoto verticale $x=0$; orizzontale $y=0$ per $x\to-\infty$ (nessun obliquo)
    \item $f'(x)=\dfrac{e^x(x-1)}{x^2}$
    \item decrescente per $x<0$ e per $0<x<1$; crescente per $x>1$
    \item minimo \textbf{relativo} $(1,e)$ (non assoluto)
    \item $f''=\dfrac{e^x(x^2-2x+2)}{x^3}$: concava per $x<0$, convessa per $x>0$; nessun flesso
\end{enumerate}
\end{solution}
\begin{explanation}
\begin{enumerate}
    \item $x\neq0$.
    \item $e^x$ non si annulla mai, quindi nessuno zero; $x=0$ è escluso.
    \item $e^x>0$: il segno è quello di $x$.
    \item Dominio simmetrico, ma $f(-x)=\dfrac{e^{-x}}{-x}$ non coincide con $\pm f(x)$.
    \item In $x=0$: $\dfrac{1}{0^\pm}=\pm\infty$. \ Per $x\to+\infty$ l'esponenziale prevale sul polinomio: $+\infty$. \ Per $x\to-\infty$: $e^x\to0$ e $x\to-\infty$, quindi $0^-$.
    \item Per $x\to+\infty$ un eventuale obliquo avrebbe $m=\lim\dfrac{f(x)}{x}=\lim\dfrac{e^x}{x^2}=+\infty$: non esiste.
    \item Regola del quoziente:
    \[ f'(x)=\dfrac{e^x\cdot x-e^x\cdot1}{x^2}=\dfrac{e^x(x-1)}{x^2} \]
    \item $e^x>0$ e $x^2>0$: il segno è quello di $x-1$.
    \item In $x=1$ la derivata passa da $-$ a $+$: minimo, $f(1)=e$. Non è assoluto, perché per $x<0$ la funzione assume valori negativi (e tende a $-\infty$ per $x\to0^-$).
    \item Derivando di nuovo e semplificando:
    \[ f''(x)=\dfrac{e^x(x^2-2x+2)}{x^3} \]
    $x^2-2x+2=(x-1)^2+1>0$ sempre: il segno è quello di $x^3$, cioè di $x$.
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=2.5, subtitle={Studio completo di una funzione logaritmica composta}, difficulty=difficile, tags={funzioni, studio di funzione, logaritmi}]
Effettua lo studio completo della funzione seguente. Determina dominio, intersezioni con gli assi cartesiani, segno, parità, limiti, eventuali asintoti, derivata prima, intervalli di monotonia, eventuali estremi relativi, derivata seconda e concavità. Rappresenta infine schematicamente il grafico.
\begin{equation*}
    f(x)=\ln\left(\dfrac{x-1}{x+1}\right)
\end{equation*}
\begin{solution}
\begin{enumerate}
    \item $D=(-\infty,-1)\cup(1,+\infty)$
    \item nessuna intersezione con gli assi
    \item $f>0$ per $x<-1$; \ $f<0$ per $x>1$
    \item dispari
    \item $\lim_{x\to-1^-}f=+\infty$, \ $\lim_{x\to1^+}f=-\infty$; \ $\lim_{x\to\mp\infty}f=0^\pm$
    \item asintoti verticali $x=\pm1$; orizzontale $y=0$
    \item $f'(x)=\dfrac{2}{x^2-1}>0$
    \item crescente in $(-\infty,-1)$ e in $(1,+\infty)$; nessun estremo
    \item $f''(x)=-\dfrac{4x}{(x^2-1)^2}$
    \item convessa per $x<-1$, concava per $x>1$; nessun flesso
\end{enumerate}
\end{solution}
\begin{explanation}
\begin{enumerate}
    \item $\dfrac{x-1}{x+1}>0$: $x<-1$ oppure $x>1$.
    \item L'argomento non vale mai $1$ e $x=0$ è escluso.
    \item L'argomento è $>1$ per $x<-1$ e $<1$ per $x>1$ (lo si vede da $\tfrac{x-1}{x+1}-1=\tfrac{-2}{x+1}$).
    \item $f(-x)=\ln\dfrac{-x-1}{-x+1}=\ln\dfrac{x+1}{x-1}=-f(x)$.
    \item Per $x\to-1^-$ l'argomento tende a $\tfrac{-2}{0^-}=+\infty$, quindi $\ln\to+\infty$. Per $x\to1^+$ tende a $0^+$, quindi $\ln\to-\infty$. All'infinito l'argomento tende a $1$ e il logaritmo a $0$.
    \item Dai limiti.
    \item Uso $f=\ln(x-1)-\ln(x+1)$ (lecito nel dominio, a meno dei moduli):
    \[ f'(x)=\dfrac{1}{x-1}-\dfrac{1}{x+1}=\dfrac{(x+1)-(x-1)}{x^2-1}=\dfrac{2}{x^2-1} \]
    \item Nel dominio $x^2>1$: la derivata è positiva.
    \item $f'=2(x^2-1)^{-1} \Rightarrow f''=-2(x^2-1)^{-2}\cdot2x$.
    \item Il segno di $f''$ è quello di $-x$: positivo per $x<-1$, negativo per $x>1$. Il cambio avverrebbe in $x=0$, fuori dal dominio.
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=3.1, subtitle={Studio di $f(x)=(2x+1)/(x+k)$ con parametro $k$}, tags={studio di funzione, parametrico}, difficulty=difficile]
Considera la funzione
\begin{equation*}
    f(x) = \frac{2x+1}{x+k}
\end{equation*}
dove $k$ è un parametro reale, e indica con $y$ il suo grafico.
\begin{enumerate}
    \item Dimostra che per $k=\frac{1}{2}$ la funzione attraversa solo il primo e secondo quadrante.
    \item Studia la funzione al variare del parametro $k$ e tracciane il grafico.
    \item Posto $k=1$ rappresenta la funzione $g(x) = |f(x)|$ e determina eventuali punti di non derivabilità della funzione.
\end{enumerate}
\begin{solution}
\begin{enumerate}
    \item Per $k=\tfrac12$: $f(x)=2$ (con $x\neq-\tfrac12$), retta orizzontale sopra l'asse $x$: solo I e II quadrante.
    \item Per $k\neq\tfrac12$: iperbole equilatera di centro $(-k,2)$, asintoti $x=-k$ e $y=2$; crescente se $k>\tfrac12$, decrescente se $k<\tfrac12$; nessun estremo.
    \item Per $k=1$: $g(x)=\left|\dfrac{2x+1}{x+1}\right|$ ha un punto angoloso in $x=-\tfrac12$ (e asintoto verticale $x=-1$).
\end{enumerate}
\end{solution}
\begin{explanation}
\begin{enumerate}
    \item Con $k=\tfrac12$ il denominatore diventa $x+\tfrac12=\dfrac{2x+1}{2}$:
    \[ f(x)=\dfrac{2x+1}{\frac{2x+1}{2}}=2 \qquad (x\neq-\dfrac12) \]
    Il grafico è la retta $y=2$ privata di un punto: sta tutta sopra l'asse $x$, quindi solo nel primo e secondo quadrante.
    \item Dominio $x\neq-k$. \\
    \textbf{Asintoti}: in $x=-k$ il numeratore vale $1-2k\neq0$ (per $k\neq\tfrac12$), quindi verticale; all'infinito il limite è il rapporto dei coefficienti, $2$: orizzontale $y=2$. \\
    \textbf{Derivata}:
    \[ f'(x)=\dfrac{2(x+k)-(2x+1)}{(x+k)^2}=\dfrac{2k-1}{(x+k)^2} \]
    Il segno è costante, quello di $2k-1$: crescente in ciascun ramo se $k>\tfrac12$, decrescente se $k<\tfrac12$, mai estremi. Il grafico è un'iperbole equilatera con centro nell'intersezione degli asintoti, $(-k,2)$.
    \item Con $k=1$: $f(x)=\dfrac{2x+1}{x+1}$, che si annulla in $x=-\tfrac12$. \\
    $g=|f|$ ribalta sopra l'asse $x$ la parte di grafico negativa (tra $-1$ e $-\tfrac12$). In $x=-\tfrac12$:
    \[ f'\left(-\dfrac12\right)=\dfrac{1}{\left(\frac12\right)^2}=4\neq0 \]
    quindi le derivate laterali di $g$ sono $-4$ e $+4$: punto angoloso.
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=3.2, subtitle={Studio di $f(x)=x^2/(x-k)$ con parametro $k$}, tags={studio di funzione, parametrico}, difficulty=difficile]
Considera la funzione
\begin{equation*}
    f(x) = \frac{x^2}{x-k}
\end{equation*}
dove $k$ è un parametro reale, e indica con $y$ il suo grafico.
\begin{enumerate}
    \item Studia la funzione al variare del parametro $k$ e tracciane il grafico.
    \item Posto $k=0$ rappresenta la funzione $g(x) = |f(x)|$ e determina eventuali punti di non derivabilità della funzione.
\end{enumerate}
\begin{solution}
\begin{enumerate}
    \item Dominio $x\neq k$. Per $k\neq0$: asintoti $x=k$ e $y=x+k$; punti stazionari in $x=0$ e $x=2k$: per $k>0$ massimo $(0,0)$ e minimo $(2k,4k)$, per $k<0$ massimo $(2k,4k)$ e minimo $(0,0)$.
    \item Per $k=0$: $g(x)=|x|$ privata dell'origine; nel suo dominio è derivabile ovunque (il punto angoloso di $|x|$ cadrebbe in $x=0$, che è escluso).
\end{enumerate}
\end{solution}
\begin{explanation}
\begin{enumerate}
    \item \textbf{Asintoti.} Per $k\neq0$, in $x=k$ il numeratore vale $k^2\neq0$: asintoto verticale. Dividendo:
    \[\begin{aligned}
    x^2 &= (x-k)(x+k)+k^2\\
    f(x) &= x+k+\dfrac{k^2}{x-k} && \textcolor{gray}{\text{asintoto obliquo $y=x+k$}}
    \end{aligned}\]
    \textbf{Derivata.}
    \[ f'(x)=\dfrac{2x(x-k)-x^2}{(x-k)^2}=\dfrac{x(x-2k)}{(x-k)^2} \]
    Si annulla in $x=0$ e $x=2k$; il segno è quello di $x(x-2k)$: positivo fuori da $0$ e $2k$, negativo tra di essi. \\
    Se $k>0$ l'ordine è $0<2k$: massimo in $0$, minimo in $2k$; se $k<0$ l'ordine si inverte. I valori: $f(0)=0$, \ $f(2k)=\dfrac{4k^2}{k}=4k$.
    \item Con $k=0$: $f(x)=\dfrac{x^2}{x}=x$ per $x\neq0$. Allora $g(x)=|x|$ senza l'origine. \\
    $|x|$ ha un punto angoloso solo in $x=0$; poiché quel punto non appartiene al dominio, in tutti i punti del dominio $g$ è derivabile.
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=3.3, subtitle={Potenziale del deuterio: $V(r)=a|r|e^{1-br^2}$}, tags={studio di funzione, parametrico, fisica}, difficulty=difficile]
Il potenziale di un atomo di deuterio può essere rappresentato dalla seguente funzione:
\begin{equation*}
    V(r) = a|r|\cdot e^{1-br^2}
\end{equation*}
dove $a$ e $b$ sono due parametri reali, e $r$ la distanza dal centro.
\begin{enumerate}
    \item Determina i valori di $a$ e $b$, sapendo che il massimo valore di potenziale si trova nei punti di ascissa $1$ e $-1$ ed ha valore $\sqrt{e}$.
    \item Verificato che, in corrispondenza dei valori trovati precedentemente, l'espressione analitica della funzione è $V(r) = |r| \cdot e^{1-\frac{1}{2}r^2}$, tracciane il grafico.
    \item Verifica la presenza di un punto di non derivabilità nel centro dell'atomo e classificalo.
\end{enumerate}
\begin{solution}
\begin{enumerate}
    \item $a=1$, \ $b=\tfrac12$
    \item Grafico pari, nullo in $r=0$, con massimi $(\pm1,\sqrt e)$ e asintoto orizzontale $V=0$.
    \item In $r=0$ \textbf{punto angoloso}: derivate laterali $-e$ e $+e$.
\end{enumerate}
\end{solution}
\begin{explanation}
\begin{enumerate}
    \item La funzione è pari (dipende da $|r|$ e $r^2$): basta studiare $r>0$, dove $V=a\,r\,e^{1-br^2}$.
    \[\begin{aligned}
    V'(r) &= a\,e^{1-br^2}+a\,r\,e^{1-br^2}\cdot(-2br) && \textcolor{gray}{\text{regola del prodotto}}\\
    &= a\,e^{1-br^2}(1-2br^2)\\
    V'(1)=0 &\;\Rightarrow\; 1-2b=0 \;\Rightarrow\; b=\dfrac12 && \textcolor{gray}{\text{il massimo è in $r=1$}}\\
    V(1) &= a\,e^{1-\frac12}=a\sqrt e=\sqrt e \;\Rightarrow\; a=1
    \end{aligned}\]
    \item $V(r)=|r|\,e^{1-\frac12r^2}\geq0$, nulla solo in $r=0$. Per $r>0$ cresce fino a $r=1$ e poi decresce; per $r\to\infty$ l'esponenziale con esponente $-\tfrac12r^2$ prevale e $V\to0$. Per simmetria lo stesso a sinistra.
    \item Per $r>0$: \ $V'(r)=e^{1-\frac12r^2}(1-r^2)\to e^1\cdot1=e$. \\
    Per $r<0$ ($|r|=-r$) la derivata è l'opposta: $\to-e$. \\
    Derivate laterali \textbf{finite} e diverse: punto angoloso (non cuspide, che richiederebbe derivate infinite).
\end{enumerate}
\end{explanation}
\end{question}
