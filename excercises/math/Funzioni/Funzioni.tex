\begin{question}[ID=1.1, subtitle={Iniettività, suriettività e biettività da diagrammi a frecce}, difficulty=facile, tags={funzioni, insiemi}]
Stabilisci se ciascuna delle funzioni da $A$ a $B$ rappresentate nei seguenti diagrammi a frecce è iniettiva, suriettiva o biiettiva.
\newline

\centering
\includegraphics[width=\linewidth]{images/insiemi_funzioni.png}
\[ \text{a)} \hspace{60pt} \text{b)} \hspace{60pt} \text{c)} \]
\begin{solution}
\begin{enumerate}
    \item iniettiva, non suriettiva
    \item né iniettiva né suriettiva
    \item suriettiva, non iniettiva
\end{enumerate}
\end{solution}
\begin{explanation}
Una funzione da $A$ a $B$ è \textbf{iniettiva} se elementi diversi di $A$ hanno immagini diverse (in ogni elemento di $B$ arriva al più una freccia), \textbf{suriettiva} se ogni elemento di $B$ è immagine di qualcosa (in ogni elemento di $B$ arriva almeno una freccia), \textbf{biiettiva} se è entrambe le cose.
\begin{enumerate}
    \item $a\to e$, $b\to f$, $c\to h$: frecce che arrivano in elementi diversi, quindi iniettiva. In $g$ non arriva nessuna freccia: non suriettiva.
    \item In $f$ arrivano tre frecce (da $a$, $b$ e $d$): non iniettiva. In $e$ non arriva nessuna freccia: non suriettiva.
    \item Tutti gli elementi di $B$ ($e$, $f$, $g$, $h$) ricevono almeno una freccia: suriettiva. In $e$ arrivano due frecce (da $a$ e da $c$): non iniettiva.
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=1.2, subtitle={Riconoscere funzioni da diagrammi a frecce}, difficulty=facile, tags={funzioni, insiemi}]
Stabilisci quali delle relazioni tra $A$ a $B$ rappresentate nei seguenti diagrammi a frecce sono funzioni.
\newline

\centering
\includegraphics[width=0.9\linewidth]{images/insiemi_funzioni_1.png}
\[ \text{a)} \hspace{50pt} \text{b)} \hspace{50pt} \text{c)} \]
\begin{solution}
\begin{enumerate}
    \item non è una funzione
    \item è una funzione
    \item non è una funzione
\end{enumerate}
\end{solution}
\begin{explanation}
Una relazione da $A$ a $B$ è una \textbf{funzione} se da \emph{ogni} elemento di $A$ parte \emph{una e una sola} freccia (in $B$ invece possono arrivare più frecce, o nessuna).
\begin{enumerate}
    \item Da $a$ partono due frecce (verso $x$ e verso $y$): l'immagine di $a$ non è unica.
    \item Da ogni elemento di $A$ parte esattamente una freccia; il fatto che in $z$ ne arrivino tre e in $w$ nessuna non conta.
    \item Da $c$ non parte nessuna freccia: $c$ non ha immagine.
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=2.1, subtitle={Dominio di funzioni razionali}, difficulty=facile, tags={dominio,razionali}]
Determinare il dominio delle seguenti funzioni:
\begin{center}\begin{minipage}{0.92\linewidth}
\begin{multicols}{2}
\begin{enumerate}[leftmargin=3.2cm]
\item $f(x)=\dfrac{2x+1}{x-3}$
\item $f(x)=\dfrac{x^2-4}{x^2-9}$
\item $f(x)=\dfrac{5}{x^3+1}$
\item $f(x)=\dfrac{x-2}{\sqrt{x^2-1}}$
\end{enumerate}
\end{multicols}
\end{minipage}\end{center}
\begin{solution}
\begin{enumerate}
    \item $D=\mathbb{R}\setminus\{3\}$
    \item $D=\mathbb{R}\setminus\{-3,3\}$
    \item $D=\mathbb{R}\setminus\{-1\}$
    \item $D=(-\infty,-1)\cup(1,+\infty)$
\end{enumerate}
\end{solution}
\begin{explanation}
Condizioni di esistenza: denominatori $\neq0$; radicandi di radici di indice pari $\geq0$ (strettamente $>0$ se la radice è al denominatore); le radici di indice dispari non pongono condizioni.
\begin{enumerate}
    \item $x-3\neq0 \Rightarrow x\neq3$.
    \item $x^2-9\neq0 \Rightarrow x\neq\pm3$ (il numeratore non conta).
    \item $x^3+1\neq0 \Rightarrow x^3\neq-1 \Rightarrow x\neq-1$.
    \item La radice è al denominatore: $x^2-1>0$, valori esterni alle radici $\pm1$.
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=2.2, subtitle={Dominio di funzioni irrazionali con radicali}, difficulty=facile, tags={dominio,radicali}]
Determinare il dominio delle seguenti funzioni:

\begin{center}\begin{minipage}{0.92\linewidth}
\begin{multicols}{2}
\begin{enumerate}[leftmargin=3.2cm]
\item $f(x)=\sqrt{4x-1}$
\item $f(x)=\sqrt{2-|x|}$
\item $f(x)=\dfrac{\sqrt{x+6}}{x-1}$
\item $f(x)=\sqrt[3]{\dfrac{x-2}{x+3}}$
\end{enumerate}
\end{multicols}
\end{minipage}\end{center}
\begin{solution}
\begin{enumerate}
    \item $D=\left[\tfrac14,+\infty\right)$
    \item $D=[-2,2]$
    \item $D=[-6,1)\cup(1,+\infty)$
    \item $D=\mathbb{R}\setminus\{-3\}$
\end{enumerate}
\end{solution}
\begin{explanation}
Condizioni di esistenza: denominatori $\neq0$; radicandi di radici di indice pari $\geq0$ (strettamente $>0$ se la radice è al denominatore); le radici di indice dispari non pongono condizioni.
\begin{enumerate}
    \item $4x-1\geq0 \Rightarrow x\geq\tfrac14$.
    \item $2-|x|\geq0 \Rightarrow |x|\leq2 \Rightarrow -2\leq x\leq2$.
    \item $\begin{cases} x+6\geq0\\ x-1\neq0\end{cases} \Rightarrow x\geq-6,\ x\neq1$.
    \item La radice cubica esiste per ogni argomento: basta che la frazione esista, $x\neq-3$.
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=2.3, subtitle={Dominio con valore assoluto e composizioni}, difficulty=media, tags={dominio,assoluto,composizione}]
Determinare il dominio delle seguenti funzioni:

\begin{center}\begin{minipage}{0.92\linewidth}
\begin{multicols}{2}
\begin{enumerate}[leftmargin=3.2cm]
\item $f(x)=\sqrt{|x|-2}$
\item $f(x)=\dfrac{1}{|x-1|-2}$
\item $f(x)=\sqrt{(x-1)(5-x)}$
\item $f(x)=\dfrac{|x+2|}{\sqrt{x-3}}$
\end{enumerate}
\end{multicols}
\end{minipage}\end{center}
\begin{solution}
\begin{enumerate}
    \item $D=(-\infty,-2]\cup[2,+\infty)$
    \item $D=\mathbb{R}\setminus\{-1,3\}$
    \item $D=[1,5]$
    \item $D=(3,+\infty)$
\end{enumerate}
\end{solution}
\begin{explanation}
Condizioni di esistenza: denominatori $\neq0$; radicandi di radici di indice pari $\geq0$ (strettamente $>0$ se la radice è al denominatore); le radici di indice dispari non pongono condizioni.
\begin{enumerate}
    \item $|x|-2\geq0 \Rightarrow |x|\geq2 \Rightarrow x\leq-2 \vee x\geq2$.
    \item $|x-1|\neq2 \Rightarrow x-1\neq\pm2 \Rightarrow x\neq3,\ x\neq-1$.
    \item $(x-1)(5-x)\geq0$: i fattori sono concordi tra le radici $1$ e $5$ (estremi inclusi).
    \item Il numeratore $|x+2|$ esiste sempre; la radice al denominatore richiede $x-3>0$.
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=2.4, subtitle={Dominio di funzioni trigonometriche}, difficulty=media, tags={dominio,trigonometria}]
Determinare il dominio delle seguenti funzioni:

\begin{center}\begin{minipage}{0.92\linewidth}
\begin{multicols}{2}
\begin{enumerate}[leftmargin=3.2cm]
\item $f(x)=\dfrac{1}{\sin x}$
\item $f(x)=\tan(3x)$
\item $f(x)=\dfrac{1}{1-\cos(2x)}$
\item $f(x)=\sqrt{\sin x + 1}$
\end{enumerate}
\end{multicols}
\end{minipage}\end{center}
\begin{solution}
\begin{enumerate}
    \item $x\neq k\pi$
    \item $x\neq\tfrac\pi6+k\tfrac\pi3$
    \item $x\neq k\pi$
    \item $D=\mathbb{R}$
\end{enumerate}
\end{solution}
\begin{explanation}
Condizioni di esistenza: denominatori $\neq0$; radicandi di radici di indice pari $\geq0$ (strettamente $>0$ se la radice è al denominatore); le radici di indice dispari non pongono condizioni.
Con $k\in\mathbb{Z}$.
\begin{enumerate}
    \item $\sin x\neq0 \Rightarrow x\neq k\pi$.
    \item $\tan t$ non esiste per $t=\tfrac\pi2+k\pi$:
    \[ 3x\neq\dfrac\pi2+k\pi \;\Rightarrow\; x\neq\dfrac\pi6+k\dfrac\pi3 \]
    \item $\cos2x\neq1 \Rightarrow 2x\neq2k\pi \Rightarrow x\neq k\pi$.
    \item $\sin x\geq-1$ sempre, quindi $\sin x+1\geq0$ per ogni $x$.
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=2.5, subtitle={Dominio di funzioni composte con radicali annidati}, difficulty=difficile, tags={dominio,funzioni composte,radicali}]
Determinare il dominio delle seguenti funzioni:

\begin{center}\begin{minipage}{0.92\linewidth}
\begin{multicols}{2}
\begin{enumerate}[leftmargin=3.2cm]
\item $f(x)=\sqrt{\sqrt{x-1}-1}$
\item $f(x)=\dfrac{\sqrt{x-2}}{\sqrt{x-5}}$
\item $f(x)=\sqrt{\dfrac{x-4}{x-6}}$
\item $f(x)=\dfrac{\sqrt{|x+1|-1}}{x-2}$
\end{enumerate}
\end{multicols}
\end{minipage}\end{center}
\begin{solution}
\begin{enumerate}
    \item $D=[2,+\infty)$
    \item $D=(5,+\infty)$
    \item $D=(-\infty,4]\cup(6,+\infty)$
    \item $D=(-\infty,-2]\cup[0,2)\cup(2,+\infty)$
\end{enumerate}
\end{solution}
\begin{explanation}
Condizioni di esistenza: denominatori $\neq0$; radicandi di radici di indice pari $\geq0$ (strettamente $>0$ se la radice è al denominatore); le radici di indice dispari non pongono condizioni.
\begin{enumerate}
    \item Due condizioni, una per ogni radice:
    \[\begin{aligned}
    x-1 &\geq 0 && \textcolor{gray}{\text{radice interna}}\\
    \sqrt{x-1} &\geq 1 && \textcolor{gray}{\text{radice esterna}}\\
    x-1 &\geq 1 \;\Rightarrow\; x\geq2 && \textcolor{gray}{\text{elevo al quadrato (entrambi $\geq0$)}}
    \end{aligned}\]
    \item $\begin{cases} x-2\geq0\\ x-5>0\end{cases} \Rightarrow x>5$.
    \item $\dfrac{x-4}{x-6}\geq0$: concordi per $x<4$ o $x>6$; $x=4$ incluso, $x=6$ escluso.
    \item $|x+1|\geq1 \Rightarrow x+1\geq1 \vee x+1\leq-1 \Rightarrow x\geq0 \vee x\leq-2$; inoltre $x\neq2$.
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=4.1, subtitle={Funzione inversa — serie 1}, difficulty=media, tags={funzioni,inversa}]
Determinare, se possibile, la funzione inversa delle seguenti funzioni:

\begin{center}\begin{minipage}{0.92\linewidth}
\begin{multicols}{2}
\begin{enumerate}[leftmargin=3.2cm]
\item $f(x)=2x+3$
\item $f(x)=\frac{x-1}{3}$
\item $f(x)=\frac{1}{x+2}$
\item $f(x)=\sqrt{x+5}$
\end{enumerate}
\end{multicols}
\end{minipage}\end{center}
\begin{solution}
\begin{enumerate}
    \item $f^{-1}(x)=\dfrac{x-3}{2}$
    \item $f^{-1}(x)=3x+1$
    \item $f^{-1}(x)=\dfrac1x-2$ \ ($x\neq0$)
    \item $f^{-1}(x)=x^2-5$ \ ($x\geq0$)
\end{enumerate}
\end{solution}
\begin{explanation}
Per trovare l'inversa si risolve $y=f(x)$ rispetto a $x$ e poi si scambiano i nomi delle variabili. L'inversa esiste se $f$ è biiettiva (tra il suo dominio e la sua immagine); il dominio di $f^{-1}$ è l'immagine di $f$.
\begin{enumerate}
    \item $y=2x+3 \Rightarrow 2x=y-3 \Rightarrow x=\dfrac{y-3}{2}$.
    \item $y=\dfrac{x-1}{3} \Rightarrow x-1=3y \Rightarrow x=3y+1$.
    \item \[\begin{aligned}
    y &= \dfrac{1}{x+2}\\
    x+2 &= \dfrac1y && \textcolor{gray}{\text{passo ai reciproci, $y\neq0$}}\\
    x &= \dfrac1y-2
    \end{aligned}\]
    \item $y=\sqrt{x+5}$ con $y\geq0$: elevo al quadrato, $y^2=x+5 \Rightarrow x=y^2-5$. L'inversa ha dominio $x\geq0$ (l'immagine della radice).
\end{enumerate}
\end{explanation}

\end{question}

\begin{question}[ID=6.1, subtitle={Composizione e inverse di $f(x)=x-1$ e $g(x)=x^3$}, difficulty=difficile, tags={funzioni,composte,inversa}]
Date le funzioni
\[
f(x)=x-1,\quad g(x)=x^3,
\]
\begin{enumerate}
    \item Determinare \(f \circ g\) e \(g \circ f\);
    \item Giustificare perché le funzioni sono invertibili e determinare le espressioni di \(f^{-1}\) e \(g^{-1}\);
    \item Verificare che risulta $(f \circ g)^{-1} = g^{-1} \circ f^{-1}$
\end{enumerate}
\begin{solution}
\begin{enumerate}
    \item $(f\circ g)(x)=x^3-1$; \ $(g\circ f)(x)=(x-1)^3$
    \item $f^{-1}(x)=x+1$; \ $g^{-1}(x)=\sqrt[3]{x}$
    \item $(f\circ g)^{-1}(x)=\sqrt[3]{x+1}=(g^{-1}\circ f^{-1})(x)$
\end{enumerate}
\end{solution}
\begin{explanation}
\begin{enumerate}
    \item $f\big(g(x)\big)=g(x)-1=x^3-1$; \quad $g\big(f(x)\big)=\big(f(x)\big)^3=(x-1)^3$.
    \item Entrambe sono strettamente crescenti su $\mathbb{R}$ e assumono tutti i valori reali, quindi sono biiettive e invertibili.
    \[\begin{aligned}
    y=x-1 &\;\Rightarrow\; x=y+1\\
    y=x^3 &\;\Rightarrow\; x=\sqrt[3]{y}
    \end{aligned}\]
    \item Inverto direttamente $f\circ g$:
    \[\begin{aligned}
    y &= x^3-1\\
    x^3 &= y+1 \;\Rightarrow\; x=\sqrt[3]{y+1}
    \end{aligned}\]
    Calcolo invece $g^{-1}\circ f^{-1}$: prima $f^{-1}(x)=x+1$, poi $g^{-1}(x+1)=\sqrt[3]{x+1}$. \\
    I risultati coincidono: l'inversa di una composizione si ottiene componendo le inverse in ordine scambiato.
\end{enumerate}
\end{explanation}
\end{question}

$a+b=5$ \ ($a=2$, $b=3$)
\end{solution}

\[\begin{aligned}
f(f(x)) &= a(ax+b)+b=a^2x+ab+b\\
f(f(f(x))) &= a(a^2x+ab+b)+b=a^3x+a^2b+ab+b
\end{aligned}\]
Due polinomi sono uguali se hanno gli stessi coefficienti:
\[\begin{aligned}
a^3 &= 8 \;\Rightarrow\; a=2 && \textcolor{gray}{\text{coefficiente di $x$}}\\
b(a^2+a+1) &= 21 \;\Rightarrow\; 7b=21 \;\Rightarrow\; b=3 && \textcolor{gray}{\text{termine noto}}
\end{aligned}\]
\end{explanation}
\end{question}

\begin{question}[ID=fac.2, subtitle={FAC: funzione con $f(x)+f(1/(1-x))=x$, trovare $f(2)$}, headings=empty, difficulty=difficile, tags={funzioni,parametrico}]
\textbf{FAC)}
Sia $f(x)$ una funzione tale che $f(x) + f\Big(\dfrac{1}{1-x}\Big) = x$ per tutte le $x$ diverse da 0 e da 1. Quanto vale $f(2)$?
\begin{solution}
$f(2)=\dfrac74$
\end{solution}
\begin{explanation}
Scrivo l'uguaglianza per $x=2$ e per i valori che vi compaiono. Con $x=2$: $\tfrac{1}{1-2}=-1$; con $x=-1$: $\tfrac{1}{1+1}=\tfrac12$; con $x=\tfrac12$: $\tfrac{1}{1-\frac12}=2$ (si torna al punto di partenza).
\[\begin{aligned}
f(2)+f(-1) &= 2 && \textcolor{gray}{\text{con $x=2$}}\\
f(-1)+f\left(\dfrac12\right) &= -1 && \textcolor{gray}{\text{con $x=-1$}}\\
f\left(\dfrac12\right)+f(2) &= \dfrac12 && \textcolor{gray}{\text{con $x=\dfrac12$}}
\end{aligned}\]
Sommo la prima e la terza e sottraggo la seconda:
\[ 2f(2)=2+\dfrac12-(-1)=\dfrac72 \;\Rightarrow\; f(2)=\dfrac74 \]
\end{explanation}
\end{question}
