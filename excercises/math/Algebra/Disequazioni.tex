\begin{question}[ID=1.1, subtitle={Disequazioni di primo grado}, difficulty=facile, tags={disequazioni lineari}]
Risolvi le seguenti disequazioni di primo grado:
\begin{multicols}{2}
\begin{enumerate}[leftmargin=3.2cm]
    \item $x + 3 > 7$
    \item $x - 5 \leq 2$
    \item $2x > 8$
    \item $-3x \leq 9$
    \item $2x + 1 > 5$
    \item $3x - 4 \geq 2$
\end{enumerate}
\end{multicols}
\begin{solution}
\begin{enumerate}
    \item $x>4$
    \item $x\leq 7$
    \item $x>4$
    \item $x\geq -3$
    \item $x>2$
    \item $x\geq 2$
\end{enumerate}
\end{solution}
\begin{explanation}
Si isola la $x$ con i principi di equivalenza: un termine si può spostare da un membro all'altro cambiandogli il segno; si possono dividere entrambi i membri per lo stesso numero, ma se il numero è \textbf{negativo} il verso della disuguaglianza si inverte.
\begin{enumerate}
    \item \[\begin{aligned}
    x+3 &> 7\\
    x &> 7-3 && \textcolor{gray}{\text{sposto il $3$ al secondo membro cambiando segno}}\\
    x &> 4 && \textcolor{gray}{\text{calcolo}}
    \end{aligned}\]
    \item \[\begin{aligned}
    x-5 &\leq 2\\
    x &\leq 2+5 && \textcolor{gray}{\text{sposto il $-5$ cambiando segno}}\\
    x &\leq 7 && \textcolor{gray}{\text{calcolo}}
    \end{aligned}\]
    \item \[\begin{aligned}
    2x &> 8\\
    x &> \dfrac{8}{2} && \textcolor{gray}{\text{divido per $2$: è positivo, il verso non cambia}}\\
    x &> 4
    \end{aligned}\]
    \item \[\begin{aligned}
    -3x &\leq 9\\
    x &\geq \dfrac{9}{-3} && \textcolor{gray}{\text{divido per $-3$: è negativo, il verso si inverte}}\\
    x &\geq -3
    \end{aligned}\]
    \item \[\begin{aligned}
    2x+1 &> 5\\
    2x &> 4 && \textcolor{gray}{\text{sposto l'$1$ e calcolo}}\\
    x &> 2 && \textcolor{gray}{\text{divido per $2$}}
    \end{aligned}\]
    \item \[\begin{aligned}
    3x-4 &\geq 2\\
    3x &\geq 6 && \textcolor{gray}{\text{sposto il $-4$ e calcolo}}\\
    x &\geq 2 && \textcolor{gray}{\text{divido per $3$}}
    \end{aligned}\]
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=1.2, subtitle={Disequazioni di primo grado}, difficulty=facile, tags={disequazioni lineari}]
Risolvi le seguenti disequazioni di primo grado:
\begin{multicols}{2}
\begin{enumerate}[leftmargin=3.2cm]
    \item $5 - x < 3$
    \item $4 - 2x \geq 6$
    \item $2(x + 1) > 6$
    \item $3(x - 2) \leq 6$
    \item $2x + 3 > x - 1$
    \item $4x - 5 \leq 2x + 3$
\end{enumerate}
\end{multicols}
\begin{solution}
\begin{enumerate}
    \item $x>2$
    \item $x\leq -1$
    \item $x>2$
    \item $x\leq 4$
    \item $x>-4$
    \item $x\leq 4$
\end{enumerate}
\end{solution}
\begin{explanation}
\begin{enumerate}
    \item \[\begin{aligned}
    5-x &< 3\\
    -x &< -2 && \textcolor{gray}{\text{sposto il $5$ al secondo membro}}\\
    x &> 2 && \textcolor{gray}{\text{moltiplico per $-1$: il verso si inverte}}
    \end{aligned}\]
    \item \[\begin{aligned}
    4-2x &\geq 6\\
    -2x &\geq 2 && \textcolor{gray}{\text{sposto il $4$ e calcolo}}\\
    x &\leq -1 && \textcolor{gray}{\text{divido per $-2$: il verso si inverte}}
    \end{aligned}\]
    \item \[\begin{aligned}
    2(x+1) &> 6\\
    2x+2 &> 6 && \textcolor{gray}{\text{svolgo la parentesi}}\\
    2x &> 4 && \textcolor{gray}{\text{sposto il $2$}}\\
    x &> 2 && \textcolor{gray}{\text{divido per $2$}}
    \end{aligned}\]
    \item \[\begin{aligned}
    3(x-2) &\leq 6\\
    3x-6 &\leq 6 && \textcolor{gray}{\text{svolgo la parentesi}}\\
    3x &\leq 12 && \textcolor{gray}{\text{sposto il $-6$}}\\
    x &\leq 4 && \textcolor{gray}{\text{divido per $3$}}
    \end{aligned}\]
    \item \[\begin{aligned}
    2x+3 &> x-1\\
    2x-x &> -1-3 && \textcolor{gray}{\text{porto le $x$ a sinistra e i numeri a destra}}\\
    x &> -4 && \textcolor{gray}{\text{calcolo}}
    \end{aligned}\]
    \item \[\begin{aligned}
    4x-5 &\leq 2x+3\\
    4x-2x &\leq 3+5 && \textcolor{gray}{\text{porto le $x$ a sinistra e i numeri a destra}}\\
    2x &\leq 8\\
    x &\leq 4 && \textcolor{gray}{\text{divido per $2$}}
    \end{aligned}\]
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=1.3, subtitle={Disequazioni di primo grado con frazioni e parentesi}, difficulty=media, tags={disequazioni lineari}]
Risolvi le seguenti disequazioni di primo grado:
\begin{multicols}{2}
\begin{enumerate}[leftmargin=3.2cm]
    \item $3x - 2 > 5x - 6$
    \item $2 - x \geq x + 4$
    \item $\dfrac{x}{2} > 3$
    \item $\dfrac{3x - 1}{2} \leq 4$
    \item $\dfrac{x + 2}{3} > \dfrac{x - 1}{2}$
    \item $\dfrac{2x - 3}{4} \leq \dfrac{x + 5}{2}$
\end{enumerate}
\end{multicols}
\begin{solution}
\begin{enumerate}
    \item $x<2$
    \item $x\leq -1$
    \item $x>6$
    \item $x\leq 3$
    \item $x<7$
    \item $x\in\mathbb{R}$
\end{enumerate}
\end{solution}
\begin{explanation}
Quando ci sono frazioni si moltiplicano entrambi i membri per il m.c.m. dei denominatori (un numero positivo, quindi il verso non cambia).
\begin{enumerate}
    \item \[\begin{aligned}
    3x-2 &> 5x-6\\
    3x-5x &> -6+2 && \textcolor{gray}{\text{porto le $x$ a sinistra e i numeri a destra}}\\
    -2x &> -4\\
    x &< 2 && \textcolor{gray}{\text{divido per $-2$: il verso si inverte}}
    \end{aligned}\]
    \item \[\begin{aligned}
    2-x &\geq x+4\\
    -x-x &\geq 4-2 && \textcolor{gray}{\text{porto le $x$ a sinistra e i numeri a destra}}\\
    -2x &\geq 2\\
    x &\leq -1 && \textcolor{gray}{\text{divido per $-2$: il verso si inverte}}
    \end{aligned}\]
    \item \[\begin{aligned}
    \dfrac{x}{2} &> 3\\
    x &> 6 && \textcolor{gray}{\text{moltiplico per $2$}}
    \end{aligned}\]
    \item \[\begin{aligned}
    \dfrac{3x-1}{2} &\leq 4\\
    3x-1 &\leq 8 && \textcolor{gray}{\text{moltiplico per $2$}}\\
    3x &\leq 9\\
    x &\leq 3 && \textcolor{gray}{\text{divido per $3$}}
    \end{aligned}\]
    \item \[\begin{aligned}
    \dfrac{x+2}{3} &> \dfrac{x-1}{2}\\
    2(x+2) &> 3(x-1) && \textcolor{gray}{\text{moltiplico per il m.c.m. $6$}}\\
    2x+4 &> 3x-3 && \textcolor{gray}{\text{svolgo le parentesi}}\\
    -x &> -7\\
    x &< 7 && \textcolor{gray}{\text{moltiplico per $-1$: il verso si inverte}}
    \end{aligned}\]
    \item \[\begin{aligned}
    \dfrac{2x-3}{4} &\leq \dfrac{x+5}{2}\\
    2x-3 &\leq 2(x+5) && \textcolor{gray}{\text{moltiplico per il m.c.m. $4$}}\\
    2x-3 &\leq 2x+10\\
    0\cdot x &\leq 13 && \textcolor{gray}{\text{le $x$ si eliminano}}
    \end{aligned}\]
    La disuguaglianza $0\leq 13$ è vera per qualunque valore di $x$, quindi la disequazione è sempre verificata: $x\in\mathbb{R}$.
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=1.4, subtitle={Disequazioni di primo grado con parentesi e frazioni}, difficulty=difficile, tags={disequazioni lineari}]
Risolvi le seguenti disequazioni di primo grado:
\begin{multicols}{2}
\begin{enumerate}[leftmargin=3.2cm]
    \item $3(2x - 1) - 4x > 5$
    \item $2(3x + 1) - (x - 4) \leq 7$
    \item $3(2x-1)-2(x+4)>5$
    \item $5-2(3x-1)\leq 3(x+2)$
    \item $\dfrac{3x-2}{4}-\dfrac{x+1}{3}>1$
    \item $2-\dfrac{x-3}{2}\geq\dfrac{x+1}{3}$
\end{enumerate}
\end{multicols}
\begin{solution}
\begin{enumerate}
    \item $x>4$
    \item $x\leq\tfrac{1}{5}$
    \item $x>4$
    \item $x\geq\tfrac{1}{9}$
    \item $x>\tfrac{22}{5}$
    \item $x\leq\tfrac{19}{5}$
\end{enumerate}
\end{solution}
\begin{explanation}
\begin{enumerate}
    \item \[\begin{aligned}
    3(2x-1)-4x &> 5\\
    6x-3-4x &> 5 && \textcolor{gray}{\text{svolgo la parentesi}}\\
    2x &> 8 && \textcolor{gray}{\text{sommo i termini simili e sposto il $-3$}}\\
    x &> 4 && \textcolor{gray}{\text{divido per $2$}}
    \end{aligned}\]
    \item \[\begin{aligned}
    2(3x+1)-(x-4) &\leq 7\\
    6x+2-x+4 &\leq 7 && \textcolor{gray}{\text{svolgo le parentesi (il meno cambia i segni)}}\\
    5x+6 &\leq 7\\
    5x &\leq 1\\
    x &\leq \dfrac15 && \textcolor{gray}{\text{divido per $5$}}
    \end{aligned}\]
    \item \[\begin{aligned}
    3(2x-1)-2(x+4) &> 5\\
    6x-3-2x-8 &> 5 && \textcolor{gray}{\text{svolgo le parentesi}}\\
    4x-11 &> 5\\
    4x &> 16\\
    x &> 4 && \textcolor{gray}{\text{divido per $4$}}
    \end{aligned}\]
    \item \[\begin{aligned}
    5-2(3x-1) &\leq 3(x+2)\\
    5-6x+2 &\leq 3x+6 && \textcolor{gray}{\text{svolgo le parentesi}}\\
    -6x-3x &\leq 6-7 && \textcolor{gray}{\text{porto le $x$ a sinistra e i numeri a destra}}\\
    -9x &\leq -1\\
    x &\geq \dfrac19 && \textcolor{gray}{\text{divido per $-9$: il verso si inverte}}
    \end{aligned}\]
    \item \[\begin{aligned}
    \dfrac{3x-2}{4}-\dfrac{x+1}{3} &> 1\\
    3(3x-2)-4(x+1) &> 12 && \textcolor{gray}{\text{moltiplico per il m.c.m. $12$}}\\
    9x-6-4x-4 &> 12 && \textcolor{gray}{\text{svolgo le parentesi}}\\
    5x &> 22\\
    x &> \dfrac{22}{5} && \textcolor{gray}{\text{divido per $5$}}
    \end{aligned}\]
    \item \[\begin{aligned}
    2-\dfrac{x-3}{2} &\geq \dfrac{x+1}{3}\\
    12-3(x-3) &\geq 2(x+1) && \textcolor{gray}{\text{moltiplico per il m.c.m. $6$}}\\
    12-3x+9 &\geq 2x+2 && \textcolor{gray}{\text{svolgo le parentesi}}\\
    -5x &\geq -19\\
    x &\leq \dfrac{19}{5} && \textcolor{gray}{\text{divido per $-5$: il verso si inverte}}
    \end{aligned}\]
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=2.1, subtitle={Disequazioni frazionarie}, difficulty=facile, tags={disequazioni frazionarie}]
Risolvi le seguenti disequazioni frazionarie:
\begin{multicols}{2}
\begin{enumerate}[leftmargin=3.2cm]
    \item $\dfrac{x}{x-1} > 0$
    \item $\dfrac{x-2}{x+1} \geq 0$
    \item $\dfrac{2x+1}{x-3} > 0$
    \item $\dfrac{x+4}{2x-1} \leq 0$
    \item $\dfrac{x-1}{x+2} < 0$
    \item $\dfrac{3x-2}{x-2} \geq 0$
\end{enumerate}
\end{multicols}
\begin{solution}
\begin{enumerate}
    \item $x<0\vee x>1$
    \item $x\in(-\infty,-1)\cup[2,+\infty)$
    \item $x<-\tfrac12\vee x>3$
    \item $x\in[-4,\tfrac12)$
    \item $x\in(-2,1)$
    \item $x\in(-\infty,\tfrac23]\cup(2,+\infty)$
\end{enumerate}
\end{solution}
\begin{explanation}
Una frazione è positiva quando numeratore $N$ e denominatore $D$ hanno lo \textbf{stesso segno}, negativa quando hanno segni \textbf{opposti}. Si studia quindi il segno di $N$ e di $D$ separatamente e si confrontano. Il denominatore non può mai essere nullo, mentre il numeratore può annullarsi solo se nella disequazione c'è $\geq$ o $\leq$.
\begin{enumerate}
    \item $N=x>0$ per $x>0$; \quad $D=x-1>0$ per $x>1$. \\
    Per $x<0$ sono entrambi negativi (quoziente positivo), per $0<x<1$ sono discordi (negativo), per $x>1$ sono entrambi positivi (positivo). \\
    Serve il segno $+$: \ $x<0 \vee x>1$.
    \item $N=x-2\geq 0$ per $x\geq 2$; \quad $D=x+1>0$ per $x>-1$. \\
    Il quoziente è positivo per $x<-1$ e per $x>2$, si annulla in $x=2$ (accettato, c'è $\geq$), non esiste in $x=-1$. \\
    Soluzione: $x\in(-\infty,-1)\cup[2,+\infty)$.
    \item $N=2x+1>0$ per $x>-\tfrac12$; \quad $D=x-3>0$ per $x>3$. \\
    Concordi per $x<-\tfrac12$ e per $x>3$: \ $x<-\tfrac12 \vee x>3$.
    \item $N=x+4\geq 0$ per $x\geq -4$; \quad $D=2x-1>0$ per $x>\tfrac12$. \\
    Serve il segno $-$ (o zero): $N$ e $D$ sono discordi tra $-4$ e $\tfrac12$; $x=-4$ è accettato (numeratore nullo), $x=\tfrac12$ no. \\
    Soluzione: $x\in[-4,\tfrac12)$.
    \item $N=x-1>0$ per $x>1$; \quad $D=x+2>0$ per $x>-2$. \\
    Serve il segno $-$: discordi tra le due radici: $x\in(-2,1)$.
    \item $N=3x-2\geq 0$ per $x\geq\tfrac23$; \quad $D=x-2>0$ per $x>2$. \\
    Concordi per $x<\tfrac23$ e per $x>2$; $x=\tfrac23$ è accettato, $x=2$ no. \\
    Soluzione: $x\in(-\infty,\tfrac23]\cup(2,+\infty)$.
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=2.2, subtitle={Disequazioni frazionarie}, difficulty=media, tags={disequazioni frazionarie}]
Risolvi le seguenti disequazioni frazionarie:
\begin{multicols}{2}
\begin{enumerate}[leftmargin=3.2cm]
    \item $\dfrac{x+1}{x-4} > 1$
    \item $\dfrac{2x}{x-3} \leq 1$
    \item $\dfrac{x-2}{x+1} > \dfrac{x}{x+1}$
    \item $\dfrac{2x+3}{x-1} \geq \dfrac{x}{x-1}$
    \item $\dfrac{x+1}{x-2} < \dfrac{2x}{x-2}$
    \item $\dfrac{3x-1}{x+2} \leq \dfrac{x+4}{x+2}$
\end{enumerate}
\end{multicols}
\begin{solution}
\begin{enumerate}
    \item $x>4$
    \item $x\in[-3,3)$
    \item $x<-1$
    \item $x\in(-\infty,-3]\cup(1,+\infty)$
    \item $x\in(-\infty,1)\cup(2,+\infty)$
    \item $x\in(-2,\tfrac52]$
\end{enumerate}
\end{solution}
\begin{explanation}
Non si può moltiplicare per il denominatore, perché non se ne conosce il segno. Si porta tutto al primo membro, si riduce a un'unica frazione e se ne studia il segno.
\begin{enumerate}
    \item \[\begin{aligned}
    \dfrac{x+1}{x-4}-1 &> 0 && \textcolor{gray}{\text{porto tutto al primo membro}}\\
    \dfrac{x+1-(x-4)}{x-4} &> 0 && \textcolor{gray}{\text{denominatore comune}}\\
    \dfrac{5}{x-4} &> 0
    \end{aligned}\]
    Il numeratore $5$ è sempre positivo, quindi basta $x-4>0$: \ $x>4$.
    \item \[\begin{aligned}
    \dfrac{2x}{x-3}-1 &\leq 0\\
    \dfrac{2x-x+3}{x-3} &\leq 0 && \textcolor{gray}{\text{denominatore comune}}\\
    \dfrac{x+3}{x-3} &\leq 0
    \end{aligned}\]
    $N\geq0$ per $x\geq-3$, $D>0$ per $x>3$: sono discordi tra $-3$ e $3$; $x=-3$ annulla la frazione ed è accettato. \ $x\in[-3,3)$.
    \item \[\begin{aligned}
    \dfrac{x-2}{x+1}-\dfrac{x}{x+1} &> 0\\
    \dfrac{-2}{x+1} &> 0 && \textcolor{gray}{\text{stesso denominatore: sottraggo i numeratori}}
    \end{aligned}\]
    Il numeratore è negativo, quindi serve $x+1<0$: \ $x<-1$.
    \item \[\begin{aligned}
    \dfrac{2x+3-x}{x-1} &\geq 0\\
    \dfrac{x+3}{x-1} &\geq 0
    \end{aligned}\]
    Concordi per $x<-3$ e $x>1$; $x=-3$ accettato: \ $x\in(-\infty,-3]\cup(1,+\infty)$.
    \item \[\begin{aligned}
    \dfrac{x+1-2x}{x-2} &< 0\\
    \dfrac{1-x}{x-2} &< 0\\
    \dfrac{x-1}{x-2} &> 0 && \textcolor{gray}{\text{cambio segno al numeratore: il verso si inverte}}
    \end{aligned}\]
    Concordi per $x<1$ e per $x>2$: \ $x\in(-\infty,1)\cup(2,+\infty)$.
    \item \[\begin{aligned}
    \dfrac{3x-1-(x+4)}{x+2} &\leq 0\\
    \dfrac{2x-5}{x+2} &\leq 0
    \end{aligned}\]
    Discordi tra $-2$ e $\tfrac52$; $x=\tfrac52$ accettato: \ $x\in(-2,\tfrac52]$.
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=2.3, subtitle={Disequazioni frazionarie con semplificazioni}, difficulty=media, tags={disequazioni frazionarie}]
Risolvi le seguenti disequazioni frazionarie:
\begin{multicols}{2}
\begin{enumerate}[leftmargin=3.2cm]
    \item $\dfrac{x}{3x-1} > 1$
    \item $\dfrac{2}{x+1} \geq 2$
    \item $\dfrac{x+1}{x-2} - 1 > 0$
    \item $\dfrac{2x}{x+3} - 2 \leq 0$
    \item $\dfrac{x^2-1}{x-1} > 0$
    \item $\dfrac{x^2-4}{x+2} \leq 0$
\end{enumerate}
\end{multicols}
\begin{solution}
\begin{enumerate}
    \item $x\in(\tfrac13,\tfrac12)$
    \item $x\in(-1,0]$
    \item $x>2$
    \item $x>-3$
    \item $x\in(-1,1)\cup(1,+\infty)$
    \item $x\in(-\infty,-2)\cup(-2,2]$
\end{enumerate}
\end{solution}
\begin{explanation}
\begin{enumerate}
    \item \[\begin{aligned}
    \dfrac{x}{3x-1}-1 &> 0 && \textcolor{gray}{\text{porto tutto al primo membro}}\\
    \dfrac{x-3x+1}{3x-1} &> 0 && \textcolor{gray}{\text{denominatore comune}}\\
    \dfrac{1-2x}{3x-1} &> 0
    \end{aligned}\]
    $N>0$ per $x<\tfrac12$, $D>0$ per $x>\tfrac13$: sono concordi (entrambi positivi) solo per $\tfrac13<x<\tfrac12$.
    \item \[\begin{aligned}
    \dfrac{2}{x+1}-2 &\geq 0\\
    \dfrac{2-2x-2}{x+1} &\geq 0 && \textcolor{gray}{\text{denominatore comune}}\\
    \dfrac{-2x}{x+1} &\geq 0\\
    \dfrac{x}{x+1} &\leq 0 && \textcolor{gray}{\text{divido per $-2$: il verso si inverte}}
    \end{aligned}\]
    Discordi tra $-1$ e $0$; $x=0$ accettato: \ $x\in(-1,0]$.
    \item \[\begin{aligned}
    \dfrac{x+1-(x-2)}{x-2} &> 0\\
    \dfrac{3}{x-2} &> 0
    \end{aligned}\]
    Il numeratore è positivo, quindi $x-2>0$: \ $x>2$.
    \item \[\begin{aligned}
    \dfrac{2x-2(x+3)}{x+3} &\leq 0\\
    \dfrac{-6}{x+3} &\leq 0
    \end{aligned}\]
    Il numeratore è negativo, quindi la frazione è negativa quando $x+3>0$: \ $x>-3$.
    \item Condizione di esistenza: $x\neq 1$. Scompongo e semplifico:
    \[\begin{aligned}
    \dfrac{(x-1)(x+1)}{x-1} &> 0\\
    x+1 &> 0 && \textcolor{gray}{\text{semplifico $(x-1)$, valido per $x\neq1$}}\\
    x &> -1
    \end{aligned}\]
    Tolgo $x=1$: \ $x\in(-1,1)\cup(1,+\infty)$.
    \item Condizione di esistenza: $x\neq -2$.
    \[\begin{aligned}
    \dfrac{(x-2)(x+2)}{x+2} &\leq 0\\
    x-2 &\leq 0 && \textcolor{gray}{\text{semplifico $(x+2)$, valido per $x\neq-2$}}\\
    x &\leq 2
    \end{aligned}\]
    Tolgo $x=-2$: \ $x\in(-\infty,-2)\cup(-2,2]$.
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=2.4, subtitle={Disequazioni frazionarie con più fattori}, difficulty=difficile, tags={disequazioni frazionarie}]
Risolvi le seguenti disequazioni frazionarie:
\begin{multicols}{2}
\begin{enumerate}[leftmargin=3.2cm]
    \item $\dfrac{3}{x-2}+\dfrac{1}{2x+1}>0$
    \item $\dfrac{1}{x+1}-\dfrac{3}{x-1}\leq 0$
    \item $\dfrac{x-1}{x+2}>\dfrac{2x+3}{x-1}$
    \item $\dfrac{2x+1}{x-2}-\dfrac{1}{x+1}\geq 2$
    \item $\dfrac{x+2}{x-1}+\dfrac{x-3}{x+2}\leq 1$
    \item $\dfrac{1}{x-2}-\dfrac{2}{x+1}\leq 1$
\end{enumerate}
\end{multicols}
\begin{solution}
\begin{enumerate}
    \item $x\in(-\tfrac12,-\tfrac17)\cup(2,+\infty)$
    \item $x\in[-2,-1)\cup(1,+\infty)$
    \item $x\in\left(\tfrac{-9-\sqrt{61}}{2},-2\right)\cup\left(\tfrac{-9+\sqrt{61}}{2},1\right)$
    \item $x\in[-\tfrac74,-1)\cup(2,+\infty)$
    \item $x\in(-2,1)$
    \item $x\in(-\infty,-\sqrt7]\cup(-1,2)\cup[\sqrt7,+\infty)$
\end{enumerate}
\end{solution}
\begin{explanation}
Con più fattori si porta tutto a un'unica frazione, si trovano gli zeri di ogni fattore e si studia il segno con una tabella: in ogni intervallo tra due zeri consecutivi il segno del quoziente è costante.
\begin{enumerate}
    \item \[\begin{aligned}
    \dfrac{3}{x-2}+\dfrac{1}{2x+1} &> 0\\
    \dfrac{3(2x+1)+(x-2)}{(x-2)(2x+1)} &> 0 && \textcolor{gray}{\text{denominatore comune}}\\
    \dfrac{7x+1}{(x-2)(2x+1)} &> 0 && \textcolor{gray}{\text{semplifico il numeratore}}
    \end{aligned}\]
    Zeri: $-\tfrac12$ (denominatore), $-\tfrac17$ (numeratore), $2$ (denominatore). \\
    Segni del quoziente da sinistra: $-,\ +,\ -,\ +$. Prendo gli intervalli con $+$: \ $x\in(-\tfrac12,-\tfrac17)\cup(2,+\infty)$.
    \item \[\begin{aligned}
    \dfrac{(x-1)-3(x+1)}{(x+1)(x-1)} &\leq 0 && \textcolor{gray}{\text{denominatore comune}}\\
    \dfrac{-2x-4}{(x+1)(x-1)} &\leq 0\\
    \dfrac{x+2}{(x+1)(x-1)} &\geq 0 && \textcolor{gray}{\text{divido per $-2$: il verso si inverte}}
    \end{aligned}\]
    Zeri: $-2$, $-1$, $1$. Segni da sinistra: $-,\ +,\ -,\ +$. \\
    $x=-2$ è accettato (numeratore nullo): \ $x\in[-2,-1)\cup(1,+\infty)$.
    \item \[\begin{aligned}
    \dfrac{(x-1)^2-(2x+3)(x+2)}{(x+2)(x-1)} &> 0 && \textcolor{gray}{\text{denominatore comune}}\\
    \dfrac{x^2-2x+1-2x^2-7x-6}{(x+2)(x-1)} &> 0 && \textcolor{gray}{\text{sviluppo i prodotti}}\\
    \dfrac{-x^2-9x-5}{(x+2)(x-1)} &> 0\\
    \dfrac{x^2+9x+5}{(x+2)(x-1)} &< 0 && \textcolor{gray}{\text{cambio segno al numeratore}}
    \end{aligned}\]
    Il numeratore si annulla per $x=\tfrac{-9\pm\sqrt{81-20}}{2}=\tfrac{-9\pm\sqrt{61}}{2}$, cioè circa $-8{,}4$ e $-0{,}6$. \\
    Zeri in ordine: $\tfrac{-9-\sqrt{61}}{2}$, $-2$, $\tfrac{-9+\sqrt{61}}{2}$, $1$. Segni da sinistra: $+,\ -,\ +,\ -,\ +$. \\
    Prendo gli intervalli con $-$: \ $x\in\left(\tfrac{-9-\sqrt{61}}{2},-2\right)\cup\left(\tfrac{-9+\sqrt{61}}{2},1\right)$.
    \item \[\begin{aligned}
    \dfrac{2x+1}{x-2}-\dfrac{1}{x+1}-2 &\geq 0\\
    \dfrac{(2x+1)(x+1)-(x-2)-2(x-2)(x+1)}{(x-2)(x+1)} &\geq 0 && \textcolor{gray}{\text{denominatore comune}}\\
    \dfrac{2x^2+3x+1-x+2-2x^2+2x+4}{(x-2)(x+1)} &\geq 0 && \textcolor{gray}{\text{sviluppo i prodotti}}\\
    \dfrac{4x+7}{(x-2)(x+1)} &\geq 0
    \end{aligned}\]
    Zeri: $-\tfrac74$, $-1$, $2$. Segni da sinistra: $-,\ +,\ -,\ +$. \\
    $x=-\tfrac74$ accettato: \ $x\in[-\tfrac74,-1)\cup(2,+\infty)$.
    \item \[\begin{aligned}
    \dfrac{(x+2)^2+(x-3)(x-1)-(x-1)(x+2)}{(x-1)(x+2)} &\leq 0 && \textcolor{gray}{\text{denominatore comune}}\\
    \dfrac{x^2+4x+4+x^2-4x+3-x^2-x+2}{(x-1)(x+2)} &\leq 0 && \textcolor{gray}{\text{sviluppo i prodotti}}\\
    \dfrac{x^2-x+9}{(x-1)(x+2)} &\leq 0
    \end{aligned}\]
    Il numeratore ha $\Delta=1-36<0$: è \textbf{sempre positivo} e non si annulla mai. \\
    Il quoziente è negativo quando lo è il denominatore, cioè tra le sue radici: \ $x\in(-2,1)$.
    \item \[\begin{aligned}
    \dfrac{(x+1)-2(x-2)-(x-2)(x+1)}{(x-2)(x+1)} &\leq 0 && \textcolor{gray}{\text{denominatore comune}}\\
    \dfrac{x+1-2x+4-x^2+x+2}{(x-2)(x+1)} &\leq 0 && \textcolor{gray}{\text{sviluppo i prodotti}}\\
    \dfrac{7-x^2}{(x-2)(x+1)} &\leq 0\\
    \dfrac{x^2-7}{(x-2)(x+1)} &\geq 0 && \textcolor{gray}{\text{cambio segno al numeratore}}
    \end{aligned}\]
    Zeri: $-\sqrt7$, $-1$, $2$, $\sqrt7$. Segni da sinistra: $+,\ -,\ +,\ -,\ +$. \\
    $x=\pm\sqrt7$ accettati: \ $x\in(-\infty,-\sqrt7]\cup(-1,2)\cup[\sqrt7,+\infty)$.
\end{enumerate}
\end{explanation}
