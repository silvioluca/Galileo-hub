%====================================================%
\begin{question}[ID=1.1, subtitle={Riconoscimento di funzioni esponenziali}, difficulty=facile, tags={definizione, esponenziali}]
Descrivere quali fra le seguenti funzioni sono esponenziali e spiegare il motivo
\begin{center}\begin{minipage}{0.92\linewidth}
\begin{multicols}{2}
\begin{enumerate}[leftmargin=3.2cm]
\item $y=3^{-x}$
\item $y=(-2)^x$
\item $y=0{,}5^{2x}$
\item $y=(\sqrt{3})^x$
\end{enumerate}
\end{multicols}
\end{minipage}\end{center}
\begin{solution}
\begin{enumerate}
    \item esponenziale
    \item non esponenziale
    \item esponenziale
    \item esponenziale
\end{enumerate}
\end{solution}
\begin{explanation}
Una funzione esponenziale ha la forma $y=a^x$ con base $a>0$ e $a\neq1$ (con base negativa, ad esempio, $a^{1/2}$ non esisterebbe).
\begin{enumerate}
    \item $3^{-x}=\left(\tfrac13\right)^x$: base $\tfrac13$, positiva e diversa da $1$.
    \item Base $-2<0$: non è definita per molti valori (es. $(-2)^{1/2}$), quindi non è esponenziale.
    \item $0{,}5^{2x}=(0{,}25)^x$: base $0{,}25$, accettabile.
    \item Base $\sqrt3\approx1{,}73$, positiva e diversa da $1$.
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=2.1, subtitle={Associazione tra funzioni esponenziali e grafici}, difficulty=facile, tags={grafici, esponenziali}]
Associare a ogni funzione il grafico corretto:
\begin{center}\begin{minipage}{0.92\linewidth}
\begin{multicols}{2}
\begin{enumerate}[label=\arabic*),leftmargin=2cm]
\item $y=\Big(\dfrac{3}{2}\Big)^x$
\item $y=\Big(\dfrac{1}{2}\Big)^x$
\item $y=4^{x}$
\item $y=\Big(\dfrac{1}{3}\Big)^{\frac{x}{2}}$
\end{enumerate}
\end{multicols}
\end{minipage}\end{center}

\begin{center}
\includegraphics[width=\textwidth]{images/grafici_esponenziali.png}
\end{center}
\vspace{-30pt}
\[ \text{a)} \hspace{30pt} \text{b)} \hspace{30pt} \text{c)} \hspace{30pt} \text{d)} \]
\begin{solution}
1) $\to$ c; \quad 2) $\to$ a; \quad 3) $\to$ b; \quad 4) $\to$ d
\end{solution}
\begin{explanation}
Tutti i grafici passano per $(0,1)$. Le funzioni crescenti hanno base $>1$ (funzioni 1 e 3), le decrescenti base $<1$ (2 e 4). Per distinguerle uso i punti segnati sui grafici:
\begin{itemize}
    \item grafico c (crescente): passa per $\left(1,\tfrac32\right)$, e $\left(\tfrac32\right)^1=\tfrac32$: è la funzione 1;
    \item grafico b (crescente): passa per $\left(\tfrac12,2\right)$, e $4^{1/2}=2$: è la funzione 3;
    \item grafico a (decrescente): passa per $(-1,2)$, e $\left(\tfrac12\right)^{-1}=2$: è la funzione 2;
    \item grafico d (decrescente): passa per $(-2,3)$, e $\left(\tfrac13\right)^{-2/2}=3$: è la funzione 4.
\end{itemize}
\end{explanation}
\end{question}

\begin{question}[ID=3.1, subtitle={Equazioni esponenziali elementari}, difficulty=facile, tags={equazioni, esponenziali}]
Calcolare il valore di $x$ nelle seguenti equazioni:
\begin{center}\begin{minipage}{0.92\linewidth}
\begin{multicols}{2}
\begin{enumerate}[leftmargin=3.2cm]
\item $3^{x} = 81$
\item $2^{x+1} = 4 \color{gray}$
\item $\color{gray}\dfrac{9}{3^x} = 27$
\item $e^{2x-1} = \dfrac{1}{e^2}$
\item $25 \cdot 5^{2x} = \dfrac{1}{5} \cdot 5^{x-1}$
\item $9^{-x} = -3$
\item $2^{2x} - 2^{x+2} + 4 = 0 \color{gray}$
\item $\color{gray} 5^{2x} + 3 \cdot 5^x + 2 = 0$
\end{enumerate}
\end{multicols}
\end{minipage}\end{center}
\begin{solution}
\begin{enumerate}
    \item $x=4$
    \item $x=1$
    \item $x=-1$
    \item $x=-\tfrac12$
    \item $x=-4$
    \item impossibile
    \item $x=1$
    \item impossibile
\end{enumerate}
\end{solution}
\begin{explanation}
\textbf{Stessa base}: si scrivono i due membri come potenze della stessa base; poiché l'esponenziale è iniettiva, $a^{f(x)}=a^{g(x)} \iff f(x)=g(x)$.
\begin{enumerate}
    \item $3^x=3^4 \;\Rightarrow\; x=4$.
    \item $2^{x+1}=2^2 \;\Rightarrow\; x+1=2 \;\Rightarrow\; x=1$.
    \item \[\begin{aligned}
    \dfrac{3^2}{3^x} &= 3^3\\
    3^{2-x} &= 3^3 && \textcolor{gray}{\text{quoziente di potenze}}\\
    2-x &= 3 \;\Rightarrow\; x=-1
    \end{aligned}\]
    \item $e^{2x-1}=e^{-2} \;\Rightarrow\; 2x-1=-2 \;\Rightarrow\; x=-\tfrac12$.
    \item \[\begin{aligned}
    5^2\cdot5^{2x} &= 5^{-1}\cdot5^{x-1}\\
    5^{2x+2} &= 5^{x-2} && \textcolor{gray}{\text{prodotto: sommo gli esponenti}}\\
    2x+2 &= x-2 \;\Rightarrow\; x=-4
    \end{aligned}\]
    \item Una potenza di base positiva è sempre positiva: non può valere $-3$. Impossibile.
    \item Con $t=2^x$ (e $2^{2x}=t^2$, $2^{x+2}=4t$):
    \[\begin{aligned}
    t^2-4t+4 &= 0\\
    (t-2)^2 &= 0 \;\Rightarrow\; t=2 \;\Rightarrow\; 2^x=2 \;\Rightarrow\; x=1
    \end{aligned}\]
    \item Con $t=5^x>0$: \ $t^2+3t+2=0 \Rightarrow (t+1)(t+2)=0 \Rightarrow t=-1,\ t=-2$. \\
    Entrambe negative, non accettabili: impossibile (del resto il primo membro è somma di termini positivi).
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=3.4, subtitle={Equazioni con radicali e raccoglimento}, difficulty=media, tags={equazioni, esponenziali, radicali, raccoglimento}]
Risolvi le seguenti equazioni esponenziali:
\begin{center}\begin{minipage}{0.92\linewidth}
\begin{multicols}{2}
\begin{enumerate}[leftmargin=3.2cm]
\item $\sqrt{8^{x}} = 4^{x-1}$
\item $3^{x+1} + 3^{x} = 36$
\item $5^{x+1} - 5^{x} + 5^{x-1} = 105$
\item $\dfrac{\sqrt[3]{2^{x-1}}}{\sqrt{4^{x+1}}} = \sqrt{8}$
\end{enumerate}
\end{multicols}
\end{minipage}\end{center}
\begin{solution}
\begin{enumerate}
    \item $x=4$
    \item $x=2$
    \item $x=2$
    \item $x=-\tfrac{17}{4}$
\end{enumerate}
\end{solution}
\begin{explanation}
\textbf{Stessa base}: si scrivono i due membri come potenze della stessa base; poiché l'esponenziale è iniettiva, $a^{f(x)}=a^{g(x)} \iff f(x)=g(x)$.
\begin{enumerate}
    \item \[\begin{aligned}
    (2^{3x})^{1/2} &= 2^{2(x-1)}\\
    \dfrac32x &= 2x-2 \;\Rightarrow\; x=4
    \end{aligned}\]
    \item Raccolgo la potenza con l'esponente più basso, $3^x$:
    \[\begin{aligned}
    3^x(3+1) &= 36\\
    3^x &= 9 \;\Rightarrow\; x=2
    \end{aligned}\]

    \[\begin{aligned}
    5^{x-1}(5^2-5+1) &= 105\\

    \end{aligned}\]
    \item \[\begin{aligned}
    \dfrac{2^{(x-1)/3}}{2^{x+1}} &= 2^{3/2} && \textcolor{gray}{\text{$\sqrt{4^{x+1}}=2^{x+1}$, \ $\sqrt8=2^{3/2}$}}\\
    \dfrac{x-1}{3}-(x+1) &= \dfrac32\\
    2x-2-6x-6 &= 9 && \textcolor{gray}{\text{moltiplico per $6$}}\\
    x &= -\dfrac{17}{4}
    \end{aligned}\]
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=3.5, subtitle={Legge di annullamento del prodotto e incognita ausiliaria}, difficulty=difficile, tags={equazioni, esponenziali, sostituzione}]
Risolvi le seguenti equazioni esponenziali:
\begin{center}\begin{minipage}{0.92\linewidth}
\begin{multicols}{2}
\begin{enumerate}[leftmargin=3.2cm]
\item $(2^{x} - 8)(3^{x} + 1) = 0$
\item $3^{x^2-2x} = 27$
\item $9^{x} - 8 \cdot 3^{x} - 9 = 0$
\item $2^{2x} - 5 \cdot 2^{x} + 4 = 0$
\end{enumerate}
\end{multicols}
\end{minipage}\end{center}
\begin{solution}
\begin{enumerate}
    \item $x=3$
    \item $x=-1\vee x=3$
    \item $x=2$
    \item $x=0\vee x=2$
\end{enumerate}
\end{solution}
\begin{explanation}
\textbf{Incognita ausiliaria}: si pone $t=a^x$ (con $t>0$), si risolve l'equazione in $t$, si scartano le soluzioni $t\leq0$ e si torna alla $x$.
\begin{enumerate}
    \item Legge di annullamento del prodotto: $2^x=8 \Rightarrow x=3$, oppure $3^x=-1$, impossibile.
    \item \[\begin{aligned}
    3^{x^2-2x} &= 3^3\\
    x^2-2x-3 &= 0 \;\Rightarrow\; (x-3)(x+1)=0
    \end{aligned}\]
    \item $t=3^x$: \ $t^2-8t-9=0 \Rightarrow t=9$ o $t=-1$ (scartata). $3^x=9 \Rightarrow x=2$.
    \item $t=2^x$: \ $t^2-5t+4=0 \Rightarrow t=1,\ t=4 \Rightarrow x=0,\ x=2$.
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=3.6, subtitle={Equazioni esponenziali con incognita ausiliaria}, difficulty=difficile, tags={equazioni, esponenziali, sostituzione}]
Risolvi le seguenti equazioni esponenziali introducendo un'opportuna incognita ausiliaria:
\begin{center}\begin{minipage}{0.92\linewidth}
\begin{multicols}{2}
\begin{enumerate}[leftmargin=3.2cm]
\item $3^{x} + 3^{2-x} = 10$
\item $\left(\dfrac{2}{3}\right)^{x} + \left(\dfrac{3}{2}\right)^{x} = \dfrac{97}{36}$
\item $8^{x} - 6 \cdot 4^{x} + 8 \cdot 2^{x} = 0$
\item $4^{x} - 2^{x+2} - 2^{x+\frac{1}{2}} + 4\sqrt{2} = 0$
\end{enumerate}
\end{multicols}
\end{minipage}\end{center}
\begin{solution}
\begin{enumerate}
    \item $x=0\vee x=2$
    \item $x=-2\vee x=2$
    \item $x=1\vee x=2$
    \item $x=\tfrac12\vee x=2$
\end{enumerate}
\end{solution}
\begin{explanation}
\textbf{Incognita ausiliaria}: si pone $t=a^x$ (con $t>0$), si risolve l'equazione in $t$, si scartano le soluzioni $t\leq0$ e si torna alla $x$.
\begin{enumerate}
    \item $3^{2-x}=\dfrac{9}{3^x}$; con $t=3^x$:
    \[\begin{aligned}
    t+\dfrac9t &= 10\\
    t^2-10t+9 &= 0 && \textcolor{gray}{\text{moltiplico per $t>0$}}\\
    t=1,\ t=9 &\;\Rightarrow\; x=0,\ x=2
    \end{aligned}\]
    \item $\left(\tfrac32\right)^x=\dfrac1t$ con $t=\left(\tfrac23\right)^x$:
    \[\begin{aligned}
    t+\dfrac1t &= \dfrac{97}{36}\\
    36t^2-97t+36 &= 0\\
    t &= \dfrac{97\pm\sqrt{9409-5184}}{72}=\dfrac{97\pm65}{72} \;\Rightarrow\; t=\dfrac94,\ t=\dfrac49
    \end{aligned}\]
    $\left(\tfrac23\right)^x=\tfrac49=\left(\tfrac23\right)^2 \Rightarrow x=2$; \ $\left(\tfrac23\right)^x=\tfrac94=\left(\tfrac23\right)^{-2} \Rightarrow x=-2$.
    \item $t=2^x$: $8^x=t^3$, $4^x=t^2$.
    \[\begin{aligned}
    t^3-6t^2+8t &= 0\\
    t(t-2)(t-4) &= 0
    \end{aligned}\]
    $t=0$ non accettabile; $t=2 \Rightarrow x=1$; $t=4 \Rightarrow x=2$.
    \item $t=2^x$: \ $2^{x+2}=4t$, \ $2^{x+1/2}=\sqrt2\,t$.
    \[\begin{aligned}
    t^2-4t-\sqrt2\,t+4\sqrt2 &= 0\\
    t(t-4)-\sqrt2(t-4) &= 0 && \textcolor{gray}{\text{raccoglimento parziale}}\\
    (t-4)(t-\sqrt2) &= 0
    \end{aligned}\]
    $2^x=4 \Rightarrow x=2$; \ $2^x=\sqrt2=2^{1/2} \Rightarrow x=\tfrac12$.
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=3.10, subtitle={Equazioni con sostituzione: forme fratte e radicali}, difficulty=difficile, tags={equazioni, esponenziali, sostituzione, fratte}]
Risolvi le seguenti equazioni esponenziali introducendo un'opportuna incognita ausiliaria:
\begin{center}\begin{minipage}{0.92\linewidth}
\begin{multicols}{2}
\begin{enumerate}[leftmargin=3.2cm]

\begin{question}[ID=6.1, subtitle={Tempo di dimezzamento dello iodio-131}, difficulty=media, tags={problemi, esponenziali, decadimento, tempo di dimezzamento}]
Lo iodio-131 è un isotopo radioattivo usato in medicina nucleare per la diagnosi e la cura delle malattie della tiroide. Il suo \textbf{tempo di dimezzamento} è di $8$ giorni: ogni $8$ giorni la quantità di sostanza ancora radioattiva si riduce alla metà. Un ospedale riceve un campione contenente $80\,\text{mg}$ di iodio-131.
\begin{enumerate}[label=\alph*),leftmargin=1.8cm]
    \item Scrivi la legge $m(t)$ che esprime la massa di iodio-131 ancora presente dopo $t$ giorni.
    \item Calcola la massa rimasta dopo $24$ giorni.
    \item Quale percentuale della massa iniziale è ancora presente dopo $16$ giorni?
    \item Il campione non è più utilizzabile quando la massa scende sotto $2{,}5\,\text{mg}$. Per quanti giorni può essere usato?
\end{enumerate}
\begin{solution}
\begin{enumerate}[label=\alph*),leftmargin=1.8cm]
    \item $m(t)=80\cdot\left(\tfrac12\right)^{t/8}$
    \item $10$ mg
    \item $25\%$ (cioè $20$ mg)
    \item per $40$ giorni
\end{enumerate}
\end{solution}
\begin{explanation}
\begin{enumerate}[label=\alph*),leftmargin=1.8cm]
    \item Ogni $8$ giorni la massa si moltiplica per $\tfrac12$; in $t$ giorni ci sono $\tfrac t8$ periodi di dimezzamento:
    \[ m(t)=80\cdot\left(\dfrac12\right)^{t/8} \]
    \item $24$ giorni sono $3$ dimezzamenti: \ $m(24)=80\cdot\tfrac18=10$ mg.
    \item $16$ giorni sono $2$ dimezzamenti: resta $\tfrac14$ della massa, cioè il $25\%$.
    \item \[\begin{aligned}
    80\left(\dfrac12\right)^{t/8} &\geq 2{,}5\\
    \left(\dfrac12\right)^{t/8} &\geq \dfrac{1}{32}=\left(\dfrac12\right)^5 && \textcolor{gray}{\text{divido per $80$}}\\
    \dfrac t8 &\leq 5 && \textcolor{gray}{\text{base $<1$: il verso si inverte}}\\
    t &\leq 40
    \end{aligned}\]
\end{enumerate}
\end{explanation}
\end{question}

\end{enumerate}
\begin{solution}
\begin{enumerate}[label=\alph*),leftmargin=1.8cm]

    \item \[\begin{aligned}
    54\left(\dfrac13\right)^{t/3} &\geq 2\\
    \left(\dfrac13\right)^{t/3} &\geq \dfrac1{27}=\left(\dfrac13\right)^3\\
    \dfrac t3 &\leq 3 \;\Rightarrow\; t\leq9 && \textcolor{gray}{\text{base $<1$: verso invertito}}
    \end{aligned}\]
    \item Dopo $6$ ore restano $6$ mg/L; la nuova dose ne aggiunge $54$: \ $6+54=60$ mg/L. Dopo altre $3$ ore la concentrazione si riduce a un terzo: $20$ mg/L.
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=6.3, subtitle={Carica virale e terapia antivirale}, difficulty=difficile, tags={problemi, esponenziali, crescita, virus}]
In un paziente appena contagiato da un virus si misura una carica virale di $500$ copie per millilitro di sangue. In assenza di cure il virus si replica rapidamente: la carica virale \textbf{raddoppia ogni $6$ ore}.
\begin{enumerate}[label=\alph*),leftmargin=1.8cm]
    \item Scrivi la legge $V(t)$ che esprime la carica virale dopo $t$ ore dal contagio.
    \item Calcola la carica virale dopo un giorno.
    \item Dopo quante ore la carica virale raggiunge le $64\,000$ copie/mL?
    \item Quando la carica virale raggiunge le $64\,000$ copie/mL il paziente inizia una terapia antivirale, grazie alla quale la carica si riduce a \textbf{un quarto ogni $12$ ore}. Dopo quante ore di terapia la carica virale scende sotto le $1000$ copie/mL?
\end{enumerate}
\begin{solution}
\begin{enumerate}[label=\alph*),leftmargin=1.8cm]
    \item $V(t)=500\cdot2^{t/6}$
    \item $8000$ copie/mL
    \item dopo $42$ ore
    \item dopo più di $36$ ore di terapia
\end{enumerate}
\end{solution}
\begin{explanation}
\begin{enumerate}[label=\alph*),leftmargin=1.8cm]
    \item Raddoppio ogni $6$ ore: $V(t)=500\cdot2^{t/6}$.
    \item $V(24)=500\cdot2^4=8000$.
    \item \[\begin{aligned}
    500\cdot2^{t/6} &= 64\,000\\
    2^{t/6} &= 128=2^7 && \textcolor{gray}{\text{divido per $500$}}\\
    t &= 42
    \end{aligned}\]
    \item Con $s$ ore di terapia: $W(s)=64\,000\cdot\left(\tfrac14\right)^{s/12}$.
    \[\begin{aligned}
    64\,000\left(\dfrac14\right)^{s/12} &< 1000\\
    \left(\dfrac14\right)^{s/12} &< \dfrac1{64}=\left(\dfrac14\right)^3\\
    \dfrac{s}{12} &> 3 \;\Rightarrow\; s>36 && \textcolor{gray}{\text{base $<1$: verso invertito}}
    \end{aligned}\]
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=6.4, subtitle={Raffreddamento di una tazza di tè}, difficulty=difficile, tags={problemi, esponenziali, decadimento, raffreddamento}]
Una tazza di tè appena versata si trova alla temperatura di $80\,{}^\circ\text{C}$ in una stanza a $20\,{}^\circ\text{C}$. Secondo la legge del raffreddamento di Newton, a diminuire in modo esponenziale non è la temperatura del tè, ma la \textbf{differenza} tra la temperatura del tè e quella dell'ambiente: in questo caso tale differenza si dimezza ogni $5$ minuti. La temperatura dopo $t$ minuti è quindi
\[
T(t) = 20 + 60 \cdot \left(\dfrac{1}{2}\right)^{\frac{t}{5}}.
\]
\begin{enumerate}[label=\alph*),leftmargin=1.8cm]
    \item Verifica che la legge fornisce la temperatura iniziale corretta e calcola la temperatura dopo $10$ minuti.
    \item Dopo quanti minuti il tè raggiunge la temperatura di $27{,}5\,{}^\circ\text{C}$?
    \item Il tè è considerato freddo quando la sua temperatura scende sotto $23{,}75\,{}^\circ\text{C}$. Dopo quanti minuti accade?
    \item Il tè può raggiungere la temperatura di $20\,{}^\circ\text{C}$? E scendere sotto questo valore? Giustifica la risposta.
\end{enumerate}
\begin{solution}
\begin{enumerate}[label=\alph*),leftmargin=1.8cm]
    \item $T(0)=80\,{}^\circ\text{C}$; \ $T(10)=35\,{}^\circ\text{C}$
    \item dopo $15$ minuti
    \item dopo $20$ minuti
    \item no: $T(t)>20\,{}^\circ\text{C}$ per ogni $t$ ($T=20$ è un asintoto orizzontale)
\end{enumerate}
\end{solution}
\begin{explanation}
\begin{enumerate}[label=\alph*),leftmargin=1.8cm]
    \item $T(0)=20+60\cdot1=80$. \quad $T(10)=20+60\cdot\left(\tfrac12\right)^2=20+15=35$.
    \item \[\begin{aligned}
    20+60\left(\dfrac12\right)^{t/5} &= 27{,}5\\
    \left(\dfrac12\right)^{t/5} &= \dfrac{7{,}5}{60}=\dfrac18=\left(\dfrac12\right)^3\\
    t &= 15
    \end{aligned}\]
    \item \[\begin{aligned}
    60\left(\dfrac12\right)^{t/5} &< 3{,}75\\
    \left(\dfrac12\right)^{t/5} &< \dfrac1{16}=\left(\dfrac12\right)^4\\
    \dfrac t5 &> 4 \;\Rightarrow\; t>20 && \textcolor{gray}{\text{base $<1$}}
    \end{aligned}\]
    \item $60\left(\tfrac12\right)^{t/5}$ è sempre positivo (una potenza di base positiva non si annulla mai), quindi $T(t)>20$ sempre. Al crescere di $t$ quel termine tende a $0$: la temperatura si avvicina a $20\,{}^\circ\text{C}$ senza mai raggiungerla.
\end{enumerate}
\end{explanation}
\end{question}

\begin{question}[ID=6.5, subtitle={Due colonie batteriche a confronto}, difficulty=difficile, tags={problemi, esponenziali, crescita, decadimento, batteri}]
In un laboratorio si studiano due colonie dello stesso batterio. La colonia A, lasciata in un terreno di coltura favorevole, contiene inizialmente $200$ batteri e il loro numero \textbf{raddoppia ogni ora}. La colonia B contiene inizialmente $51\,200$ batteri e viene trattata con un antibiotico, per effetto del quale il numero di batteri \textbf{si dimezza ogni ora}.
\begin{enumerate}[label=\alph*),leftmargin=1.8cm]
    \item Scrivi le leggi $N_A(t)$ e $N_B(t)$ che esprimono il numero di batteri delle due colonie dopo $t$ ore.
    \item Calcola il numero di batteri di ciascuna colonia dopo $2$ ore.
    \item Dopo quante ore le due colonie hanno lo stesso numero di batteri? Quanti sono?
    \item L'antibiotico è considerato efficace quando la colonia B scende sotto i $100$ batteri. Dopo quante ore accade? Quanti batteri contiene in quel momento la colonia A?
\end{enumerate}
\begin{solution}
\begin{enumerate}[label=\alph*),leftmargin=1.8cm]
    \item $N_A(t)=200\cdot2^t$; \ $N_B(t)=51\,200\cdot2^{-t}$
    \item $N_A(2)=800$; \ $N_B(2)=12\,800$
    \item dopo $4$ ore, con $3200$ batteri ciascuna
    \item dopo $9$ ore; la colonia A ha $102\,400$ batteri
\end{enumerate}
\end{solution}
\begin{explanation}
\begin{enumerate}[label=\alph*),leftmargin=1.8cm]
    \item A raddoppia ogni ora: $200\cdot2^t$. B si dimezza ogni ora: $51\,200\cdot\left(\tfrac12\right)^t=51\,200\cdot2^{-t}$.
    \item $N_A(2)=200\cdot4=800$; \ $N_B(2)=\tfrac{51\,200}{4}=12\,800$.
    \item \[\begin{aligned}
    200\cdot2^t &= 51\,200\cdot2^{-t}\\
    2^{2t} &= \dfrac{51\,200}{200}=256=2^8 && \textcolor{gray}{\text{moltiplico per $2^t$ e divido per $200$}}\\
    t &= 4, \qquad N_A(4)=200\cdot16=3200
    \end{aligned}\]
    \item \[\begin{aligned}
    51\,200\left(\dfrac12\right)^t &< 100\\
    \left(\dfrac12\right)^t &< \dfrac1{512}=\left(\dfrac12\right)^9 \;\Rightarrow\; t>9
    \end{aligned}\]
    $N_A(9)=200\cdot512=102\,400$.
\end{enumerate}
\end{explanation}
\end{question}
