\begin{question}[ID=1.1, subtitle={Scomposizione di polinomi — serie 1}, tags={scomposizione, prodotti notevoli}, difficulty=facile]
Scomponi i seguenti polinomi:
\begin{enumerate}
  \item $6x^4y - 9x^3y^2 + 3x^2y$
  \item $3a^2b - 2a + 12ab - 8$
  \item $25x^2 -9$
  \item $x^2 - 6x  + 9$
  \color{gray}\item $9y^2 +\dfrac{1}{4}-3y$
  \color{gray}\item $8x^3 - 12x^2y + 6xy^2 - y^3$
  \color{black}\item $x^2 +4x  - 5$
  \color{gray}\item $2x^2 - 7x - 15$
  \color{black}\item $27x^3 - 125y^3$
  \color{gray}\item $x^3 + 8$
\end{enumerate}
\begin{solution}
\begin{enumerate}
    \item $3x^2y(2x^2-3xy+1)$
    \item $(a+4)(3ab-2)$
    \item $(5x-3)(5x+3)$
    \item $(x-3)^2$
    \item $\left(3y-\tfrac12\right)^2$
    \item $(2x-y)^3$
    \item $(x+5)(x-1)$
    \item $(2x+3)(x-5)$
    \item $(3x-5y)(9x^2+15xy+25y^2)$
    \item $(x+2)(x^2-2x+4)$
\end{enumerate}
\end{solution}
\begin{explanation}
Per scomporre si prova, nell'ordine: raccoglimento totale, raccoglimento parziale, prodotti notevoli (differenza di quadrati, quadrato e cubo di binomio, somma e differenza di cubi), trinomio di secondo grado.
\begin{enumerate}
    \item \textbf{Raccoglimento totale}: il M.C.D. dei coefficienti $6,9,3$ è $3$; le lettere comuni con il minimo esponente sono $x^2$ e $y$.
    \[ 6x^4y-9x^3y^2+3x^2y=3x^2y\,(2x^2-3xy+1) \]
    \item \textbf{Raccoglimento parziale}: riordino i termini e raccolgo a coppie ($3ab$ dai primi due, $-2$ dagli altri due):
    \[\begin{aligned}
    3a^2b+12ab-2a-8 &= 3ab(a+4)-2(a+4) && \textcolor{gray}{\text{raccolgo a coppie}}\\
    &= (a+4)(3ab-2) && \textcolor{gray}{\text{raccolgo il fattore comune $(a+4)$}}
    \end{aligned}\]
    \item \textbf{Differenza di quadrati}: $25x^2=(5x)^2$ e $9=3^2$.
    \[ (5x)^2-3^2=(5x-3)(5x+3) \]
    \item \textbf{Quadrato di binomio}: $x^2$ e $9=3^2$ sono quadrati e $-6x=-2\cdot x\cdot3$ è il doppio prodotto.
    \[ x^2-6x+9=(x-3)^2 \]
    \item \textbf{Quadrato di binomio}: riordino $9y^2-3y+\tfrac14$; $9y^2=(3y)^2$, $\tfrac14=\left(\tfrac12\right)^2$, $-3y=-2\cdot3y\cdot\tfrac12$.
    \[ 9y^2-3y+\dfrac14=\left(3y-\dfrac12\right)^2 \]
    \item \textbf{Cubo di binomio}: $8x^3=(2x)^3$, $-y^3=(-y)^3$; controllo i termini centrali: $3(2x)^2(-y)=-12x^2y$ e $3(2x)(-y)^2=6xy^2$.
    \[ 8x^3-12x^2y+6xy^2-y^3=(2x-y)^3 \]
    \item \textbf{Trinomio} $x^2+sx+p$: cerco due numeri con somma $4$ e prodotto $-5$: sono $5$ e $-1$.
    \[ x^2+4x-5=(x+5)(x-1) \]
    \item \textbf{Trinomio con primo coefficiente $\neq1$}: cerco due numeri con somma $-7$ e prodotto $2\cdot(-15)=-30$: sono $-10$ e $3$. Spezzo il termine centrale:
    \[\begin{aligned}
    2x^2-7x-15 &= 2x^2-10x+3x-15 && \textcolor{gray}{\text{$-7x=-10x+3x$}}\\
    &= 2x(x-5)+3(x-5) && \textcolor{gray}{\text{raccoglimento parziale}}\\
    &= (2x+3)(x-5)
    \end{aligned}\]
    \item \textbf{Differenza di cubi} con $A=3x$, $B=5y$: \ $A^3-B^3=(A-B)(A^2+AB+B^2)$.
    \[ (3x)^3-(5y)^3=(3x-5y)(9x^2+15xy+25y^2) \]
    \item \textbf{Somma di cubi} con $A=x$, $B=2$: \ $A^3+B^3=(A+B)(A^2-AB+B^2)$.
    \[ x^3+2^3=(x+2)(x^2-2x+4) \]
\end{enumerate}
\end{explanation}
\end{question}


\begin{question}[ID=1.2, subtitle={Scomposizione di polinomi — serie 2}, tags={scomposizione, prodotti notevoli}, difficulty=facile]
Scomponi i seguenti polinomi:
\begin{enumerate}
  \item $6a^2b + 9ab + 3a^3b$
  \item $3a^2 + 3ab + 5a + 5b$
  \item $x^2 - 16$
  \item $x^2 + 10x + 25$
\color{gray}  \item $8x^3 - 12x^2y + 6xy^2 - y^3$
\color{gray}  \item $2x^2 - 7x - 15$
\color{black}  \item $x^2+9x+14$
\color{gray}  \item $x^2 - 6x + 9$
\color{gray}  \item $27x^3 - 125y^3$
\color{black}  \item $x^3 + 8$
\end{enumerate}
\begin{solution}
\begin{enumerate}
    \item $3ab(a^2+2a+3)$
    \item $(3a+5)(a+b)$
    \item $(x-4)(x+4)$
    \item $(x+5)^2$
    \item $(2x-y)^3$
    \item $(2x+3)(x-5)$
    \item $(x+7)(x+2)$
    \item $(x-3)^2$
    \item $(3x-5y)(9x^2+15xy+25y^2)$
    \item $(x+2)(x^2-2x+4)$
\end{enumerate}
\end{solution}
\begin{explanation}
Per scomporre si prova, nell'ordine: raccoglimento totale, raccoglimento parziale, prodotti notevoli (differenza di quadrati, quadrato e cubo di binomio, somma e differenza di cubi), trinomio di secondo grado.
\begin{enumerate}
    \item \textbf{Raccoglimento totale}: M.C.D. dei coefficienti $6,9,3$ è $3$; lettere comuni $a$ e $b$ con esponente minimo $1$.
    \[ 6a^2b+9ab+3a^3b=3ab\,(2a+3+a^2)=3ab\,(a^2+2a+3) \]
    \item \textbf{Raccoglimento parziale}:
    \[\begin{aligned}
    3a^2+3ab+5a+5b &= 3a(a+b)+5(a+b) && \textcolor{gray}{\text{raccolgo $3a$ e $5$}}\\
    &= (3a+5)(a+b) && \textcolor{gray}{\text{raccolgo $(a+b)$}}
    \end{aligned}\]
    \item \textbf{Differenza di quadrati}: \ $x^2-4^2=(x-4)(x+4)$.
    \item \textbf{Quadrato di binomio}: $10x=2\cdot x\cdot5$ e $25=5^2$: \ $(x+5)^2$.
    \item \textbf{Cubo di binomio} con $A=2x$, $B=-y$ (come nella serie 1): \ $(2x-y)^3$.
    \item \textbf{Trinomio}: somma $-7$, prodotto $2\cdot(-15)=-30$: $-10$ e $3$.
    \[ 2x^2-10x+3x-15=2x(x-5)+3(x-5)=(2x+3)(x-5) \]
    \item \textbf{Trinomio}: due numeri con somma $9$ e prodotto $14$: $7$ e $2$. \ $(x+7)(x+2)$.
    \item \textbf{Quadrato di binomio}: $-6x=-2\cdot x\cdot3$, $9=3^2$: \ $(x-3)^2$.
    \item \textbf{Differenza di cubi} con $A=3x$, $B=5y$: \ $(3x-5y)(9x^2+15xy+25y^2)$.
    \item \textbf{Somma di cubi} con $A=x$, $B=2$: \ $(x+2)(x^2-2x+4)$.
\end{enumerate}
\end{explanation}
\end{question}
