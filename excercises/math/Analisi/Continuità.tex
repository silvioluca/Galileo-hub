\begin{question}[ID=1.1, subtitle={Discontinuità da grafico — serie 1}, tags={continuità, discontinuità, grafico}, difficulty=facile]
Determina se esistono punti di discontinuità nei grafici seguenti e indicane la tipologia.
\newline

\includegraphics[width=0.9\textwidth]{images/grafici_continuità}
\begin{center}
  fig. 1 \hspace{200pt} fig. 2
\end{center}
\begin{solution}
\textbf{Fig. 1:} in $x=0$ discontinuità \textbf{eliminabile}; \ in $x=1$ \textbf{salto} (1ª specie). \\
\textbf{Fig. 2:} in $x=-1$ discontinuità \textbf{eliminabile}; \ in $x=0$ \textbf{salto} (1ª specie); \ in $x=1$ discontinuità di \textbf{2ª specie}.
\end{solution}
\begin{explanation}
In un punto $x_0$ si confrontano il limite sinistro, il limite destro e il valore $f(x_0)$:
\begin{itemize}
    \item limiti uguali e finiti, ma valore mancante o diverso: discontinuità \textbf{eliminabile} (di 3ª specie);
    \item limiti finiti ma diversi: \textbf{salto} (1ª specie), di ampiezza $|\lim_{x\to x_0^+}f-\lim_{x\to x_0^-}f|$;
    \item almeno un limite infinito o inesistente: discontinuità di \textbf{2ª specie}.
\end{itemize}
\textbf{Fig. 1}
\begin{itemize}
    \item $x=0$: il grafico arriva da entrambi i lati allo stesso punto vuoto (i due limiti coincidono e sono finiti), ma il valore $f(0)$ è il punto pieno più in alto, diverso dal limite: \textbf{eliminabile}.
    \item $x=1$: da sinistra il grafico arriva a un punto vuoto in alto, da destra riparte dal punto pieno sull'asse $x$. I limiti sono finiti ma diversi: \textbf{salto}.
\end{itemize}
\textbf{Fig. 2}
\begin{itemize}
    \item $x=-1$: la curva passa per un punto vuoto (limite destro e sinistro uguali), mentre $f(-1)$ è il punto pieno sull'asse $x$: \textbf{eliminabile}.
    \item $x=0$: da sinistra il grafico arriva a un punto vuoto in basso, da destra parte da un punto pieno più in alto: limiti finiti diversi, \textbf{salto}.
    \item $x=1$: da sinistra il limite è finito (punto vuoto), ma da destra la curva scende verso $-\infty$ lungo l'asintoto verticale tratteggiato: \textbf{2ª specie}.
\end{itemize}
\end{explanation}
\end{question}

\begin{question}[ID=2.1, subtitle={Punti singolari — serie 1}, tags={continuità, punti singolari}, difficulty=media]
Individua e classifica gli eventuali punti singolari delle seguenti funzioni:
\begin{center}\begin{minipage}{0.92\linewidth}
\begin{multicols}{2}
\begin{enumerate}[label=\arabic*),leftmargin=3.2cm]
\item $f(x) = \frac{x^2-4}{x+2}$
\item $f(x) = \frac{\sqrt{x+2}-2}{x-2}$
\item $f(x) = \frac{|x|}{x}$
\item $f(x) = \frac{(x-2)^2}{|x-2|}$
\item $f(x) = \frac{1}{e^x-1}$
\item $f(x) = \frac{1}{1-\cos x}$
\item $f(x) = \begin{cases} x - 2 \quad & x \ge 2 \\ \frac{2x-4}{x-3} \quad & x < 2 \end{cases}$
\item $f(x) = \begin{cases} \frac{\sin x}{x} & x < 0 \\ e^{1/\ln x} & x > 0 \end{cases}$
\end{enumerate}
\end{multicols}
\end{minipage}\end{center}
\begin{solution}
\begin{enumerate}[label=\arabic*)]
    \item eliminabile in $x=-2$
    \item eliminabile in $x=2$
    \item salto in $x=0$ (ampiezza $2$)
    \item eliminabile in $x=2$
    \item 2ª specie in $x=0$
    \item 2ª specie in $x=2k\pi$
    \item nessun punto singolare (continua)
    \item eliminabile in $x=0$; \ 2ª specie in $x=1$
\end{enumerate}
\end{solution}
\begin{explanation}
In un punto $x_0$ si confrontano il limite sinistro, il limite destro e il valore $f(x_0)$:
\begin{itemize}
    \item limiti uguali e finiti, ma valore mancante o diverso: discontinuità \textbf{eliminabile} (di 3ª specie);
    \item limiti finiti ma diversi: \textbf{salto} (1ª specie), di ampiezza $|\lim_{x\to x_0^+}f-\lim_{x\to x_0^-}f|$;
    \item almeno un limite infinito o inesistente: discontinuità di \textbf{2ª specie}.
\end{itemize}
\begin{enumerate}[label=\arabic*)]
    \item Il punto da studiare è $x=-2$ (denominatore nullo). Scompongo e semplifico:
    \[ \lim_{x\to-2}\dfrac{(x-2)(x+2)}{x+2}=\lim_{x\to-2}(x-2)=-4 \]
    Limite finito ma funzione non definita: eliminabile.
    \item In $x=2$ si ha la forma $\tfrac00$. Razionalizzo moltiplicando per $\sqrt{x+2}+2$:
    \[\begin{aligned}
    \dfrac{\sqrt{x+2}-2}{x-2} &= \dfrac{(x+2)-4}{(x-2)(\sqrt{x+2}+2)} && \textcolor{gray}{\text{differenza di quadrati al numeratore}}\\
    &= \dfrac{1}{\sqrt{x+2}+2}\;\to\;\dfrac14
    \end{aligned}\]
    Limite finito: eliminabile.
    \item Per $x>0$ $|x|=x$ e $f=1$; per $x<0$ $|x|=-x$ e $f=-1$. Limiti destro $1$ e sinistro $-1$: salto di ampiezza $2$.
    \item $(x-2)^2=|x-2|^2$, quindi $f(x)=|x-2|$ per $x\neq2$, che tende a $0$: eliminabile.
    \item $e^x-1=0$ per $x=0$. Il numeratore vale $1$: per $x\to0^+$ il denominatore tende a $0^+$ e $f\to+\infty$; per $x\to0^-$ tende a $0^-$ e $f\to-\infty$. Limiti infiniti: 2ª specie.
    \item $1-\cos x=0$ quando $\cos x=1$, cioè $x=2k\pi$. Lì il denominatore tende a $0^+$ e $f\to+\infty$: 2ª specie in ogni $x=2k\pi$.
    \item L'unico punto da controllare è il raccordo $x=2$ (il ramo sinistro non si annulla al denominatore per $x<2$):
    \[ \lim_{x\to2^-}\dfrac{2x-4}{x-3}=\dfrac{0}{-1}=0, \qquad f(2)=2-2=0 \]
    Coincidono: la funzione è continua.
    \item Due punti da studiare.
    \begin{itemize}
        \item $x=0$ (non appartiene a nessun ramo): da sinistra $\dfrac{\sin x}{x}\to1$ (limite notevole); da destra $\ln x\to-\infty$, quindi $\dfrac{1}{\ln x}\to0$ e $e^{1/\ln x}\to e^0=1$. Limiti uguali e finiti: eliminabile.
        \item $x=1$ (lì $\ln x=0$): per $x\to1^+$ si ha $\dfrac{1}{\ln x}\to+\infty$ e $f\to+\infty$: 2ª specie.
    \end{itemize}
\end{enumerate}
\end{explanation}
\end{question}

    \item nessun punto singolare (continua)
    \item nessun punto singolare (continua)
\end{enumerate}
\end{solution}
\begin{explanation}
In un punto $x_0$ si confrontano il limite sinistro, il limite destro e il valore $f(x_0)$:
\begin{itemize}
    \item limiti uguali e finiti, ma valore mancante o diverso: discontinuità \textbf{eliminabile} (di 3ª specie);
    \item limiti finiti ma diversi: \textbf{salto} (1ª specie), di ampiezza $|\lim_{x\to x_0^+}f-\lim_{x\to x_0^-}f|$;
    \item almeno un limite infinito o inesistente: discontinuità di \textbf{2ª specie}.
\end{itemize}
\begin{enumerate}[label=\arabic*)]
    \item In $x=1$ il denominatore si annulla, il numeratore vale $2\neq0$: $f\to\pm\infty$, 2ª specie.
    \item In $x=0$ forma $\tfrac00$; è il limite notevole $\displaystyle\lim_{x\to0}\dfrac{\ln(1+x)}{x}=1$: eliminabile.

\end{enumerate}
\end{explanation}

\begin{question}[ID=2.5, subtitle={Punti singolari — serie 5}, tags={continuità, punti singolari}, difficulty=media]
Individua e classifica gli eventuali punti singolari delle seguenti funzioni:
\begin{center}\begin{minipage}{0.92\linewidth}
\begin{multicols}{2}
\begin{enumerate}[label=\arabic*),leftmargin=3.2cm]
\item $f(x) = \frac{x^2 + 4x + 4}{x^2 - 4}$
\item $f(x) = \frac{\sqrt{x^2 - 1}}{x-1}$
\item $f(x) = \frac{|x-1|}{x^2 - 1}$
\item $f(x) = \frac{x+2}{|x+2|}$
\item $f(x) = \frac{1}{e^x+1}$
\item $f(x) = \frac{1}{\sin x - x}$
\item $f(x) = \begin{cases} x^2 \quad & x \geq 2 \\ 2x - 1 \quad & x < 2 \end{cases}$
\item $f(x) = \begin{cases} e^{-x} & x > 0 \\ x + 3 & x \le 0 \end{cases}$
\end{enumerate}
\end{multicols}
\end{minipage}\end{center}
\begin{solution}
\begin{enumerate}[label=\arabic*)]
    \item eliminabile in $x=-2$; \ 2ª specie in $x=2$
    \item 2ª specie in $x=1$
    \item salto in $x=1$ (ampiezza $1$); \ 2ª specie in $x=-1$
    \item salto in $x=-2$ (ampiezza $2$)
    \item nessun punto singolare
    \item 2ª specie in $x=0$
    \item salto in $x=2$ (ampiezza $1$)
    \item salto in $x=0$ (ampiezza $2$)
\end{enumerate}
\end{solution}
\begin{explanation}
In un punto $x_0$ si confrontano il limite sinistro, il limite destro e il valore $f(x_0)$:
\begin{itemize}
    \item limiti uguali e finiti, ma valore mancante o diverso: discontinuità \textbf{eliminabile} (di 3ª specie);
    \item limiti finiti ma diversi: \textbf{salto} (1ª specie), di ampiezza $|\lim_{x\to x_0^+}f-\lim_{x\to x_0^-}f|$;
    \item almeno un limite infinito o inesistente: discontinuità di \textbf{2ª specie}.
\end{itemize}
\begin{enumerate}[label=\arabic*)]
    \item Il numeratore è $(x+2)^2$, il denominatore $(x-2)(x+2)$. Semplifico: $\dfrac{x+2}{x-2}$ per $x\neq-2$. \\
    In $x=-2$ il limite è $\dfrac{0}{-4}=0$: eliminabile. In $x=2$ il numeratore vale $4\neq0$: 2ª specie.
    \item Dominio $x\leq-1 \vee x>1$. Per $x\to1^+$:
    \[ \dfrac{\sqrt{(x-1)(x+1)}}{x-1}=\dfrac{\sqrt{x+1}}{\sqrt{x-1}}\to+\infty \]
    2ª specie.
    \item Scrivo il denominatore come $(x-1)(x+1)$:
    \[\begin{aligned}
    x\to1^+: &\quad \dfrac{x-1}{(x-1)(x+1)}=\dfrac{1}{x+1}\to\dfrac12\\
    x\to1^-: &\quad \dfrac{-(x-1)}{(x-1)(x+1)}=-\dfrac{1}{x+1}\to-\dfrac12
    \end{aligned}\]

Asintoto verticale $x=1$; \ asintoto obliquo $y=x+1$; \ nessun asintoto orizzontale.
\end{solution}
\begin{explanation}
\textbf{Asintoti verticali.} Il denominatore si annulla in $x=1$, dove il numeratore vale $2\neq0$: $f\to\pm\infty$, quindi $x=1$ è asintoto verticale.

\textbf{Asintoti orizzontali.} Il grado del numeratore ($2$) supera quello del denominatore ($1$): $f\to\pm\infty$ all'infinito, nessun asintoto orizzontale.

\textbf{Asintoto obliquo.} I gradi differiscono di $1$: c'è un asintoto obliquo. Divido i polinomi:
\[\begin{aligned}
x^2+1 &= (x-1)(x+1)+2\\
\dfrac{x^2+1}{x-1} &= x+1+\dfrac{2}{x-1}
\end{aligned}\]
Il termine $\dfrac{2}{x-1}$ tende a $0$ all'infinito: l'asintoto è $y=x+1$.
\end{explanation}
\end{question}

\begin{question}[ID=5.2, subtitle={Asintoti di $f(x)=(x^2-4)/(x+2)$}, tags={continuità, asintoti}, difficulty=facile]
Studia la seguente funzione ponendo particolare attenzione a eventuali asintoti:
\begin{equation*}
    f(x) = \frac{x^2 - 4}{x+2}
\end{equation*}
\begin{solution}
Nessun asintoto; \ discontinuità eliminabile in $x=-2$ (il grafico è la retta $y=x-2$ privata del punto $(-2,-4)$).
\end{solution}
\begin{explanation}
Scompongo il numeratore: $x^2-4=(x-2)(x+2)$. Per $x\neq-2$:
\[ f(x)=\dfrac{(x-2)(x+2)}{x+2}=x-2 \]
In $x=-2$ il limite vale $-4$, finito: non c'è asintoto verticale ma un \emph{buco} (discontinuità eliminabile). \\
Il grafico è una retta, quindi non ci sono asintoti orizzontali o obliqui (la retta coincide con se stessa).
\end{explanation}
\end{question}

\begin{question}[ID=5.4, subtitle={Asintoti di $f(x)=(x^3-2x)/(x^2+1)$}, tags={continuità, asintoti}, difficulty=media]
Studia la funzione seguente, soffermandoti su eventuali asintoti e sul comportamento per $x \to \pm\infty$:
\begin{equation*}
    f(x) = \frac{x^3 - 2x}{x^2+1}
\end{equation*}
\begin{solution}
Nessun asintoto verticale né orizzontale; \ asintoto obliquo $y=x$.
\end{solution}
\begin{explanation}
\textbf{Verticali.} $x^2+1>0$ sempre: nessun asintoto verticale.

\textbf{Orizzontale.} Grado del numeratore $3$, del denominatore $2$: $f\to\pm\infty$, nessun asintoto orizzontale.

\textbf{Obliquo.} I gradi differiscono di $1$. Divido:
\[\begin{aligned}
x^3-2x &= x(x^2+1)-3x\\
f(x) &= x-\dfrac{3x}{x^2+1}
\end{aligned}\]
$\dfrac{3x}{x^2+1}\to0$ all'infinito (grado minore al numeratore): l'asintoto è $y=x$.
\end{explanation}
\end{question}

\begin{question}[ID=5.5, subtitle={Asintoti di $f(x)=(x^2+x+1)/(x^2-2x)$}, tags={continuità, asintoti}, difficulty=media]
Determina il dominio della funzione e analizza eventuali asintoti:
\begin{equation*}
    f(x) = \frac{x^2 + x + 1}{x^2 - 2x}
\end{equation*}
\begin{solution}
Dominio $x\neq0,\ x\neq2$; \ asintoti verticali $x=0$ e $x=2$; \ asintoto orizzontale $y=1$.
\end{solution}
\begin{explanation}
\textbf{Dominio.} $x^2-2x=x(x-2)\neq0$: \ $x\neq0$ e $x\neq2$.

\textbf{Verticali.} Il numeratore vale $1$ in $x=0$ e $7$ in $x=2$ (non nullo): entrambi asintoti verticali. \\
(Il numeratore $x^2+x+1$ ha $\Delta<0$ e non si annulla mai, quindi non ci sono semplificazioni.)

\textbf{Orizzontale.} Numeratore e denominatore hanno lo stesso grado: il limite all'infinito è il rapporto dei coefficienti direttivi:
\[ \lim_{x\to\pm\infty}\dfrac{x^2+x+1}{x^2-2x}=\dfrac11=1 \]
\end{explanation}
\end{question}

\begin{question}[ID=5.6, subtitle={Asintoti di $f(x)=\sqrt{x^2+1}/(x-3)$}, tags={continuità, asintoti}, difficulty=media]
Studia la funzione seguente, con particolare attenzione agli asintoti verticali e orizzontali:
\begin{equation*}
    f(x) = \frac{\sqrt{x^2+1}}{x-3}
\end{equation*}
\begin{solution}
Asintoto verticale $x=3$; \ asintoti orizzontali $y=1$ (per $x\to+\infty$) e $y=-1$ (per $x\to-\infty$).
\end{solution}
\begin{explanation}
\textbf{Dominio.} Il radicando $x^2+1$ è sempre positivo; resta $x\neq3$.

\textbf{Verticale.} In $x=3$ il numeratore vale $\sqrt{10}\neq0$: asintoto verticale.

\textbf{Orizzontali.} Raccolgo $x^2$ sotto radice, ricordando che $\sqrt{x^2}=|x|$:
\[\begin{aligned}
\dfrac{\sqrt{x^2+1}}{x-3} &= \dfrac{|x|\sqrt{1+\frac1{x^2}}}{x\left(1-\frac3x\right)}\\
x\to+\infty: &\quad |x|=x \;\Rightarrow\; f\to1\\
x\to-\infty: &\quad |x|=-x \;\Rightarrow\; f\to-1
\end{aligned}\]
Due asintoti orizzontali diversi: $y=1$ a destra e $y=-1$ a sinistra.
\end{explanation}
\end{question}
